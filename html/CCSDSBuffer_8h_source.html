26 [[nodiscard]] inline Result<std::size_t> declaredPacketSize(
27 const std::uint8_t *data, const std::size_t size) {
28 RET_IF_ERR_MSG(data == nullptr, ErrorCode::NULL_POINTER,
29 "Cannot inspect packet size: raw buffer pointer is null.");
30 RET_IF_ERR_MSG(size < 6U, ErrorCode::INVALID_HEADER_DATA,
31 "Cannot inspect packet size: at least six primary-header bytes are required.");
32 const auto version = static_cast<std::uint8_t>((data[0] >> 5U) & 0x07U);
33 RET_IF_ERR_MSG(version != 0U, ErrorCode::INVALID_HEADER_DATA,
34 "Cannot inspect packet size: unsupported CCSDS packet version.");
35 const auto encodedLength = static_cast<std::uint16_t>(
36 (static_cast<std::uint16_t>(data[4]) << 8U) | static_cast<std::uint16_t>(data[5]));
37 return 6U + static_cast<std::size_t>(encodedLength) + 1U;
38 }
