29 [[nodiscard]] inline Result<std::size_t> declaredPacketSize(
30 const std::uint8_t *data, const std::size_t size) {
31 RET_IF_ERR_MSG(data == nullptr, ErrorCode::NULL_POINTER,
32 "Cannot inspect packet size: raw buffer pointer is null.");
33 RET_IF_ERR_MSG(size < CCSDS_PRIMARY_HEADER_SIZE, ErrorCode::INVALID_HEADER_DATA,
34 "Cannot inspect packet size: at least six primary-header bytes are required.");
35
36 std::size_t declaredSize = 0U;
37 const auto status = ccsds_packet_declared_size(data, size, &declaredSize);
38 RET_IF_ERR_MSG(status != CCSDS_STATUS_OK, ErrorCode::INVALID_HEADER_DATA,
39 "Cannot inspect packet size: unsupported CCSDS packet version.");
40 return declaredSize;
41 }
