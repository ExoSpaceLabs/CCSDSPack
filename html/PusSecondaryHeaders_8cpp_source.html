9 ccsds::Result<std::uint64_t> requiredUnsigned(const ccsds::Config &config,
10 const char *key,
11 const std::uint64_t maximum) {
12 RET_IF_ERR_MSG(!config.isKey(key), ccsds::ErrorCode::CONFIG_FILE_ERROR,
13 std::string{"Config: Missing uint field: "} + key);
14 const auto unsignedResult = config.get<std::uint64_t>(key);
15 if (unsignedResult) {
16 RET_IF_ERR_MSG(unsignedResult.value() > maximum,
17 ccsds::ErrorCode::CONFIG_FILE_ERROR,
18 std::string{"Config: Out-of-range uint field: "} + key);
19 return unsignedResult.value();
20 }
21 const auto intResult = config.get<int>(key);
22 RET_IF_ERR_MSG(!intResult || intResult.value() < 0
23 || static_cast<std::uint64_t>(intResult.value()) > maximum,
24 ccsds::ErrorCode::CONFIG_FILE_ERROR,
25 std::string{"Config: Invalid unsigned field: "} + key);
26 return static_cast<std::uint64_t>(intResult.value());
27 }
28
29 ccsds::Result<std::uint64_t> optionalUnsigned(const ccsds::Config &config,
30 const char *key,
31 const std::uint64_t maximum,
32 const std::uint64_t fallback) {
33 if (!config.isKey(key)) return fallback;
34 return requiredUnsigned(config, key, maximum);
35 }

55bool ccsds::pus::SecondaryHeader::identifierFits(const std::uint32_t value,
56 const std::uint8_t octets) const {
57 if (octets == 0U) return value == 0U;
58 if (octets >= 4U) return true;
59 return value < (1UL << (octets * 8U));
60}

62void ccsds::pus::SecondaryHeader::appendIdentifier(std::vector<std::uint8_t> &bytes,
63 const std::uint32_t value,
64 const std::uint8_t octets) {
65 for (std::uint8_t i = 0; i < octets; ++i) {
66 const auto shift = static_cast<std::uint8_t>((octets - i - 1U) * 8U);
67 bytes.push_back(static_cast<std::uint8_t>((value >> shift) & 0xFFU));
68 }
69}

71std::uint32_t ccsds::pus::SecondaryHeader::readIdentifier(
72 const std::vector<std::uint8_t> &bytes, const std::size_t offset,
73 const std::uint8_t octets) {
74 std::uint32_t result = 0U;
75 for (std::uint8_t i = 0; i < octets; ++i) {
76 result = static_cast<std::uint32_t>((result << 8U) | bytes[offset + i]);
77 }
78 return result;
79}

81bool ccsds::pus::SecondaryHeader::trailingSpareIsZero(
82 const std::vector<std::uint8_t> &bytes) const {
83 const auto spare = static_cast<std::size_t>(m_secondaryHeaderSpareOctets);
84 if (spare > bytes.size()) return false;
85 for (std::size_t i = bytes.size() - spare; i < bytes.size(); ++i) {
86 if (bytes[i] != 0U) return false;
87 }
88 return true;
89}

156void ccsds::pus::TcSecondaryHeader::appendTcBody(std::vector<std::uint8_t> &bytes) const {
157 bytes.push_back(m_serviceType);
158 bytes.push_back(m_serviceSubtype);
159 appendIdentifier(bytes, m_sourceId, m_sourceIdOctets);
160 bytes.insert(bytes.end(), m_secondaryHeaderSpareOctets, 0U);
161}

163ccsds::ResultBool ccsds::pus::TcSecondaryHeader::parseTcBody(
164 const std::vector<std::uint8_t> &data, const std::size_t offset) {
165 RET_IF_ERR_MSG(data.size() != tcSize(), ErrorCode::INVALID_SECONDARY_HEADER_DATA,
166 "PUS TC secondary-header size does not match its tailoring.");
167 RET_IF_ERR_MSG(!trailingSpareIsZero(data), ErrorCode::INVALID_SECONDARY_HEADER_DATA,
168 "PUS TC secondary-header spare octets must be zero.");
169 const auto source = readIdentifier(data, offset + 2U, m_sourceIdOctets);
170 m_serviceType = data[offset];
171 m_serviceSubtype = data[offset + 1U];
172 m_sourceId = source;
173 return true;
174}

209ccsds::ResultBool ccsds::pus::TmSecondaryHeader::loadFromConfig(
210 const ccsds::Config &config) {
211 FORWARD_RESULT(SecondaryHeader::loadFromConfig(config));
212
213 if (getRevision() == Revision::A) {
214 std::uint64_t width{};
215 ASSIGN_CP(width, optionalUnsigned(config, "pus_destination_id_octets", 4U,
216 m_destinationIdOctets));
217 RET_IF_ERR_MSG(!validIdentifierWidth(static_cast<std::uint8_t>(width)),
218 ErrorCode::CONFIG_FILE_ERROR,
219 "Config: pus_destination_id_octets must be 0, 1, 2, or 4.");
220 m_destinationIdOctets = static_cast<std::uint8_t>(width);
221 } else if (config.isKey("pus_destination_id_octets")) {
222 std::uint64_t width{};
223 ASSIGN_CP(width, requiredUnsigned(config, "pus_destination_id_octets", 4U));
224 RET_IF_ERR_MSG(width != 2U, ErrorCode::CONFIG_FILE_ERROR,
225 "Config: PUS-C TM destination ID is fixed at two octets.");
226 }
227
228 if (config.isKey("pus_time_format")) {
229 const auto format = config.get<std::string>("pus_time_format");
230 RET_IF_ERR_MSG(!format, ErrorCode::CONFIG_FILE_ERROR,
231 "Config: invalid pus_time_format.");
232 if (format.value() == "none") {
233 m_timestampPresent = false;
234 m_cuc = {};
235 RET_IF_ERR_MSG(config.isKey("pus_time_epoch") || config.isKey("pus_time_p_field")
236 || config.isKey("pus_time_coarse_octets")
237 || config.isKey("pus_time_fine_octets")
238 || config.isKey("pus_time_coarse") || config.isKey("pus_time_fine"),
239 ErrorCode::CONFIG_FILE_ERROR,
240 "Config: disabled PUS time cannot contain CUC fields.");
241 } else if (format.value() == "cuc") {
242 m_timestampPresent = true;
243 RET_IF_ERR_MSG(!config.isKey("pus_time_epoch") || !config.isKey("pus_time_p_field")
244 || !config.isKey("pus_time_coarse_octets")
245 || !config.isKey("pus_time_fine_octets"),
246 ErrorCode::CONFIG_FILE_ERROR,
247 "Config: CUC tailoring requires epoch, P-field, coarse, and fine width fields.");
248 const auto epoch = config.get<std::string>("pus_time_epoch");
249 const auto pField = config.get<std::string>("pus_time_p_field");
250 RET_IF_ERR_MSG(!epoch || !pField, ErrorCode::CONFIG_FILE_ERROR,
251 "Config: invalid PUS CUC string field.");
252 m_cuc.epoch = epoch.value() == "ccsds-1958-tai" ? time::Epoch::Ccsds1958Tai
253 : epoch.value() == "agency-defined" ? time::Epoch::AgencyDefined
254 : time::Epoch::Unspecified;
255 m_cuc.pField = pField.value() == "implicit" ? time::PFieldMode::Implicit
256 : pField.value() == "explicit" ? time::PFieldMode::Explicit
257 : static_cast<time::PFieldMode>(0xFFU);
258 std::uint64_t coarse{};
259 std::uint64_t fine{};
260 ASSIGN_CP(coarse, requiredUnsigned(config, "pus_time_coarse_octets", 4U));
261 ASSIGN_CP(fine, requiredUnsigned(config, "pus_time_fine_octets", 3U));
262 RET_IF_ERR_MSG(coarse < 1U, ErrorCode::CONFIG_FILE_ERROR,
263 "Config: pus_time_coarse_octets must be 1..4.");
264 m_cuc.coarseOctets = static_cast<std::uint8_t>(coarse);
265 m_cuc.fineOctets = static_cast<std::uint8_t>(fine);
266 FORWARD_RESULT(time::validate(m_cuc));
267 } else {
268 return Error{ErrorCode::CONFIG_FILE_ERROR,
269 "Config: pus_time_format must be 'none' or 'cuc'."};
270 }
271 }
272
273 std::uint64_t serviceType{};
274 std::uint64_t serviceSubtype{};
275 std::uint64_t destinationId{};
276 ASSIGN_CP(serviceType, requiredUnsigned(config, "pus_service_type", UINT8_MAX));
277 ASSIGN_CP(serviceSubtype, requiredUnsigned(config, "pus_service_subtype", UINT8_MAX));
278 ASSIGN_CP(destinationId, requiredUnsigned(config, "pus_destination_id", UINT32_MAX));
279 m_serviceType = static_cast<std::uint8_t>(serviceType);
280 m_serviceSubtype = static_cast<std::uint8_t>(serviceSubtype);
281 FORWARD_RESULT(setDestinationId(static_cast<std::uint32_t>(destinationId)));
282
283 if (m_timestampPresent) {
284 time::CucTime timestamp;
285 ASSIGN_CP(timestamp.coarse, requiredUnsigned(config, "pus_time_coarse", UINT32_MAX));
286 ASSIGN_CP(timestamp.fine, requiredUnsigned(config, "pus_time_fine", 0xFFFFFFU));
287 FORWARD_RESULT(setTimestamp(timestamp));
288 } else {
289 m_timestamp = {};
290 }
291 return true;
292}

295ccsds::ResultBool ccsds::pus::TmSecondaryHeader::appendTmTail(
296 std::vector<std::uint8_t> &bytes) const {
297 appendIdentifier(bytes, m_destinationId, m_destinationIdOctets);
298 if (m_timestampPresent) {
299 std::vector<std::uint8_t> encoded;
300 ASSIGN_MV(encoded, time::serialize(m_timestamp, m_cuc));
301 bytes.insert(bytes.end(), encoded.begin(), encoded.end());
302 }
303 bytes.insert(bytes.end(), m_secondaryHeaderSpareOctets, 0U);
304 return true;
305}

307ccsds::ResultBool ccsds::pus::TmSecondaryHeader::parseTmTail(
308 const std::vector<std::uint8_t> &data, const std::size_t offset) {
309 const auto tail = tmTailSize();
310 RET_IF_ERR_MSG(offset + tail != data.size(), ErrorCode::INVALID_SECONDARY_HEADER_DATA,
311 "PUS TM secondary-header size does not match its tailoring.");
312 RET_IF_ERR_MSG(!trailingSpareIsZero(data), ErrorCode::INVALID_SECONDARY_HEADER_DATA,
313 "PUS TM secondary-header spare octets must be zero.");
314 const auto destination = readIdentifier(data, offset, m_destinationIdOctets);
315 const auto timestampOffset = offset + m_destinationIdOctets;
316 const auto timestampSize = m_timestampPresent ? time::encodedSize(m_cuc) : 0U;
317 const auto timestampEnd = timestampOffset + timestampSize;
318 m_destinationId = destination;
319 if (m_timestampPresent) {
320 const std::vector<std::uint8_t> encoded(
321 data.begin() + static_cast<std::ptrdiff_t>(timestampOffset),
322 data.begin() + static_cast<std::ptrdiff_t>(timestampEnd));
323 ASSIGN_CP(m_timestamp, time::deserialize(encoded, m_cuc));
324 } else {
325 m_timestamp = {};
326 }
327 return true;
328}

330std::uint16_t ccsds::pus::TmSecondaryHeader::tmTailSize() const {
331 return static_cast<std::uint16_t>(m_destinationIdOctets
332 + (m_timestampPresent ? time::encodedSize(m_cuc) : 0U)
333 + m_secondaryHeaderSpareOctets);
334}

352ccsds::ResultBool ccsds::pus::rev_a::TcHeader::deserialize(
353 const std::vector<std::uint8_t> &data) {
354 FORWARD_RESULT(validateTailoring(getTailoring()));
355 RET_IF_ERR_MSG(data.size() != getSize(), ErrorCode::INVALID_SECONDARY_HEADER_DATA,
356 "PUS-A TC secondary-header size mismatch.");
357 RET_IF_ERR_MSG((data[0] & 0x80U) != 0U || ((data[0] >> 4U) & 0x07U) != 1U,
358 ErrorCode::INVALID_SECONDARY_HEADER_DATA,
359 "PUS-A TC reserved bit or version is invalid.");
360 FORWARD_RESULT(parseTcBody(data, 1U));
361 m_acknowledgementFlags = data[0] & 0x0FU;
362 return true;
363}

398ccsds::ResultBool ccsds::pus::rev_a::TmHeader::deserialize(
399 const std::vector<std::uint8_t> &data) {
400 FORWARD_RESULT(validateTailoring(getTailoring()));
401 RET_IF_ERR_MSG(data.size() != getSize(), ErrorCode::INVALID_SECONDARY_HEADER_DATA,
402 "PUS-A TM secondary-header size mismatch.");
403 RET_IF_ERR_MSG(data[0] != 0x10U, ErrorCode::INVALID_SECONDARY_HEADER_DATA,
404 "PUS-A TM reserved bits or version are invalid.");
405 const auto subcounterOffset = 3U;
406 const auto tailOffset = subcounterOffset + (m_packetSubcounterPresent ? 1U : 0U);
407 FORWARD_RESULT(parseTmTail(data, tailOffset));
408 m_serviceType = data[1];
409 m_serviceSubtype = data[2];
410 m_packetSubcounter = m_packetSubcounterPresent ? data[subcounterOffset] : 0U;
411 return true;
412}

418 if (m_packetSubcounterPresent) bytes.push_back(m_packetSubcounter);

515ccsds::ResultBool ccsds::pus::rev_c::TmHeader::deserialize(
516 const std::vector<std::uint8_t> &data) {
517 FORWARD_RESULT(validateTailoring(getTailoring()));
518 RET_IF_ERR_MSG(data.size() != getSize(), ErrorCode::INVALID_SECONDARY_HEADER_DATA,
519 "PUS-C TM secondary-header size mismatch.");
520 RET_IF_ERR_MSG((data[0] >> 4U) != 2U, ErrorCode::INVALID_SECONDARY_HEADER_DATA,
521 "PUS-C TM version is invalid.");
522 FORWARD_RESULT(parseTmTail(data, 5U));
523 m_timeReferenceStatus = data[0] & 0x0FU;
524 m_serviceType = data[1];
525 m_serviceSubtype = data[2];
526 m_messageTypeCounter = static_cast<std::uint16_t>(
527 (static_cast<std::uint16_t>(data[3]) << 8U) | data[4]);
528 return true;
529}

531std::vector<std::uint8_t> ccsds::pus::rev_c::TmHeader::serialize() const {
532 if (!validateTailoring(getTailoring()) || m_timeReferenceStatus > 0x0FU
533 || !identifierFits(m_destinationId, 2U)) return {};
534 std::vector<std::uint8_t> bytes{
535 static_cast<std::uint8_t>(0x20U | m_timeReferenceStatus),
536 m_serviceType, m_serviceSubtype,
537 static_cast<std::uint8_t>(m_messageTypeCounter >> 8U),
538 static_cast<std::uint8_t>(m_messageTypeCounter & 0xFFU)
539 };
540 if (!appendTmTail(bytes)) return {};
541 return bytes;
542}
