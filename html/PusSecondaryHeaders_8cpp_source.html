10 ccsds::Error pusTcDecodeError(const ccsds_status_t status, const bool revisionA) {
11 using ccsds::Error;
12 using ccsds::ErrorCode;
13 switch (status) {
14 case CCSDS_STATUS_PUS_INVALID_IDENTIFIER_WIDTH:
15 return Error(ErrorCode::INVALID_SECONDARY_HEADER_DATA,
16 "PUS-A TC source-ID width must be 0, 1, 2, or 4 octets.");
17 case CCSDS_STATUS_PUS_SIZE_MISMATCH:
18 return Error(ErrorCode::INVALID_SECONDARY_HEADER_DATA,
19 revisionA ? "PUS-A TC secondary-header size mismatch."
20 : "PUS-C TC secondary-header size mismatch.");
21 case CCSDS_STATUS_PUS_INVALID_VERSION:
22 return Error(ErrorCode::INVALID_SECONDARY_HEADER_DATA,
23 revisionA ? "PUS-A TC reserved bit or version is invalid."
24 : "PUS-C TC version is invalid.");
25 case CCSDS_STATUS_PUS_NONZERO_SPARE:
26 return Error(ErrorCode::INVALID_SECONDARY_HEADER_DATA,
27 "PUS TC secondary-header spare octets must be zero.");
28 case CCSDS_STATUS_NULL_POINTER:
29 return Error(ErrorCode::NULL_POINTER, "PUS TC core received a null pointer.");
30 default:
31 return Error(ErrorCode::INVALID_SECONDARY_HEADER_DATA,
32 "PUS TC secondary-header decoding failed.");
33 }
34 }

39 switch (status) {
40 case CCSDS_STATUS_PUS_SIZE_MISMATCH:
41 return Error(ErrorCode::INVALID_SECONDARY_HEADER_DATA,

43 : "PUS-C TM secondary-header size mismatch.");

45 return Error(ErrorCode::INVALID_SECONDARY_HEADER_DATA,
46 revisionA ? "PUS-A TM reserved bits or version are invalid."
47 : "PUS-C TM version is invalid.");
48 case CCSDS_STATUS_PUS_NONZERO_SPARE:
49 return Error(ErrorCode::INVALID_SECONDARY_HEADER_DATA,
50 "PUS TM secondary-header spare octets must be zero.");
51 case CCSDS_STATUS_CUC_PFIELD_MISMATCH:

59 default:
60 return Error(ErrorCode::INVALID_SECONDARY_HEADER_DATA,
61 "PUS TM secondary-header decoding failed.");
62 }

66 ccsds::Result<std::uint64_t> requiredUnsigned(const ccsds::Config &config,
67 const char *key,
68 const std::uint64_t maximum) {
69 RET_IF_ERR_MSG(!config.isKey(key), ccsds::ErrorCode::CONFIG_FILE_ERROR,
70 std::string{"Config: Missing uint field: "} + key);
71 const auto unsignedResult = config.get<std::uint64_t>(key);
72 if (unsignedResult) {
73 RET_IF_ERR_MSG(unsignedResult.value() > maximum,
74 ccsds::ErrorCode::CONFIG_FILE_ERROR,
75 std::string{"Config: Out-of-range uint field: "} + key);
76 return unsignedResult.value();
77 }
78 const auto intResult = config.get<int>(key);
79 RET_IF_ERR_MSG(!intResult || intResult.value() < 0
80 || static_cast<std::uint64_t>(intResult.value()) > maximum,
81 ccsds::ErrorCode::CONFIG_FILE_ERROR,
82 std::string{"Config: Invalid unsigned field: "} + key);
83 return static_cast<std::uint64_t>(intResult.value());
84 }
85
86 ccsds::Result<std::uint64_t> optionalUnsigned(const ccsds::Config &config,
87 const char *key,
88 const std::uint64_t maximum,
89 const std::uint64_t fallback) {
90 if (!config.isKey(key)) return fallback;
91 return requiredUnsigned(config, key, maximum);
92 }

112bool ccsds::pus::SecondaryHeader::identifierFits(const std::uint32_t value,
113 const std::uint8_t octets) const {
114 if (octets == 0U) return value == 0U;
115 if (octets >= 4U) return true;
116 return value < (1UL << (octets * 8U));
117}
118
119void ccsds::pus::SecondaryHeader::appendIdentifier(std::vector<std::uint8_t> &bytes,
120 const std::uint32_t value,
121 const std::uint8_t octets) {
122 for (std::uint8_t i = 0; i < octets; ++i) {
123 const auto shift = static_cast<std::uint8_t>((octets - i - 1U) * 8U);
124 bytes.push_back(static_cast<std::uint8_t>((value >> shift) & 0xFFU));
125 }
126}
127
128std::uint32_t ccsds::pus::SecondaryHeader::readIdentifier(
129 const std::vector<std::uint8_t> &bytes, const std::size_t offset,
130 const std::uint8_t octets) {
131 std::uint32_t result = 0U;
132 for (std::uint8_t i = 0; i < octets; ++i) {
133 result = static_cast<std::uint32_t>((result << 8U) | bytes[offset + i]);
134 }
135 return result;
136}

138bool ccsds::pus::SecondaryHeader::trailingSpareIsZero(
139 const std::vector<std::uint8_t> &bytes) const {
140 const auto spare = static_cast<std::size_t>(m_secondaryHeaderSpareOctets);
141 if (spare > bytes.size()) return false;
142 for (std::size_t i = bytes.size() - spare; i < bytes.size(); ++i) {
143 if (bytes[i] != 0U) return false;
144 }
145 return true;
146}

220ccsds::ResultBool ccsds::pus::TcSecondaryHeader::parseTcBody(
221 const std::vector<std::uint8_t> &data, const std::size_t offset) {
222 RET_IF_ERR_MSG(data.size() != tcSize(), ErrorCode::INVALID_SECONDARY_HEADER_DATA,
223 "PUS TC secondary-header size does not match its tailoring.");
224 RET_IF_ERR_MSG(!trailingSpareIsZero(data), ErrorCode::INVALID_SECONDARY_HEADER_DATA,
225 "PUS TC secondary-header spare octets must be zero.");
226 const auto source = readIdentifier(data, offset + 2U, m_sourceIdOctets);
227 m_serviceType = data[offset];
228 m_serviceSubtype = data[offset + 1U];
229 m_sourceId = source;
230 return true;
231}

266ccsds::ResultBool ccsds::pus::TmSecondaryHeader::loadFromConfig(
267 const ccsds::Config &config) {
268 FORWARD_RESULT(SecondaryHeader::loadFromConfig(config));
269
270 if (getRevision() == Revision::A) {
271 std::uint64_t width{};
272 ASSIGN_CP(width, optionalUnsigned(config, "pus_destination_id_octets", 4U,
273 m_destinationIdOctets));
274 RET_IF_ERR_MSG(!validIdentifierWidth(static_cast<std::uint8_t>(width)),
275 ErrorCode::CONFIG_FILE_ERROR,
276 "Config: pus_destination_id_octets must be 0, 1, 2, or 4.");
277 m_destinationIdOctets = static_cast<std::uint8_t>(width);
278 } else if (config.isKey("pus_destination_id_octets")) {
279 std::uint64_t width{};
280 ASSIGN_CP(width, requiredUnsigned(config, "pus_destination_id_octets", 4U));
281 RET_IF_ERR_MSG(width != 2U, ErrorCode::CONFIG_FILE_ERROR,
282 "Config: PUS-C TM destination ID is fixed at two octets.");
283 }
284
285 if (config.isKey("pus_time_format")) {
286 const auto format = config.get<std::string>("pus_time_format");
287 RET_IF_ERR_MSG(!format, ErrorCode::CONFIG_FILE_ERROR,
288 "Config: invalid pus_time_format.");
289 if (format.value() == "none") {
290 m_timestampPresent = false;
291 m_cuc = {};
292 RET_IF_ERR_MSG(config.isKey("pus_time_epoch") || config.isKey("pus_time_p_field")
293 || config.isKey("pus_time_coarse_octets")
294 || config.isKey("pus_time_fine_octets")
295 || config.isKey("pus_time_coarse") || config.isKey("pus_time_fine"),
296 ErrorCode::CONFIG_FILE_ERROR,
297 "Config: disabled PUS time cannot contain CUC fields.");
298 } else if (format.value() == "cuc") {
299 m_timestampPresent = true;
300 RET_IF_ERR_MSG(!config.isKey("pus_time_epoch") || !config.isKey("pus_time_p_field")
301 || !config.isKey("pus_time_coarse_octets")
302 || !config.isKey("pus_time_fine_octets"),
303 ErrorCode::CONFIG_FILE_ERROR,
304 "Config: CUC tailoring requires epoch, P-field, coarse, and fine width fields.");
305 const auto epoch = config.get<std::string>("pus_time_epoch");
306 const auto pField = config.get<std::string>("pus_time_p_field");
307 RET_IF_ERR_MSG(!epoch || !pField, ErrorCode::CONFIG_FILE_ERROR,
308 "Config: invalid PUS CUC string field.");
309 m_cuc.epoch = epoch.value() == "ccsds-1958-tai" ? time::Epoch::Ccsds1958Tai
310 : epoch.value() == "agency-defined" ? time::Epoch::AgencyDefined
311 : time::Epoch::Unspecified;
312 m_cuc.pField = pField.value() == "implicit" ? time::PFieldMode::Implicit
313 : pField.value() == "explicit" ? time::PFieldMode::Explicit
314 : static_cast<time::PFieldMode>(0xFFU);
315 std::uint64_t coarse{};
316 std::uint64_t fine{};
317 ASSIGN_CP(coarse, requiredUnsigned(config, "pus_time_coarse_octets", 4U));
318 ASSIGN_CP(fine, requiredUnsigned(config, "pus_time_fine_octets", 3U));
319 RET_IF_ERR_MSG(coarse < 1U, ErrorCode::CONFIG_FILE_ERROR,
320 "Config: pus_time_coarse_octets must be 1..4.");
321 m_cuc.coarseOctets = static_cast<std::uint8_t>(coarse);
322 m_cuc.fineOctets = static_cast<std::uint8_t>(fine);
323 FORWARD_RESULT(time::validate(m_cuc));
324 } else {
325 return Error{ErrorCode::CONFIG_FILE_ERROR,
326 "Config: pus_time_format must be 'none' or 'cuc'."};
327 }
328 }
329
330 std::uint64_t serviceType{};
331 std::uint64_t serviceSubtype{};
332 std::uint64_t destinationId{};
333 ASSIGN_CP(serviceType, requiredUnsigned(config, "pus_service_type", UINT8_MAX));
334 ASSIGN_CP(serviceSubtype, requiredUnsigned(config, "pus_service_subtype", UINT8_MAX));
335 ASSIGN_CP(destinationId, requiredUnsigned(config, "pus_destination_id", UINT32_MAX));
336 m_serviceType = static_cast<std::uint8_t>(serviceType);
337 m_serviceSubtype = static_cast<std::uint8_t>(serviceSubtype);
338 FORWARD_RESULT(setDestinationId(static_cast<std::uint32_t>(destinationId)));
339
340 if (m_timestampPresent) {
341 time::CucTime timestamp;
342 ASSIGN_CP(timestamp.coarse, requiredUnsigned(config, "pus_time_coarse", UINT32_MAX));
343 ASSIGN_CP(timestamp.fine, requiredUnsigned(config, "pus_time_fine", 0xFFFFFFU));
344 FORWARD_RESULT(setTimestamp(timestamp));
345 } else {
346 m_timestamp = {};
347 }
348 return true;
349}

352ccsds::ResultBool ccsds::pus::TmSecondaryHeader::appendTmTail(
353 std::vector<std::uint8_t> &bytes) const {
354 appendIdentifier(bytes, m_destinationId, m_destinationIdOctets);
355 if (m_timestampPresent) {
356 std::vector<std::uint8_t> encoded;
357 ASSIGN_MV(encoded, time::serialize(m_timestamp, m_cuc));
358 bytes.insert(bytes.end(), encoded.begin(), encoded.end());
359 }
360 bytes.insert(bytes.end(), m_secondaryHeaderSpareOctets, 0U);
361 return true;
362}

364ccsds::ResultBool ccsds::pus::TmSecondaryHeader::parseTmTail(
365 const std::vector<std::uint8_t> &data, const std::size_t offset) {
366 const auto tail = tmTailSize();
367 RET_IF_ERR_MSG(offset + tail != data.size(), ErrorCode::INVALID_SECONDARY_HEADER_DATA,
368 "PUS TM secondary-header size does not match its tailoring.");
369 RET_IF_ERR_MSG(!trailingSpareIsZero(data), ErrorCode::INVALID_SECONDARY_HEADER_DATA,
370 "PUS TM secondary-header spare octets must be zero.");
371 const auto destination = readIdentifier(data, offset, m_destinationIdOctets);
372 const auto timestampOffset = offset + m_destinationIdOctets;
373 const auto timestampSize = m_timestampPresent ? time::encodedSize(m_cuc) : 0U;
374 const auto timestampEnd = timestampOffset + timestampSize;
375 m_destinationId = destination;
376 if (m_timestampPresent) {
377 const std::vector<std::uint8_t> encoded(
378 data.begin() + static_cast<std::ptrdiff_t>(timestampOffset),
379 data.begin() + static_cast<std::ptrdiff_t>(timestampEnd));
380 ASSIGN_CP(m_timestamp, time::deserialize(encoded, m_cuc));
381 } else {
382 m_timestamp = {};
383 }
384 return true;
385}

387std::uint16_t ccsds::pus::TmSecondaryHeader::tmTailSize() const {
388 return static_cast<std::uint16_t>(m_destinationIdOctets
389 + (m_timestampPresent ? time::encodedSize(m_cuc) : 0U)
390 + m_secondaryHeaderSpareOctets);
391}

427std::vector<std::uint8_t> ccsds::pus::rev_a::TcHeader::serialize() const {
428 const ccsds_pus_a_tc_tailoring_t tailoring{
429 m_sourceIdOctets,
430 m_secondaryHeaderSpareOctets
431 };
432 const ccsds_pus_tc_fields_t fields{
433 m_acknowledgementFlags,
434 m_serviceType,
435 m_serviceSubtype,
436 m_sourceId
437 };
438 std::vector<std::uint8_t> bytes(ccsds_pus_a_tc_encoded_size(&tailoring));
439 std::size_t written = 0U;
440 if (bytes.empty()
441 || ccsds_pus_a_tc_encode(&fields, &tailoring, bytes.data(), bytes.size(), &written)
442 != CCSDS_STATUS_OK
443 || written != bytes.size()) {
444 return {};
445 }
446 return bytes;
447}

474ccsds::ResultBool ccsds::pus::rev_a::TmHeader::deserialize(
475 const std::vector<std::uint8_t> &data) {
476 FORWARD_RESULT(validateTailoring(getTailoring()));
477 const ccsds_pus_a_tm_tailoring_t tailoring{
478 m_destinationIdOctets,
479 static_cast<std::uint8_t>(m_packetSubcounterPresent ? 1U : 0U),
480 static_cast<std::uint8_t>(m_timestampPresent ? 1U : 0U),
481 {
482 static_cast<ccsds_cuc_epoch_t>(m_cuc.epoch),
483 static_cast<ccsds_cuc_pfield_mode_t>(m_cuc.pField),
484 m_cuc.coarseOctets,
485 m_cuc.fineOctets
486 },
487 m_secondaryHeaderSpareOctets
488 };
489 ccsds_pus_a_tm_fields_t fields{};
490 const auto status = ccsds_pus_a_tm_decode(
491 data.data(), data.size(), &tailoring, &fields);
492 if (status != CCSDS_STATUS_OK) return pusTmDecodeError(status, true);
493
494 m_serviceType = fields.service_type;
495 m_serviceSubtype = fields.service_subtype;
496 m_packetSubcounter = fields.packet_subcounter;
497 m_destinationId = fields.destination_id;
498 m_timestamp = {fields.timestamp.coarse, fields.timestamp.fine};
499 return true;
500}

502std::vector<std::uint8_t> ccsds::pus::rev_a::TmHeader::serialize() const {
503 if (!validateTailoring(getTailoring())) return {};
504 const ccsds_pus_a_tm_tailoring_t tailoring{
505 m_destinationIdOctets,
506 static_cast<std::uint8_t>(m_packetSubcounterPresent ? 1U : 0U),
