9 bool identifierFits(const std::uint32_t value, const std::uint8_t octets) noexcept {
10 if (octets == 0U) return value == 0U;
11 if (octets >= 4U) return true;
12 return value < (1UL << (octets * 8U));
13 }

24 bool reservedBitsValid(const std::vector<std::uint8_t> &bytes,
25 const ccsds::pus::Revision revision,
26 const ccsds::PacketDirection direction) noexcept {
27 if (bytes.empty()) return false;
28 if (revision == ccsds::pus::Revision::A
29 && direction == ccsds::PacketDirection::Telecommand) {
30 return (bytes[0] & 0x80U) == 0U && ((bytes[0] >> 4U) & 0x07U) == 1U;
31 }
32 if (revision == ccsds::pus::Revision::A
33 && direction == ccsds::PacketDirection::Telemetry) {
34 return bytes[0] == 0x10U;
35 }
36 if (revision == ccsds::pus::Revision::C) return (bytes[0] >> 4U) == 2U;
37 return false;
38 }

40 bool spareFieldsValid(const std::vector<std::uint8_t> &bytes,
41 const std::uint8_t spareOctets) noexcept {
42 const auto spare = static_cast<std::size_t>(spareOctets);
43 if (spare > bytes.size()) return false;
44 for (std::size_t i = bytes.size() - spare; i < bytes.size(); ++i)
45 if (bytes[i] != 0U) return false;
46 return true;
47 }

55 bool pusTailoringValid(const ccsds::pus::SecondaryHeader &header) {
56 if (!validRevision(header.getRevision()) || !validDirection(header.getDirection())) return false;
57 if (header.getDirection() == ccsds::PacketDirection::Telecommand) {
58 const auto &tc = static_cast<const ccsds::pus::TcSecondaryHeader &>(header);
59 if (!ccsds::pus::validIdentifierWidth(tc.getSourceIdOctets())) return false;
60 return header.getRevision() != ccsds::pus::Revision::C || tc.getSourceIdOctets() == 2U;
61 }
62 const auto &tm = static_cast<const ccsds::pus::TmSecondaryHeader &>(header);
63 if (!ccsds::pus::validIdentifierWidth(tm.getDestinationIdOctets())) return false;
64 if (header.getRevision() == ccsds::pus::Revision::C && tm.getDestinationIdOctets() != 2U)
65 return false;
66 if (tm.timestampPresent()) return static_cast<bool>(ccsds::time::validate(tm.getCucConfiguration()));
67 return cucIsEmpty(tm.getCucConfiguration());
68 }

70 bool sameSecondaryContract(const ccsds::Packet &lhs, const ccsds::Packet &rhs) noexcept {
71 const auto lhsHeader = lhs.getSecondaryHeader();
72 const auto rhsHeader = rhs.getSecondaryHeader();
73 if (static_cast<bool>(lhsHeader) != static_cast<bool>(rhsHeader)) return false;
74 if (!lhsHeader) return true;
75 if (lhsHeader->getType() != rhsHeader->getType()
76 || lhsHeader->getDirection() != rhsHeader->getDirection()
77 || lhsHeader->isPusHeader() != rhsHeader->isPusHeader()) return false;
78 if (!lhsHeader->isPusHeader()) return true;
79 return ccsds::pus::sameTailoring(
80 static_cast<const ccsds::pus::SecondaryHeader &>(*lhsHeader),
81 static_cast<const ccsds::pus::SecondaryHeader &>(*rhsHeader));
82 }

125void ccsds::Validator::acceptSequence(const Header &header) noexcept {
126 const auto next = static_cast<std::uint16_t>((header.getSequenceCount() + 1U) & SEQUENCE_COUNT_MASK);
127 auto state = static_cast<std::uint16_t>(SEQUENCE_INITIALIZED_MASK | next);
128 if (header.getSequenceFlags() == FIRST_SEGMENT
129 || header.getSequenceFlags() == CONTINUING_SEGMENT) state |= SEGMENT_OPEN_MASK;
130 m_sequenceCounter = state;
131}

133ccsds::ValidationReport ccsds::Validator::validate(const Packet &packet) {
134 m_report = {};
135 const auto &header = packet.getPrimaryHeader();
136 const auto headerData = header.serialize();
137 const bool primaryHeaderValid = header.getHeaderStatus() != INVALID && headerData.size() == 6U;
138 setCheck(ValidationCode::PrimaryHeader, primaryHeaderValid);
139 if (!primaryHeaderValid) return m_report;
140
141 bool sequenceFlagsValid{true};
142 bool sequenceCountValid{true};
143
144 if (m_validatePacketCoherence) {
145 setCheck(ValidationCode::PacketVersion, header.getVersionNumber() == 0U);
146 const auto serializedSize = packet.getSerializedSize();
147 const auto packetDataFieldSize = serializedSize >= 6U ? serializedSize - 6U : 0U;
148 setCheck(ValidationCode::PacketDataLength,
149 packetDataFieldSize > 0U && header.getDataLength() == packetDataFieldSize - 1U);
150
151 if (packet.getPacketErrorControlMode() == PacketErrorControlMode::CRC16) {
152 auto crcInput = headerData;
153 const auto dataFieldBytes = packet.getFullDataFieldBytes();
154 crcInput.insert(crcInput.end(), dataFieldBytes.begin(), dataFieldBytes.end());
155 const CRC16Config crcConfig{};
156 const auto calculatedCRC = ccsds::crc16(
157 crcInput, crcConfig.polynomial, crcConfig.initialValue, crcConfig.finalXorValue);
158 setCheck(ValidationCode::Crc16, calculatedCRC == packet.getCRC());
159 }
160
161 const auto secondary = packet.getSecondaryHeader();
162 const bool secondaryPresence =
163 (header.getSecondaryHeaderFlag() != 0U) == static_cast<bool>(secondary);
164 setCheck(ValidationCode::SecondaryHeaderPresence, secondaryPresence);
165
166 if (secondary && secondary->getDirection() != PacketDirection::Unspecified) {
167 setCheck(ValidationCode::SecondaryHeaderDirection,
168 header.getType() == packetTypeForDirection(secondary->getDirection()));
169 }
170
171 if (secondary && secondary->isPusHeader()) {
172 setCheck(ValidationCode::PusHeader, true);
173 const auto &pusHeader = static_cast<const pus::SecondaryHeader &>(*secondary);
174 const bool revisionValid = validRevision(pusHeader.getRevision());
175 const bool directionValid = validDirection(pusHeader.getDirection());
176 setCheck(ValidationCode::PusRevision, revisionValid);
177 setCheck(ValidationCode::PusDirection, directionValid);
178 setCheck(ValidationCode::PusPacketType,
179 directionValid
180 && header.getType() == packetTypeForDirection(pusHeader.getDirection()));
181 setCheck(ValidationCode::PusTailoring, pusTailoringValid(pusHeader));
182
183 const auto bytes = secondary->serialize();
184 const bool sizeValid = !bytes.empty() && bytes.size() == secondary->getSize();
185 setCheck(ValidationCode::PusSecondaryHeaderSize, sizeValid);
186 setCheck(ValidationCode::PusReservedBits,
187 sizeValid && reservedBitsValid(bytes, pusHeader.getRevision(), pusHeader.getDirection()));
188 setCheck(ValidationCode::PusSpareFields,
189 sizeValid && spareFieldsValid(bytes, pusHeader.getSecondaryHeaderSpareOctets()));
190
191 if (pusHeader.getDirection() == PacketDirection::Telecommand) {
192 const auto &tc = static_cast<const pus::TcSecondaryHeader &>(pusHeader);
193 setCheck(ValidationCode::PusAcknowledgement, tc.getAcknowledgementFlags() <= 0x0FU);
194 setCheck(ValidationCode::PusSourceId,
195 identifierFits(tc.getSourceId(), tc.getSourceIdOctets()));
196 } else if (pusHeader.getDirection() == PacketDirection::Telemetry) {
197 const auto &tm = static_cast<const pus::TmSecondaryHeader &>(pusHeader);
198 setCheck(ValidationCode::PusDestinationId,
199 identifierFits(tm.getDestinationId(), tm.getDestinationIdOctets()));
200 if (tm.timestampPresent()) {
201 const auto timestamp = time::serialize(tm.getTimestamp(), tm.getCucConfiguration());
202 setCheck(ValidationCode::PusTimestamp, static_cast<bool>(timestamp));
203 } else {
204 setCheck(ValidationCode::PusTimestamp, tm.getTimestamp() == time::CucTime{});
205 }
206 if (pusHeader.getRevision() == pus::Revision::A) {
207 const auto &aTm = static_cast<const pus::rev_a::TmHeader &>(pusHeader);
208 setCheck(ValidationCode::PusPacketSubcounter,
209 aTm.packetSubcounterPresent() || aTm.getPacketSubcounter() == 0U);
210 } else if (pusHeader.getRevision() == pus::Revision::C) {
211 const auto &cTm = static_cast<const pus::rev_c::TmHeader &>(pusHeader);
212 setCheck(ValidationCode::PusTimeReferenceStatus,
213 cTm.getTimeReferenceStatus() <= 0x0FU);
214 }
215 }
216 }
217
218 const auto flags = static_cast<ESequenceFlag>(header.getSequenceFlags());
219 const bool open = segmentOpen();
220 switch (flags) {
221 case UNSEGMENTED:
222 case FIRST_SEGMENT: sequenceFlagsValid = !open; break;
223 case CONTINUING_SEGMENT:
224 case LAST_SEGMENT: sequenceFlagsValid = open; break;
225 default: sequenceFlagsValid = false; break;
226 }
227 setCheck(ValidationCode::SequenceFlags, sequenceFlagsValid);
228 if (m_validateSequenceCount) {
229 sequenceCountValid = !sequenceInitialized()
230 || header.getSequenceCount() == expectedSequenceCount();
231 setCheck(ValidationCode::SequenceCount, sequenceCountValid);
232 }
233 }
234
235 if (m_validateAgainstTemplate) {
236 const auto &templateHeader = m_templatePacket.getPrimaryHeader();
237 const auto templateHeaderData = templateHeader.serialize();
238 const bool templateHeaderValid = templateHeader.getHeaderStatus() != INVALID
239 && templateHeaderData.size() == 6U;
240 setCheck(ValidationCode::PacketIdentifier,
241 templateHeaderValid && templateHeaderData[0] == headerData[0]
242 && templateHeaderData[1] == headerData[1]);
243 bool segmentationClassValid{false};
244 if (templateHeaderValid) {
245 segmentationClassValid = templateHeader.getSequenceFlags() == UNSEGMENTED
246 ? header.getSequenceFlags() == UNSEGMENTED
247 : header.getSequenceFlags() != UNSEGMENTED;
248 }
249 setCheck(ValidationCode::SegmentationClass, segmentationClassValid);
250 setCheck(ValidationCode::TemplatePacketErrorControl,
251 m_templatePacket.getPacketErrorControlMode() == packet.getPacketErrorControlMode());
252 setCheck(ValidationCode::TemplateSecondaryHeader,
253 sameSecondaryContract(m_templatePacket, packet));
254 }
255
256 if (m_validatePacketCoherence && sequenceFlagsValid
257 && (!m_validateSequenceCount || sequenceCountValid) && m_report.valid()) {
258 acceptSequence(header);
259 }
260 return m_report;
261}
