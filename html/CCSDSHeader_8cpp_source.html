74CCSDS::ResultBool CCSDS::Header::deserialize(const std::vector<std::uint8_t> &data) {
75 if (data.size() != 6U) {
76 m_status = INVALID;
77 return Error(INVALID_HEADER_DATA, "Invalid Header Data provided: size != 6");
78 }
79
80 std::uint64_t headerData = 0;
81 for (std::uint8_t i = 0; i < 6; ++i) {
82 headerData |= static_cast<std::uint64_t>(data[i]) << (40 - i * 8);
83 }
84 FORWARD_RESULT(setData(headerData));
85 return true;
86}

88CCSDS::ResultBool CCSDS::Header::setData(const std::uint64_t &data) {
89 if (data > 0xFFFFFFFFFFFFULL) {
90 m_status = INVALID;
91 return Error(INVALID_HEADER_DATA, "Input data exceeds expected bit size for version or size.");
92 }
93
94 m_dataLength = static_cast<std::uint16_t>(data & 0xFFFFU);
95 m_packetSequenceControl = static_cast<std::uint16_t>((data >> 16) & 0xFFFFU);
96 m_packetIdentificationAndVersion = static_cast<std::uint16_t>((data >> 32) & 0xFFFFU);
97
98 m_versionNumber = static_cast<std::uint8_t>(m_packetIdentificationAndVersion >> 13);
99 m_type = static_cast<std::uint8_t>((m_packetIdentificationAndVersion >> 12) & 0x01U);
100 m_dataFieldHeaderFlag = static_cast<std::uint8_t>((m_packetIdentificationAndVersion >> 11) & 0x01U);
101 m_APID = m_packetIdentificationAndVersion & 0x07FFU;
102 m_sequenceFlags = static_cast<std::uint8_t>(m_packetSequenceControl >> 14);
103 m_sequenceCount = m_packetSequenceControl & 0x3FFFU;
104 refreshStatus();
105 return true;
106}

112std::vector<std::uint8_t> CCSDS::Header::serialize() const {
113 if (m_status == INVALID) {
114 return {};
115 }
116
117 const auto packetSequenceControl = static_cast<std::uint16_t>(
118 (static_cast<std::uint16_t>(m_sequenceFlags) << 14U) | m_sequenceCount);
119 const auto packetIdentificationAndVersion = static_cast<std::uint16_t>(
120 (static_cast<std::uint16_t>(m_versionNumber) << 13U)
121 | (static_cast<std::uint16_t>(m_type) << 12U)
122 | (static_cast<std::uint16_t>(m_dataFieldHeaderFlag) << 11U)
123 | m_APID);
124
125 return {
126 static_cast<std::uint8_t>(packetIdentificationAndVersion >> 8U),
127 static_cast<std::uint8_t>(packetIdentificationAndVersion & 0xFFU),
128 static_cast<std::uint8_t>(packetSequenceControl >> 8U),
129 static_cast<std::uint8_t>(packetSequenceControl & 0xFFU),
130 static_cast<std::uint8_t>(m_dataLength >> 8U),
131 static_cast<std::uint8_t>(m_dataLength & 0xFFU),
132 };
133}
