75ccsds::ResultBool ccsds::Header::deserialize(const std::vector<std::uint8_t> &data) {
76 if (data.size() != CCSDS_PRIMARY_HEADER_SIZE) {
77 m_status = INVALID;
78 return Error(INVALID_HEADER_DATA, "Invalid Header Data provided: size != 6");
79 }
80
81 ccsds_primary_header_t decoded{};
82 const auto status = ccsds_primary_header_decode(data.data(), data.size(), &decoded);
83 if (status != CCSDS_STATUS_OK) {
84 m_status = INVALID;
85 return Error(INVALID_HEADER_DATA, "Invalid Header Data provided.");
86 }
87
88 FORWARD_RESULT(setData(PrimaryHeader{
89 decoded.version_number,
90 decoded.type,
91 decoded.secondary_header_flag,
92 decoded.apid,
93 decoded.sequence_flags,
94 decoded.sequence_count,
95 decoded.data_length
96 }));
97 return true;
98}

124std::vector<std::uint8_t> ccsds::Header::serialize() const {
125 if (m_status == INVALID) {
126 return {};
127 }
128
129 const ccsds_primary_header_t header{
130 m_versionNumber,
131 m_type,
132 m_secondaryHeaderFlag,
133 m_APID,
134 m_sequenceFlags,
135 m_sequenceCount,
136 m_dataLength
137 };
138 std::uint8_t bytes[CCSDS_PRIMARY_HEADER_SIZE]{};
139 if (ccsds_primary_header_encode(&header, bytes, sizeof(bytes)) != CCSDS_STATUS_OK) {
140 return {};
141 }
142 return std::vector<std::uint8_t>(bytes, bytes + CCSDS_PRIMARY_HEADER_SIZE);
143}

167ccsds::ResultBool ccsds::Header::setData(const PrimaryHeader &data) {
168 if (data.versionNumber > 0x07U) {
169 m_status = INVALID;
170 return Error(INVALID_HEADER_DATA, "Invalid version number, value > 7");
171 }
172 if (data.type > 0x01U) {
173 m_status = INVALID;
174 return Error(INVALID_HEADER_DATA, "Invalid type, value > 1");
175 }
176 if (data.secondaryHeaderFlag > 0x01U) {
177 m_status = INVALID;
178 return Error(INVALID_HEADER_DATA, "Invalid secondary header flag, value > 1");
179 }
180 if (data.APID > IDLE_APID) {
181 m_status = INVALID;
182 return Error(INVALID_HEADER_DATA, "Invalid APID, value > 2047");
183 }
184 if (data.sequenceFlags > 0x03U) {
185 m_status = INVALID;
186 return Error(INVALID_HEADER_DATA, "Invalid sequence flag, value > 3");
187 }
188 if (data.sequenceCount > 0x3FFFU) {
189 m_status = INVALID;
190 return Error(INVALID_HEADER_DATA, "Invalid sequence count, value > 16383");
191 }
192
193 m_versionNumber = data.versionNumber;
194 m_type = data.type;
195 m_secondaryHeaderFlag = data.secondaryHeaderFlag;
196 m_APID = data.APID;
197 m_sequenceFlags = data.sequenceFlags;
198 m_sequenceCount = data.sequenceCount;
199 m_dataLength = data.dataLength;
200 m_packetSequenceControl = static_cast<std::uint16_t>(
201 (static_cast<std::uint16_t>(m_sequenceFlags) << 14U) | m_sequenceCount);
202 m_packetIdentificationAndVersion = static_cast<std::uint16_t>(
203 (static_cast<std::uint16_t>(m_versionNumber) << 13U)
204 | (static_cast<std::uint16_t>(m_type) << 12U)
205 | (static_cast<std::uint16_t>(m_secondaryHeaderFlag) << 11U)
206 | m_APID);
207 refreshStatus();
208 return true;
209}
