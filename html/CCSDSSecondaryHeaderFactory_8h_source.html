18 class SecondaryHeaderFactory {
19 public:
20 using CreatorFunc = std::function<std::shared_ptr<SecondaryHeaderAbstract>()>;
21
22 template <typename T>
23 ResultBool registerType() {
24 static_assert(std::is_base_of<SecondaryHeaderAbstract, T>::value,
25 "T must derive from SecondaryHeaderAbstract");
26 static_assert(std::is_default_constructible<T>::value,
27 "Custom secondary-header types must be default constructible");
28 const auto probe = std::make_shared<T>();
29 return registerCreator(probe->getType(), [] {
30 return std::static_pointer_cast<SecondaryHeaderAbstract>(std::make_shared<T>());
31 });
32 }
33
34 ResultBool registerCreator(const std::string &type, CreatorFunc creator) {
35 RET_IF_ERR_MSG(type.empty(), ErrorCode::INVALID_HEADER_DATA,
36 "Cannot register an empty secondary-header type.");
37 RET_IF_ERR_MSG(type.rfind("PUS:", 0U) == 0U, ErrorCode::INVALID_HEADER_DATA,
38 "The PUS: namespace is reserved for standards-defined codecs.");
39 RET_IF_ERR_MSG(!creator, ErrorCode::INVALID_HEADER_DATA,
40 "Cannot register an empty secondary-header creator.");
41 RET_IF_ERR_MSG(m_creators.find(type) != m_creators.end(), ErrorCode::INVALID_HEADER_DATA,
42 "Secondary-header type is already registered: " + type);
43 m_creators.emplace(type, std::move(creator));
44 return true;
45 }
46
47 [[nodiscard]] std::shared_ptr<SecondaryHeaderAbstract> create(const std::string &type) const {
48 const auto it = m_creators.find(type);
49 return it == m_creators.end() ? nullptr : it->second();
50 }
51
52 [[nodiscard]] bool typeIsRegistered(const std::string &type) const {
53 return m_creators.find(type) != m_creators.end();
54 }
55
56 private:
57 std::unordered_map<std::string, CreatorFunc> m_creators;
58 };
