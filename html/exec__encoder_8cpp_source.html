40ccsds::ResultBool configureManagerFraming(ccsds::Manager &manager, const ccsds::Config &cfg) {
41 if (!cfg.isKey("sync_pattern_enable")) return true;
42
43 bool enabled{};
44 ASSIGN_CP(enabled, cfg.get<bool>("sync_pattern_enable"));
45 manager.setSyncPatternEnable(enabled);
46 if (enabled && cfg.isKey("sync_pattern")) {
47 int encoded{};
48 ASSIGN_CP(encoded, cfg.get<int>("sync_pattern"));
49 manager.setSyncPattern(static_cast<std::uint32_t>(encoded));
50 }
51 return true;
52}

55int main(const int argc, char *argv[]) {
56 const std::string appName = "ccsds_encoder";
57 std::unordered_map<std::string, std::string> allowed{
58 {"h", "help"}, {"v", "verbose"}, {"i", "input"},
59 {"o", "output"}, {"c", "config"}, {"e", "packet-error-control"}
60 };
61 const std::set<std::string> booleanArgs{"verbose", "help"};
62 std::unordered_map<std::string, std::string> args{
63 {"verbose", "false"}, {"help", "false"}
64 };
65
66 const auto start = std::chrono::high_resolution_clock::now();
67 if (const auto result = parseArguments(argc, argv, allowed, args, booleanArgs); !result) {
68 return printError(result.error());
69 }
70 if (args["help"] == "true") {
71 printHelp();
72 return 0;
73 }
74
75 const auto requirePath = [&](const char *key, const char *description)
76 -> ccsds::Result<std::string> {
77 const auto it = args.find(key);
78 if (it == args.end() || it->second.empty()) {
79 return ccsds::Error{static_cast<ccsds::ErrorCode>(ARG_PARSE_ERROR),
80 std::string(description) + " must be specified"};
81 }
82 return it->second;
83 };
84
85 std::string input;
86 std::string output;
87 std::string configFile;
88 {
89 auto result = requirePath("input", "Input file");
90 if (!result) { printHelp(); return printError(result.error()); }
91 input = result.value();
92 }
93 {
94 auto result = requirePath("output", "Output file");
95 if (!result) { printHelp(); return printError(result.error()); }
96 output = result.value();
97 }
98 {
99 auto result = requirePath("config", "Configuration file");
100 if (!result) { printHelp(); return printError(result.error()); }
101 configFile = result.value();
102 }
103
104 if (!ccsds::fileExists(input)) {
105 return printError(ccsds::Error{static_cast<ccsds::ErrorCode>(ARG_PARSE_ERROR),
106 "Input file does not exist: " + input});
107 }
108 if (!ccsds::fileExists(configFile)) {
109 return printError(ccsds::Error{static_cast<ccsds::ErrorCode>(ARG_PARSE_ERROR),
110 "Configuration file does not exist: " + configFile});
111 }
112
113 customConsole(appName, "reading CCSDS configuration file: " + configFile);
114 ccsds::Config cfg;
115 if (const auto result = cfg.load(configFile); !result) return printError(result.error());
116
117 ccsds::Packet templatePacket;
118 if (const auto result = templatePacket.loadFromConfig(cfg); !result) {
119 return printError(result.error());
120 }
121
122 if (const auto it = args.find("packet-error-control"); it != args.end()) {
123 const auto result = parsePacketErrorControlMode(it->second);
124 if (!result) return printError(result.error());
125 if (const auto applied = applyPacketErrorControlMode(templatePacket, result.value());
126 !applied) return printError(applied.error());
127 }
128
129 ccsds::Manager manager;
130 if (const auto result = manager.setPacketTemplate(std::move(templatePacket)); !result) {
131 return printError(result.error());
132 }
133 if (const auto result = configureManagerFraming(manager, cfg); !result) {
134 return printError(result.error());
135 }
136
137 customConsole(appName,
138 std::string("packet error control: ")
139 + packetErrorControlModeName(manager.getTemplate().getPacketErrorControlMode()));
140
141 std::vector<std::uint8_t> inputBytes;
142 customConsole(appName, "reading data from " + input);
143 {
144 const auto result = ccsds::readBinaryFile(input);
145 if (!result) return printError(result.error());
146 inputBytes = result.value();
147 }
148
149 if (manager.getTemplate().getPrimaryHeader().getSequenceFlags() == ccsds::UNSEGMENTED
150 && inputBytes.size() > manager.getDataFieldSize()) {
151 return printError(ccsds::Error{
152 static_cast<ccsds::ErrorCode>(INVALID_INPUT_DATA),
153 "Input contains " + std::to_string(inputBytes.size())
154 + " bytes, exceeding the unsegmented packet capacity of "
155 + std::to_string(manager.getDataFieldSize())});
156 }
157
158 customConsole(appName, "generating CCSDS packets using input data");
159 if (const auto result = manager.setApplicationData(inputBytes); !result) {
160 return printError(result.error());
161 }
162
163 if (args["verbose"] == "true") ccsds::printPackets(manager);
164
165 const auto packets = manager.getPacketsBuffer();
166 if (!packets) return printError(packets.error());
167
168 customConsole(appName, "writing data to " + output);
169 if (const auto result = ccsds::writeBinaryFile(packets.value(), output); !result) {
170 return printError(result.error());
171 }
172
173 const auto duration = std::chrono::duration_cast<std::chrono::microseconds>(
174 std::chrono::high_resolution_clock::now() - start);
175 customConsole(appName, "execution time: " + std::to_string(duration.count()) + " [us]");
176 customConsole(appName, "[ Exit code 0 ]");
177 return 0;
178}
