40CCSDS::ResultBool configureManagerFraming(CCSDS::Manager &manager, const Config &cfg) {
41 if (!cfg.isKey("sync_pattern_enable")) return true;
42
43 bool enabled{};
44 ASSIGN_CP(enabled, cfg.get<bool>("sync_pattern_enable"));
45 manager.setSyncPatternEnable(enabled);
46 if (enabled && cfg.isKey("sync_pattern")) {
47 int encoded{};
48 ASSIGN_CP(encoded, cfg.get<int>("sync_pattern"));
49 manager.setSyncPattern(static_cast<std::uint32_t>(encoded));
50 }
51 return true;
52}

55int main(const int argc, char *argv[]) {
56 const std::string appName = "ccsds_encoder";
57 std::unordered_map<std::string, std::string> allowed{
58 {"h", "help"}, {"v", "verbose"}, {"i", "input"},
59 {"o", "output"}, {"c", "config"}, {"e", "packet-error-control"}
60 };
61 const std::set<std::string> booleanArgs{"verbose", "help"};
62 std::unordered_map<std::string, std::string> args{
63 {"verbose", "false"}, {"help", "false"}
64 };
65
66 const auto start = std::chrono::high_resolution_clock::now();
67 if (const auto result = parseArguments(argc, argv, allowed, args, booleanArgs); !result) {
68 return printError(result.error());
69 }
70 if (args["help"] == "true") {
71 printHelp();
72 return 0;
73 }
74
75 const auto requirePath = [&](const char *key, const char *description)
76 -> CCSDS::Result<std::string> {
77 const auto it = args.find(key);
78 if (it == args.end() || it->second.empty()) {
79 return CCSDS::Error{static_cast<CCSDS::ErrorCode>(ARG_PARSE_ERROR),
80 std::string(description) + " must be specified"};
81 }
82 return it->second;
83 };
84
85 std::string input;
86 std::string output;
87 std::string configFile;
88 {
89 auto result = requirePath("input", "Input file");
90 if (!result) { printHelp(); return printError(result.error()); }
91 input = result.value();
92 }
93 {
94 auto result = requirePath("output", "Output file");
95 if (!result) { printHelp(); return printError(result.error()); }
96 output = result.value();
97 }
98 {
99 auto result = requirePath("config", "Config file");
100 if (!result) { printHelp(); return printError(result.error()); }
101 configFile = result.value();
102 }
103
104 if (!fileExists(input)) {
105 return printError(CCSDS::Error{static_cast<CCSDS::ErrorCode>(ARG_PARSE_ERROR),
106 "Input file does not exist: " + input});
107 }
108 if (!fileExists(configFile)) {
109 return printError(CCSDS::Error{static_cast<CCSDS::ErrorCode>(ARG_PARSE_ERROR),
110 "Config file does not exist: " + configFile});
111 }
112
113 customConsole(appName, "reading CCSDS configuration file: " + configFile);
114 Config cfg;
115 if (const auto result = cfg.load(configFile); !result) return printError(result.error());
116
117 CCSDS::Packet templatePacket;
118 if (const auto result = templatePacket.loadFromConfig(cfg); !result) {
119 return printError(result.error());
120 }
121
122 if (const auto it = args.find("packet-error-control"); it != args.end()) {
123 const auto result = parsePacketErrorControlMode(it->second);
124 if (!result) return printError(result.error());
125 templatePacket.setPacketErrorControlMode(result.value());
126 }
127
128 CCSDS::Manager manager;
129 if (const auto result = manager.setPacketTemplate(std::move(templatePacket)); !result) {
130 return printError(result.error());
131 }
132 if (const auto result = configureManagerFraming(manager, cfg); !result) {
133 return printError(result.error());
134 }
135
136 customConsole(appName,
137 std::string("packet error control: ")
138 + packetErrorControlModeName(manager.getTemplate().getPacketErrorControlMode()));
139
140 std::vector<std::uint8_t> inputBytes;
141 customConsole(appName, "reading data from " + input);
142 {
143 const auto result = readBinaryFile(input);
144 if (!result) return printError(result.error());
145 inputBytes = result.value();
146 }
147
148 if (manager.getTemplate().getPrimaryHeader().getSequenceFlags() == CCSDS::UNSEGMENTED
149 && inputBytes.size() > manager.getDataFieldSize()) {
150 return printError(CCSDS::Error{
151 static_cast<CCSDS::ErrorCode>(INVALID_INPUT_DATA),
152 "Input contains " + std::to_string(inputBytes.size())
153 + " bytes, exceeding the unsegmented packet capacity of "
154 + std::to_string(manager.getDataFieldSize())});
155 }
156
157 customConsole(appName, "generating CCSDS packets using input data");
158 if (const auto result = manager.setApplicationData(inputBytes); !result) {
159 return printError(result.error());
160 }
161
162 if (args["verbose"] == "true") printPackets(manager);
163
164 const auto packets = manager.getPacketsBuffer();
165 if (!manager.getPackets().empty() && packets.empty()) {
166 return printError(CCSDS::Error{CCSDS::ErrorCode::INVALID_HEADER_DATA,
167 "Unable to serialize generated packet stream"});
168 }
169
170 customConsole(appName, "writing data to " + output);
171 if (const auto result = writeBinaryFile(packets, output); !result) {
172 return printError(result.error());
173 }
174
175 const auto duration = std::chrono::duration_cast<std::chrono::microseconds>(
176 std::chrono::high_resolution_clock::now() - start);
177 customConsole(appName, "execution time: " + std::to_string(duration.count()) + " [us]");
178 customConsole(appName, "[ Exit code 0 ]");
179 return 0;
180}
