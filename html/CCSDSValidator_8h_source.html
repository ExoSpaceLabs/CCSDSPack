52 class ValidationReport {
53 public:
54 static constexpr std::size_t Capacity{32U};
55
56 [[nodiscard]] bool valid() const noexcept {
57 for (std::size_t i = 0U; i < m_size; ++i) if (!m_checks[i].passed) return false;
58 return true;
59 }
60 [[nodiscard]] explicit operator bool() const noexcept { return valid(); }
61 [[nodiscard]] std::size_t size() const noexcept { return m_size; }
62 [[nodiscard]] bool contains(const ValidationCode code) const noexcept {
63 for (std::size_t i = 0U; i < m_size; ++i) if (m_checks[i].code == code) return true;
64 return false;
65 }
66 [[nodiscard]] bool passed(const ValidationCode code) const noexcept {
67 for (std::size_t i = 0U; i < m_size; ++i)
68 if (m_checks[i].code == code) return m_checks[i].passed;
69 return false;
70 }
71 [[nodiscard]] bool failed(const ValidationCode code) const noexcept {
72 return contains(code) && !passed(code);
73 }
74 [[nodiscard]] const ValidationCheck *begin() const noexcept { return m_checks.data(); }
75 [[nodiscard]] const ValidationCheck *end() const noexcept { return m_checks.data() + m_size; }
76
77 private:
78 void set(const ValidationCode code, const bool passed) noexcept {
79 for (std::size_t i = 0U; i < m_size; ++i) {
80 if (m_checks[i].code == code) {
81 m_checks[i].passed = m_checks[i].passed && passed;
82 return;
83 }
84 }
85 if (m_size < Capacity) m_checks[m_size++] = {code, passed};
86 }
87 std::array<ValidationCheck, Capacity> m_checks{};
88 std::size_t m_size{0U};
89 friend class Validator;
90 };
