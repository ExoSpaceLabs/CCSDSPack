21uint16_t crc16(
22 const std::vector<std::uint8_t> &data, const std::uint16_t polynomial, const std::uint16_t initialValue,
23 const std::uint16_t finalXorValue) {
24 std::uint16_t crc = initialValue;
25
26 for (const auto &byte: data) {
27 crc ^= static_cast<std::uint16_t>(byte) << 8; // Align byte with MSB of 16-bit CRC
28 for (std::int32_t i = 0; i < 8; ++i) {
29 // Process each bit
30 if (crc & 0x8000) {
31 // Check if MSB is set
32 crc = (crc << 1) ^ polynomial; // Shift and XOR with polynomial
33 } else {
34 crc = crc << 1; // Shift only
35 }
36 }
37 }
38
39 return crc ^ finalXorValue; // Apply final XOR
40}

47std::string getBinaryString(const std::uint32_t value, const std::int32_t bits) {
48 std::string binaryString;
49 // Calculate the minimum number of bits required to represent in groups of 4
50 const std::int32_t paddedBits = ((bits + 3) / 4) * 4; // Round up to the nearest multiple of 4
51
52 for (std::int32_t i = paddedBits - 1; i >= 0; --i) {
53 binaryString += ((value >> i) & 1) ? '1' : '0';
54
55 // Add a space after every 4 bits, except at the end
56 if (i % 4 == 0 && i != 0) {
57 binaryString += ' ';
58 }
59 }
60 return binaryString;
61}

216ccsds::ResultBool writeBinaryFile(const std::vector<std::uint8_t>& data, const std::string& filename) {
217 RET_IF_ERR_MSG(filename.empty(),ccsds::ErrorCode::FILE_WRITE_ERROR, "No filename provided");
218 RET_IF_ERR_MSG(data.empty(),ccsds::ErrorCode::FILE_WRITE_ERROR, "No data provided");
219 std::ofstream out(filename, std::ios::binary);
220
221 RET_IF_ERR_MSG(!out,ccsds::ErrorCode::FILE_WRITE_ERROR, "Failed to open file for writing");
222
223 // Write the entire vector data to the file in one go
224 out.write(reinterpret_cast<const char*>(data.data()), static_cast<std::streamsize>(data.size()));
225 RET_IF_ERR_MSG(!out,ccsds::ErrorCode::FILE_WRITE_ERROR, "Failed to write the data to the file");
226
227 return true;
228}

230ccsds::ResultBuffer readBinaryFile(const std::string& filename) {
231 RET_IF_ERR_MSG(filename.empty(),ccsds::ErrorCode::FILE_READ_ERROR, "No filename provided");
232
233 std::ifstream in(filename, std::ios::binary | std::ios::ate);
234 RET_IF_ERR_MSG(!in,ccsds::ErrorCode::FILE_READ_ERROR, "Failed to open file for reading");
235
236 // Get the file size using the 'ate' flag (seeks to the end automatically)
237 const std::streamsize size = in.tellg();
238 in.seekg(0, std::ios::beg);
239
240 // Read the entire file content into the vector
241 std::vector<std::uint8_t> data(size);
242 in.read(reinterpret_cast<char*>(data.data()), size);
243
244 RET_IF_ERR_MSG(!in,ccsds::ErrorCode::FILE_READ_ERROR, "Failed to read the entire file");
245 return data;
246}
