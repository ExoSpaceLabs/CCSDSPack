11ccsds_status_t ccsds_packet_stream_write_prefix(const int sync_enabled,
12 const uint32_t sync_pattern,
13 uint8_t *output,
14 const size_t capacity,
15 size_t *written_out) {
16 const size_t required = ccsds_packet_stream_prefix_size(sync_enabled);
17
18 if (written_out == NULL) return CCSDS_STATUS_NULL_POINTER;
19 if (capacity < required) return CCSDS_STATUS_BUFFER_TOO_SMALL;
20 if (required == 0U) {
21 *written_out = 0U;
22 return CCSDS_STATUS_OK;
23 }
24 if (output == NULL) return CCSDS_STATUS_NULL_POINTER;
25
26 ccsds_store_be_uint(output, (uint8_t)CCSDS_PACKET_STREAM_SYNC_SIZE,
27 (uint64_t)sync_pattern);
28 *written_out = required;
29 return CCSDS_STATUS_OK;
30}

32ccsds_status_t ccsds_packet_stream_init(ccsds_packet_stream_t *stream,
33 const uint8_t *data,
34 const size_t size,
35 const ccsds_packet_error_control_t error_control,
36 const ccsds_crc16_config_t *crc_config,
37 const int sync_enabled,
38 const uint32_t sync_pattern) {
39 if (stream == NULL || data == NULL) {
40 return CCSDS_STATUS_NULL_POINTER;
41 }
42 if (size == 0U) {
43 return CCSDS_STATUS_NO_DATA;
44 }
45 if (error_control != CCSDS_PACKET_ERROR_CONTROL_NONE
46 && error_control != CCSDS_PACKET_ERROR_CONTROL_CRC16) {
47 return CCSDS_STATUS_INVALID_DATA;
48 }
49
50 stream->data = data;
51 stream->size = size;
52 stream->offset = 0U;
53 stream->sync_pattern = sync_pattern;
54 stream->error_control = error_control;
55 stream->crc_config =
56 crc_config != NULL ? *crc_config : ccsds_crc16_default_config();
57 stream->sync_enabled = (uint8_t)(sync_enabled != 0);
58 return CCSDS_STATUS_OK;
59}

68ccsds_status_t ccsds_packet_stream_next(ccsds_packet_stream_t *stream,
69 ccsds_packet_view_t *view_out,
70 size_t *frame_consumed_out) {
71 size_t packet_offset;
72 size_t frame_size;
73 ccsds_packet_view_t view;
74 ccsds_status_t status;
75
76 if (stream == NULL || view_out == NULL || frame_consumed_out == NULL) {
77 return CCSDS_STATUS_NULL_POINTER;
78 }
79 if (stream->data == NULL || stream->offset > stream->size) {
80 return CCSDS_STATUS_INVALID_DATA;
81 }
82 if (stream->offset == stream->size) {
83 return CCSDS_STATUS_NO_DATA;
84 }
85
86 packet_offset = stream->offset;
87 if (stream->sync_enabled != 0U) {
88 if (stream->size - packet_offset < 4U) {
89 return CCSDS_STATUS_INVALID_DATA;
90 }
91 if ((uint32_t)ccsds_load_be_uint(stream->data + packet_offset, 4U) != stream->sync_pattern) {
92 return CCSDS_STATUS_INVALID_DATA;
93 }
94 packet_offset += 4U;
95 }
96
97 status = ccsds_packet_view_parse(
98 stream->data + packet_offset,
99 stream->size - packet_offset,
100 stream->error_control,
101 &stream->crc_config,
102 &view);
103 if (status != CCSDS_STATUS_OK) {
104 return status;
105 }
106
107 frame_size = (packet_offset - stream->offset) + view.consumed;
108 stream->offset += frame_size;
109 *view_out = view;
110 *frame_consumed_out = frame_size;
111 return CCSDS_STATUS_OK;
112}
