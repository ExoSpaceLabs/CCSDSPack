7ccsds_status_t ccsds_primary_header_validate(const ccsds_primary_header_t *header) {
8 if (header == NULL) {
9 return CCSDS_STATUS_NULL_POINTER;
10 }
11 if (header->version_number > 0x07U
12 || header->type > 0x01U
13 || header->secondary_header_flag > 0x01U
14 || header->apid > CCSDS_IDLE_APID
15 || header->sequence_flags > 0x03U
16 || header->sequence_count > CCSDS_SEQUENCE_COUNT_MAX) {
17 return CCSDS_STATUS_INVALID_HEADER_DATA;
18 }
19 return CCSDS_STATUS_OK;
20}

22ccsds_status_t ccsds_primary_header_decode(const uint8_t *data,
23 const size_t size,
24 ccsds_primary_header_t *header_out) {
25 uint16_t packet_id;
26 uint16_t sequence;
27
28 if (data == NULL || header_out == NULL) {
29 return CCSDS_STATUS_NULL_POINTER;
30 }
31 if (size < CCSDS_PRIMARY_HEADER_SIZE) {
32 return CCSDS_STATUS_INVALID_HEADER_DATA;
33 }
34
35 packet_id = ccsds_load_be16(data);
36 sequence = ccsds_load_be16(data + 2U);
37
38 header_out->version_number = (uint8_t)((packet_id >> 13U) & 0x07U);
39 header_out->type = (uint8_t)((packet_id >> 12U) & 0x01U);
40 header_out->secondary_header_flag = (uint8_t)((packet_id >> 11U) & 0x01U);
41 header_out->apid = (uint16_t)(packet_id & 0x07FFU);
42 header_out->sequence_flags = (uint8_t)((sequence >> 14U) & 0x03U);
43 header_out->sequence_count = (uint16_t)(sequence & CCSDS_SEQUENCE_COUNT_MAX);
44 header_out->data_length = ccsds_load_be16(data + 4U);
45
46 return CCSDS_STATUS_OK;
47}

49ccsds_status_t ccsds_primary_header_encode(const ccsds_primary_header_t *header,
50 uint8_t *data_out,
51 const size_t capacity) {
52 uint16_t packet_id;
53 uint16_t sequence;
54 ccsds_status_t status;
55
56 if (data_out == NULL) {
57 return CCSDS_STATUS_NULL_POINTER;
58 }
59 if (capacity < CCSDS_PRIMARY_HEADER_SIZE) {
60 return CCSDS_STATUS_BUFFER_TOO_SMALL;
61 }
62
63 status = ccsds_primary_header_validate(header);
64 if (status != CCSDS_STATUS_OK) {
65 return status;
66 }
67
68 packet_id = (uint16_t)(((uint16_t)header->version_number << 13U)
69 | ((uint16_t)header->type << 12U)
70 | ((uint16_t)header->secondary_header_flag << 11U)
71 | header->apid);
72 sequence = (uint16_t)(((uint16_t)header->sequence_flags << 14U)
73 | header->sequence_count);
74
75 ccsds_store_be16(data_out, packet_id);
76 ccsds_store_be16(data_out + 2U, sequence);
77 ccsds_store_be16(data_out + 4U, header->data_length);
78
79 return CCSDS_STATUS_OK;
80}

82ccsds_status_t ccsds_primary_header_pack(const ccsds_primary_header_t *header,
83 uint64_t *packed_out) {
84 uint8_t bytes[CCSDS_PRIMARY_HEADER_SIZE];
85 ccsds_status_t status;
86
87 if (packed_out == NULL) {
88 return CCSDS_STATUS_NULL_POINTER;
89 }
90

92 if (status != CCSDS_STATUS_OK) {

100ccsds_status_t ccsds_primary_header_unpack(const uint64_t packed,
101 ccsds_primary_header_t *header_out) {
102 uint8_t bytes[CCSDS_PRIMARY_HEADER_SIZE];
103
104 if (header_out == NULL) {
105 return CCSDS_STATUS_NULL_POINTER;
106 }
107 if (packed > UINT64_C(0xFFFFFFFFFFFF)) {
108 return CCSDS_STATUS_INVALID_HEADER_DATA;
109 }
110
111 ccsds_store_be_uint(bytes, (uint8_t)CCSDS_PRIMARY_HEADER_SIZE, packed);
112
113 return ccsds_primary_header_decode(bytes, sizeof(bytes), header_out);
114}

116ccsds_status_t ccsds_packet_declared_size(const uint8_t *data,
117 const size_t size,
118 size_t *packet_size_out) {
119 ccsds_primary_header_t header;
120 ccsds_status_t status;
121
122 if (packet_size_out == NULL) {
123 return CCSDS_STATUS_NULL_POINTER;
124 }
125
126 status = ccsds_primary_header_decode(data, size, &header);
127 if (status != CCSDS_STATUS_OK) {
128 return status;
129 }
130 if (header.version_number != 0U) {
131 return CCSDS_STATUS_INVALID_HEADER_DATA;
132 }
133
134 *packet_size_out = CCSDS_PRIMARY_HEADER_SIZE
135 + (size_t)header.data_length
136 + (size_t)1U;
137 return CCSDS_STATUS_OK;
138}
