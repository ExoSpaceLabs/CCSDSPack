7void ccsds_validation_report_reset(ccsds_validation_report_t *report) {
8 size_t index;
9 if (report == NULL) {
10 return;
11 }
12 for (index = 0U; index < CCSDS_VALIDATION_REPORT_CAPACITY; ++index) {
13 report->checks[index].code = 0U;
14 report->checks[index].passed = 0U;
15 }
16 report->size = 0U;
17}
18
19ccsds_status_t ccsds_validation_report_set(ccsds_validation_report_t *report,
20 const ccsds_validation_code_t code,
21 const int passed) {
22 size_t index;
23 if (report == NULL) {
24 return CCSDS_STATUS_NULL_POINTER;
25 }
26 if (code >= CCSDS_VALIDATION_CODE_COUNT) {
27 return CCSDS_STATUS_INVALID_DATA;
28 }
29
30 for (index = 0U; index < report->size; ++index) {
31 if (report->checks[index].code == code) {
32 report->checks[index].passed =
33 (uint8_t)(report->checks[index].passed != 0U && passed != 0);
34 return CCSDS_STATUS_OK;
35 }
36 }
37
38 if (report->size >= CCSDS_VALIDATION_REPORT_CAPACITY) {
39 return CCSDS_STATUS_BUFFER_TOO_SMALL;
40 }
41
42 report->checks[report->size].code = code;
43 report->checks[report->size].passed = (uint8_t)(passed != 0);
44 ++report->size;
45 return CCSDS_STATUS_OK;
46}

48int ccsds_validation_report_valid(const ccsds_validation_report_t *report) {
49 size_t index;
50 if (report == NULL) {
51 return 0;
52 }
53 for (index = 0U; index < report->size; ++index) {
54 if (report->checks[index].passed == 0U) {
55 return 0;
56 }
57 }
58 return 1;
59}

61int ccsds_validation_report_contains(const ccsds_validation_report_t *report,
62 const ccsds_validation_code_t code) {
63 size_t index;
64 if (report == NULL || code >= CCSDS_VALIDATION_CODE_COUNT) {
65 return 0;
66 }
67 for (index = 0U; index < report->size; ++index) {
68 if (report->checks[index].code == code) {
69 return 1;
70 }
71 }
72 return 0;
73}

75int ccsds_validation_report_passed(const ccsds_validation_report_t *report,
76 const ccsds_validation_code_t code) {
77 size_t index;
78 if (report == NULL || code >= CCSDS_VALIDATION_CODE_COUNT) {
79 return 0;
80 }
81 for (index = 0U; index < report->size; ++index) {
82 if (report->checks[index].code == code) {
83 return report->checks[index].passed != 0U;
84 }
85 }
86 return 0;
87}

89int ccsds_validation_report_failed(const ccsds_validation_report_t *report,
90 const ccsds_validation_code_t code) {
91 return ccsds_validation_report_contains(report, code)
92 && !ccsds_validation_report_passed(report, code);

149int ccsds_sequence_flags_valid(const ccsds_sequence_validator_t *validator,
150 const uint8_t sequence_flags) {
151 const int open = ccsds_sequence_validator_segment_open(validator);
152 switch (sequence_flags) {

154 case 1U:

156 case 0U:
157 case 2U:
158 return open;

160 return 0;

164int ccsds_sequence_count_valid(const ccsds_sequence_validator_t *validator,
165 const uint16_t sequence_count) {
166 if (validator == NULL || sequence_count > CCSDS_SEQUENCE_COUNT_MAX) {
167 return 0;
168 }
169 return !ccsds_sequence_validator_initialized(validator)
170 || sequence_count == ccsds_sequence_validator_expected_count(validator);

173ccsds_status_t ccsds_sequence_validator_accept(ccsds_sequence_validator_t *validator,
174 const ccsds_primary_header_t *header) {
175 uint16_t state;
176 uint16_t next;
177
178 if (validator == NULL || header == NULL) {
179 return CCSDS_STATUS_NULL_POINTER;
180 }
181 if (ccsds_primary_header_validate(header) != CCSDS_STATUS_OK) {
182 return CCSDS_STATUS_INVALID_HEADER_DATA;
183 }
184
185 next = (uint16_t)((header->sequence_count + 1U) & CCSDS_SEQUENCE_COUNT_MASK);
186 state = (uint16_t)(CCSDS_SEQUENCE_INITIALIZED_MASK | next);
187 if (header->sequence_flags == 1U || header->sequence_flags == 0U) {
188 state = (uint16_t)(state | CCSDS_SEQUENCE_OPEN_MASK);
189 }
190 validator->state = state;
191 return CCSDS_STATUS_OK;
192}

210ccsds_status_t ccsds_validate_packet_coherence(
211 const ccsds_packet_coherence_input_t *input,
212 const ccsds_sequence_validator_t *sequence,
213 const int validate_sequence_count,
214 ccsds_validation_report_t *report) {
215 ccsds_status_t status;
216 size_t packet_data_field_size;
217 int primary_valid;
218 int sequence_flags_valid;
219
220 if (input == NULL || sequence == NULL || report == NULL) {
221 return CCSDS_STATUS_NULL_POINTER;
222 }
223
224 primary_valid =
225 input->primary_header_state_valid != 0U
226 && ccsds_primary_header_validate(&input->header) == CCSDS_STATUS_OK;
227 status = set_check(report, CCSDS_VALIDATION_PRIMARY_HEADER, primary_valid);
228 if (status != CCSDS_STATUS_OK || !primary_valid) {
229 return status;
230 }
231
232 status = set_check(report, CCSDS_VALIDATION_PACKET_VERSION,
233 input->header.version_number == 0U);
234 if (status != CCSDS_STATUS_OK) return status;
235
236 packet_data_field_size =
237 input->serialized_size >= CCSDS_PRIMARY_HEADER_SIZE
238 ? input->serialized_size - CCSDS_PRIMARY_HEADER_SIZE
239 : 0U;
240 status = set_check(
241 report,
242 CCSDS_VALIDATION_PACKET_DATA_LENGTH,
243 packet_data_field_size > 0U
244 && (size_t)input->header.data_length == packet_data_field_size - 1U);
245 if (status != CCSDS_STATUS_OK) return status;
246
247 if (input->crc_checked != 0U) {
248 status = set_check(report, CCSDS_VALIDATION_CRC16,
249 input->crc_valid != 0U);
250 if (status != CCSDS_STATUS_OK) return status;
251 }
252
253 status = set_check(
254 report,
255 CCSDS_VALIDATION_SECONDARY_HEADER_PRESENCE,
256 (input->header.secondary_header_flag != 0U)
257 == (input->secondary_header_present != 0U));
258 if (status != CCSDS_STATUS_OK) return status;
259
260 if (input->secondary_header_present != 0U
261 && input->secondary_direction != CCSDS_PACKET_DIRECTION_UNSPECIFIED) {
262 status = set_check(
263 report,
264 CCSDS_VALIDATION_SECONDARY_HEADER_DIRECTION,
265 direction_valid(input->secondary_direction)
266 && input->header.type
267 == packet_type_for_direction(input->secondary_direction));
268 if (status != CCSDS_STATUS_OK) return status;
269 }
270
271 sequence_flags_valid = ccsds_sequence_flags_valid(
272 sequence, input->header.sequence_flags);
273 status = set_check(report, CCSDS_VALIDATION_SEQUENCE_FLAGS,
274 sequence_flags_valid);
275 if (status != CCSDS_STATUS_OK) return status;
276
277 if (validate_sequence_count != 0) {
278 status = set_check(
279 report,
280 CCSDS_VALIDATION_SEQUENCE_COUNT,
281 ccsds_sequence_count_valid(sequence, input->header.sequence_count));
282 if (status != CCSDS_STATUS_OK) return status;
283 }
284
285 return CCSDS_STATUS_OK;
286}

289ccsds_status_t ccsds_validate_pus_coherence(
290 const ccsds_pus_coherence_input_t *input,
291 ccsds_validation_report_t *report) {
292 ccsds_status_t status;
293 int revision_valid;
294 int direction_is_valid;
295 int size_valid;
296 int reserved_valid = 0;
297
298 if (input == NULL || report == NULL) return CCSDS_STATUS_NULL_POINTER;
299
300 status = set_check(report, CCSDS_VALIDATION_PUS_HEADER, 1);
301 if (status != CCSDS_STATUS_OK) return status;
302
303 revision_valid = input->revision == 1U || input->revision == 2U;
304 direction_is_valid = direction_valid(input->direction);
305
306 status = set_check(report, CCSDS_VALIDATION_PUS_REVISION, revision_valid);
307 if (status != CCSDS_STATUS_OK) return status;
308 status = set_check(report, CCSDS_VALIDATION_PUS_DIRECTION, direction_is_valid);
309 if (status != CCSDS_STATUS_OK) return status;
310 status = set_check(
311 report, CCSDS_VALIDATION_PUS_PACKET_TYPE,
312 direction_is_valid
313 && input->packet_type == packet_type_for_direction(input->direction));
314 if (status != CCSDS_STATUS_OK) return status;
315 status = set_check(report, CCSDS_VALIDATION_PUS_TAILORING,
316 input->tailoring_valid != 0U);
317 if (status != CCSDS_STATUS_OK) return status;
318
319 size_valid = input->serialized.data != NULL
320 && input->serialized.size != 0U
321 && input->serialized.size == input->expected_size;
322 status = set_check(report, CCSDS_VALIDATION_PUS_SECONDARY_HEADER_SIZE,
323 size_valid);
324 if (status != CCSDS_STATUS_OK) return status;
325
326 if (size_valid) {
327 if (input->revision == 1U
328 && input->direction == CCSDS_PACKET_DIRECTION_TELECOMMAND) {
329 reserved_valid = (input->serialized.data[0] & 0x80U) == 0U
330 && ((input->serialized.data[0] >> 4U) & 0x07U) == 1U;
331 } else if (input->revision == 1U
332 && input->direction == CCSDS_PACKET_DIRECTION_TELEMETRY) {
333 reserved_valid = input->serialized.data[0] == 0x10U;
334 } else if (input->revision == 2U) {
335 reserved_valid = (input->serialized.data[0] >> 4U) == 2U;
336 }
337 }
338 status = set_check(report, CCSDS_VALIDATION_PUS_RESERVED_BITS,
339 size_valid && reserved_valid);
340 if (status != CCSDS_STATUS_OK) return status;
341 status = set_check(
342 report, CCSDS_VALIDATION_PUS_SPARE_FIELDS,
343 size_valid && ccsds_pus_spare_is_zero(
344 input->serialized.data, input->serialized.size, input->spare_octets));
345 if (status != CCSDS_STATUS_OK) return status;
346
347 if (input->direction == CCSDS_PACKET_DIRECTION_TELECOMMAND) {
348 status = set_check(report, CCSDS_VALIDATION_PUS_ACKNOWLEDGEMENT,
349 input->acknowledgement_flags <= 0x0FU);
350 if (status != CCSDS_STATUS_OK) return status;
351 return set_check(
352 report, CCSDS_VALIDATION_PUS_SOURCE_ID,
353 ccsds_pus_identifier_fits(
354 input->identifier_value, input->identifier_octets));
355 }
356
357 if (input->direction == CCSDS_PACKET_DIRECTION_TELEMETRY) {
358 status = set_check(
359 report, CCSDS_VALIDATION_PUS_DESTINATION_ID,
360 ccsds_pus_identifier_fits(
361 input->identifier_value, input->identifier_octets));
362 if (status != CCSDS_STATUS_OK) return status;
363 status = set_check(
364 report, CCSDS_VALIDATION_PUS_TIMESTAMP,
365 input->timestamp_present != 0U
366 ? input->timestamp_valid != 0U
367 : input->timestamp_zero_when_absent != 0U);
368 if (status != CCSDS_STATUS_OK) return status;
369
370 if (input->revision == 1U) {
371 return set_check(
372 report, CCSDS_VALIDATION_PUS_PACKET_SUBCOUNTER,
373 input->packet_subcounter_present != 0U
374 || input->packet_subcounter == 0U);
375 }
376 if (input->revision == 2U) {
377 return set_check(
378 report, CCSDS_VALIDATION_PUS_TIME_REFERENCE_STATUS,
379 input->time_reference_status <= 0x0FU);
380 }
381 }
382
383 return CCSDS_STATUS_OK;
384}

386ccsds_status_t ccsds_validate_template_coherence(
387 const ccsds_primary_header_t *packet_header,
388 const ccsds_template_coherence_input_t *template_input,
389 ccsds_validation_report_t *report) {
390 ccsds_status_t status;
391 int template_header_valid;
392 int packet_identifier_valid;
393 int segmentation_class_valid = 0;
394
395 if (packet_header == NULL || template_input == NULL || report == NULL) {
396 return CCSDS_STATUS_NULL_POINTER;
397 }
398
399 template_header_valid =
400 template_input->primary_header_state_valid != 0U
401 && ccsds_primary_header_validate(&template_input->header)
402 == CCSDS_STATUS_OK;
403
404 packet_identifier_valid =
405 template_header_valid
406 && template_input->header.version_number == packet_header->version_number
407 && template_input->header.type == packet_header->type
408 && template_input->header.secondary_header_flag
409 == packet_header->secondary_header_flag
410 && template_input->header.apid == packet_header->apid;
411 status = set_check(report, CCSDS_VALIDATION_PACKET_IDENTIFIER,
412 packet_identifier_valid);
413 if (status != CCSDS_STATUS_OK) return status;
414
415 if (template_header_valid) {
416 segmentation_class_valid =
417 template_input->header.sequence_flags == 3U
418 ? packet_header->sequence_flags == 3U
419 : packet_header->sequence_flags != 3U;
420 }
421 status = set_check(report, CCSDS_VALIDATION_SEGMENTATION_CLASS,
422 segmentation_class_valid);
423 if (status != CCSDS_STATUS_OK) return status;
424
425 status = set_check(
426 report,
427 CCSDS_VALIDATION_TEMPLATE_PACKET_ERROR_CONTROL,
428 template_input->packet_error_control_equal != 0U);
429 if (status != CCSDS_STATUS_OK) return status;
430
431 return set_check(
432 report,
433 CCSDS_VALIDATION_TEMPLATE_SECONDARY_HEADER,
434 template_input->secondary_contract_equal != 0U);
435}
