6ccsds_status_t ccsds_segmentation_packet_count(const size_t total_size,
7 const size_t max_data_size,
8 size_t *packet_count_out) {
9 if (packet_count_out == NULL) {
10 return CCSDS_STATUS_NULL_POINTER;
11 }
12 if (total_size == 0U) {
13 return CCSDS_STATUS_NO_DATA;
14 }
15 if (max_data_size == 0U) {
16 return CCSDS_STATUS_INVALID_APPLICATION_DATA;
17 }
18
19 *packet_count_out = 1U + ((total_size - 1U) / max_data_size);
20 return CCSDS_STATUS_OK;
21}

39ccsds_status_t ccsds_segmentation_plan(const size_t total_size,
40 const size_t max_data_size,
41 const size_t packet_index,
42 const uint16_t start_sequence_count,
43 const int auto_sequence,
44 ccsds_segment_plan_t *plan_out) {
45 size_t packet_count;
46 size_t offset;
47 size_t remaining;
48 ccsds_status_t status;
49 ccsds_segment_plan_t plan;
50
51 if (plan_out == NULL) {
52 return CCSDS_STATUS_NULL_POINTER;
53 }
54 if (start_sequence_count > CCSDS_SEQUENCE_COUNT_MAX) {
55 return CCSDS_STATUS_INVALID_HEADER_DATA;
56 }
57
58 status = ccsds_segmentation_packet_count(
59 total_size, max_data_size, &packet_count);
60 if (status != CCSDS_STATUS_OK) {
61 return status;
62 }
63 if (packet_index >= packet_count) {
64 return CCSDS_STATUS_INVALID_DATA;
65 }
66
67 offset = packet_index * max_data_size;
68 remaining = total_size - offset;
69
70 plan.packet_count = packet_count;
71 plan.packet_index = packet_index;
72 plan.offset = offset;
73 plan.size = remaining < max_data_size ? remaining : max_data_size;
74
75 if (packet_count == 1U) {
76 plan.sequence_flags = CCSDS_SEQUENCE_UNSEGMENTED;
77 } else if (packet_index == 0U) {
78 plan.sequence_flags = CCSDS_SEQUENCE_FIRST;
79 } else if (packet_index + 1U == packet_count) {
80 plan.sequence_flags = CCSDS_SEQUENCE_LAST;
81 } else {
82 plan.sequence_flags = CCSDS_SEQUENCE_CONTINUING;
83 }
84
85 plan.sequence_count = auto_sequence != 0
86 ? ccsds_sequence_after_packets(start_sequence_count, packet_index, 1)
87 : start_sequence_count;
88
89 *plan_out = plan;
90 return CCSDS_STATUS_OK;
91}
