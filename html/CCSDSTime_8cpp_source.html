7 std::uint64_t maximumForOctets(const std::uint8_t octets) {
8 return octets >= 8U ? UINT64_MAX : ((std::uint64_t{1U} << (octets * 8U)) - 1U);
9 }

11 void appendCounter(std::vector<std::uint8_t> &output,
12 const std::uint64_t value,
13 const std::uint8_t octets) {
14 for (std::uint8_t index = 0U; index < octets; ++index) {
15 const auto shift = static_cast<std::uint8_t>((octets - index - 1U) * 8U);
16 output.push_back(static_cast<std::uint8_t>((value >> shift) & 0xFFU));
17 }
18 }

20 std::uint64_t readCounter(const std::vector<std::uint8_t> &data,
21 const std::size_t offset,
22 const std::uint8_t octets) {
23 std::uint64_t value{};
24 for (std::uint8_t index = 0U; index < octets; ++index) {
25 value = (value << 8U) | data[offset + index];
26 }
27 return value;
28 }

30 std::uint8_t pField(const ccsds::time::CucConfiguration &configuration) {
31 const std::uint8_t epoch = configuration.epoch == ccsds::time::Epoch::Ccsds1958Tai
32 ? 0x10U : 0x20U;
33 return static_cast<std::uint8_t>(epoch
34 | ((configuration.coarseOctets - 1U) << 2U)
35 | configuration.fineOctets);
36 }

39ccsds::ResultBool ccsds::time::validate(const CucConfiguration &configuration) {
40 RET_IF_ERR_MSG(configuration.epoch != Epoch::Ccsds1958Tai
41 && configuration.epoch != Epoch::AgencyDefined,
42 ErrorCode::INVALID_DATA,
43 "CUC requires the CCSDS-1958 or an agency-defined epoch.");
44 RET_IF_ERR_MSG(configuration.pField != PFieldMode::Implicit
45 && configuration.pField != PFieldMode::Explicit,
46 ErrorCode::INVALID_DATA, "CUC P-field policy is invalid.");
47 RET_IF_ERR_MSG(configuration.coarseOctets < 1U || configuration.coarseOctets > 4U,
48 ErrorCode::INVALID_DATA,
49 "Basic CUC coarse time requires 1 to 4 octets.");
50 RET_IF_ERR_MSG(configuration.fineOctets > 3U, ErrorCode::INVALID_DATA,
51 "Basic CUC fine time requires 0 to 3 octets.");
52 return true;
53}

61ccsds::ResultBuffer ccsds::time::serialize(const CucTime &value,
62 const CucConfiguration &configuration) {
63 const auto validation = validate(configuration);
64 if (!validation) return validation.error();
65 RET_IF_ERR_MSG(value.coarse > maximumForOctets(configuration.coarseOctets),
66 ErrorCode::INVALID_DATA,
67 "CUC coarse time does not fit the configured width.");
68 RET_IF_ERR_MSG(value.fine > maximumForOctets(configuration.fineOctets),
69 ErrorCode::INVALID_DATA,
70 "CUC fine time does not fit the configured width.");
71
72 std::vector<std::uint8_t> output;
73 output.reserve(encodedSize(configuration));
74 if (configuration.pField == PFieldMode::Explicit) output.push_back(pField(configuration));
75 appendCounter(output, value.coarse, configuration.coarseOctets);
76 appendCounter(output, value.fine, configuration.fineOctets);
77 return output;
78}

80ccsds::Result<ccsds::time::CucTime> ccsds::time::deserialize(
81 const std::vector<std::uint8_t> &data,
82 const CucConfiguration &configuration) {
83 const auto validation = validate(configuration);
84 if (!validation) return validation.error();
85 RET_IF_ERR_MSG(data.size() != encodedSize(configuration), ErrorCode::INVALID_DATA,
86 "CUC encoded size does not match the configured layout.");
87
88 std::size_t offset{};
89 if (configuration.pField == PFieldMode::Explicit) {
90 RET_IF_ERR_MSG(data.front() != pField(configuration), ErrorCode::INVALID_DATA,
91 "CUC P-field does not match the configured epoch or widths.");
92 offset = 1U;
93 }
94
95 CucTime value;
96 value.coarse = readCounter(data, offset, configuration.coarseOctets);
97 value.fine = readCounter(data, offset + configuration.coarseOctets,
98 configuration.fineOctets);
99 return value;
100}
