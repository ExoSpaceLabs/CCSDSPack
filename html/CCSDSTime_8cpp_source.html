17 ccsds::Error cucError(const ccsds_status_t status) {
18 using ccsds::Error;
19 using ccsds::ErrorCode;
20 switch (status) {
21 case CCSDS_STATUS_CUC_INVALID_EPOCH:
22 return Error(ErrorCode::INVALID_DATA,
23 "CUC requires the CCSDS-1958 or an agency-defined epoch.");
24 case CCSDS_STATUS_CUC_INVALID_PFIELD_MODE:
25 return Error(ErrorCode::INVALID_DATA, "CUC P-field policy is invalid.");
26 case CCSDS_STATUS_CUC_INVALID_COARSE_WIDTH:
27 return Error(ErrorCode::INVALID_DATA,
28 "Basic CUC coarse time requires 1 to 4 octets.");
29 case CCSDS_STATUS_CUC_INVALID_FINE_WIDTH:
30 return Error(ErrorCode::INVALID_DATA,
31 "Basic CUC fine time requires 0 to 3 octets.");
32 case CCSDS_STATUS_CUC_COARSE_OVERFLOW:
33 return Error(ErrorCode::INVALID_DATA,
34 "CUC coarse time does not fit the configured width.");
35 case CCSDS_STATUS_CUC_FINE_OVERFLOW:
36 return Error(ErrorCode::INVALID_DATA,
37 "CUC fine time does not fit the configured width.");
38 case CCSDS_STATUS_CUC_SIZE_MISMATCH:
39 return Error(ErrorCode::INVALID_DATA,
40 "CUC encoded size does not match the configured layout.");
41 case CCSDS_STATUS_CUC_PFIELD_MISMATCH:
42 return Error(ErrorCode::INVALID_DATA,
43 "CUC P-field does not match the configured epoch or widths.");
44 case CCSDS_STATUS_NULL_POINTER:
45 return Error(ErrorCode::NULL_POINTER, "CUC core received a null pointer.");
46 default:
47 return Error(ErrorCode::INVALID_DATA, "CUC encoding or decoding failed.");
48 }
49 }
50}

64ccsds::ResultBuffer ccsds::time::serialize(const CucTime &value,
65 const CucConfiguration &configuration) {
66 const auto coreConfig = toCoreConfig(configuration);
67 const auto validation = ccsds_cuc_validate(&coreConfig);
68 if (validation != CCSDS_STATUS_OK) return cucError(validation);
69
70 const ccsds_cuc_time_t coreValue{value.coarse, value.fine};
71 std::vector<std::uint8_t> output(ccsds_cuc_encoded_size(&coreConfig));
72 std::size_t written = 0U;
73
74 const auto status = ccsds_cuc_encode(
75 &coreValue, &coreConfig, output.data(), output.size(), &written);
76 if (status != CCSDS_STATUS_OK) return cucError(status);
77 output.resize(written);
78 return output;
79}

81ccsds::Result<ccsds::time::CucTime> ccsds::time::deserialize(
82 const std::vector<std::uint8_t> &data,
83 const CucConfiguration &configuration) {
84 const auto coreConfig = toCoreConfig(configuration);
85 ccsds_cuc_time_t value{};
86
87 const auto status = ccsds_cuc_decode(data.data(), data.size(), &coreConfig, &value);
88 if (status != CCSDS_STATUS_OK) return cucError(status);
89 return CucTime{value.coarse, value.fine};
