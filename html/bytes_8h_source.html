37static inline uint64_t ccsds_load_uint(const uint8_t *data,
38 const uint8_t octets,
39 const ccsds_byte_order_t order) {
40 uint64_t value = 0U;
41 uint8_t index;
42
43 if (order == CCSDS_BYTE_ORDER_BIG) {
44 for (index = 0U; index < octets; ++index) {
45 value = (value << 8U) | (uint64_t)data[index];
46 }
47 } else {
48 for (index = 0U; index < octets; ++index) {
49 value |= (uint64_t)data[index] << ((uint64_t)index * UINT64_C(8));
50 }
51 }
52
53 return value;
54}

62static inline void ccsds_store_uint(uint8_t *data,
63 const uint8_t octets,
64 const uint64_t value,
65 const ccsds_byte_order_t order) {
66 uint8_t index;
67
68 if (order == CCSDS_BYTE_ORDER_BIG) {
69 for (index = 0U; index < octets; ++index) {
70 const uint8_t shift = (uint8_t)((octets - index - 1U) * 8U);
71 data[index] = (uint8_t)((value >> shift) & UINT64_C(0xFF));
72 }
73 } else {
74 for (index = 0U; index < octets; ++index) {
75 const uint8_t shift = (uint8_t)(index * 8U);
76 data[index] = (uint8_t)((value >> shift) & UINT64_C(0xFF));
77 }
78 }
79}
