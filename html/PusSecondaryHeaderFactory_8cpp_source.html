6ccsds::Result<std::shared_ptr<ccsds::SecondaryHeaderAbstract>>
7ccsds::pus::SecondaryHeaderFactory::create(const Revision revision,
8 const PacketDirection direction) const {
9 if (revision == Revision::A && direction == PacketDirection::Telecommand)
10 return std::static_pointer_cast<SecondaryHeaderAbstract>(std::make_shared<rev_a::TcHeader>());
11 if (revision == Revision::A && direction == PacketDirection::Telemetry)
12 return std::static_pointer_cast<SecondaryHeaderAbstract>(std::make_shared<rev_a::TmHeader>());
13 if (revision == Revision::C && direction == PacketDirection::Telecommand)
14 return std::static_pointer_cast<SecondaryHeaderAbstract>(std::make_shared<rev_c::TcHeader>());
15 if (revision == Revision::C && direction == PacketDirection::Telemetry)
16 return std::static_pointer_cast<SecondaryHeaderAbstract>(std::make_shared<rev_c::TmHeader>());
17 return Error{ErrorCode::INVALID_SECONDARY_HEADER_DATA,
18 "Unsupported PUS revision/direction selection."};
19}

21ccsds::Result<std::shared_ptr<ccsds::SecondaryHeaderAbstract>>
22ccsds::pus::SecondaryHeaderFactory::create(const std::string &value) const {
23 RET_IF_ERR_MSG(!typeIsSupported(value), ErrorCode::INVALID_SECONDARY_HEADER_DATA,
24 "Unsupported PUS secondary-header selector: " + value);
25 if (value == "PUS:revA:TC") return create(Revision::A, PacketDirection::Telecommand);
26 if (value == "PUS:revA:TM") return create(Revision::A, PacketDirection::Telemetry);
27 if (value == "PUS:revC:TC") return create(Revision::C, PacketDirection::Telecommand);
28 return create(Revision::C, PacketDirection::Telemetry);
29}
