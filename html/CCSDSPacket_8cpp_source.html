23 ccsds::ResultBool validatePacketForSerialization(
24 const ccsds::Header &header,
25 const ccsds::DataField &dataField) {
26 RET_IF_ERR_MSG(header.getHeaderStatus() == ccsds::INVALID,
27 ccsds::ErrorCode::INVALID_HEADER_DATA,
28 "Cannot serialize packet: primary header is invalid.");
29 RET_IF_ERR_MSG(header.getVersionNumber() != 0U,
30 ccsds::ErrorCode::INVALID_HEADER_DATA,
31 "Cannot serialize packet: unsupported CCSDS packet version.");
32
33 if (header.getAPID() == ccsds::IDLE_APID) {
34 RET_IF_ERR_MSG(header.getSecondaryHeaderFlag() != 0U,
35 ccsds::ErrorCode::INVALID_HEADER_DATA,
36 "Cannot serialize Idle Packet: secondary-header flag must be zero.");
37 RET_IF_ERR_MSG(dataField.getSecondaryHeaderFlag(),
38 ccsds::ErrorCode::INVALID_SECONDARY_HEADER_DATA,
39 "Cannot serialize Idle Packet with a secondary header.");
40 RET_IF_ERR_MSG(dataField.getApplicationDataBytesSize() == 0U,
41 ccsds::ErrorCode::INVALID_DATA,
42 "Cannot serialize Idle Packet without mission-defined idle user data.");
43 return true;
44 }
45
46 const auto secondary = dataField.getSecondaryHeader();
47 RET_IF_ERR_MSG((header.getSecondaryHeaderFlag() != 0U) != static_cast<bool>(secondary),
48 ccsds::ErrorCode::INVALID_SECONDARY_HEADER_DATA,
49 "Cannot serialize packet: primary-header flag and secondary-header state differ.");
50
51 if (secondary && secondary->getDirection() != ccsds::PacketDirection::Unspecified) {
52 RET_IF_ERR_MSG(header.getType()
53 != ccsds::packetTypeForDirection(secondary->getDirection()),
54 ccsds::ErrorCode::INVALID_HEADER_DATA,
55 "Cannot serialize packet: primary-header Packet Type differs from secondary-header direction.");
56 }
57 return true;
58 }

60 ccsds::Result<std::size_t> declaredPacketSize(const std::vector<std::uint8_t> &data) {
61 if (data.size() < 6U) {
62 return ccsds::Error{ccsds::ErrorCode::INVALID_HEADER_DATA,
63 "Cannot deserialize packet: truncated CCSDS primary header."};
64 }
65 const std::vector<std::uint8_t> headerData(data.begin(), data.begin() + 6);
66 ccsds::Header header;
67 const auto headerResult = header.deserialize(headerData);
68 if (!headerResult) return headerResult.error();
69 if (header.getVersionNumber() != 0U) {
70 return ccsds::Error{ccsds::ErrorCode::INVALID_HEADER_DATA,
71 "Cannot deserialize packet: unsupported CCSDS packet version."};
72 }
73 const auto packetSize = 6U + static_cast<std::size_t>(header.getDataLength()) + 1U;
74 if (data.size() < packetSize) {
75 return ccsds::Error{ccsds::ErrorCode::INVALID_DATA,
76 "Cannot deserialize packet: truncated packet body."};
77 }
78 return packetSize;
79 }

81 ccsds::Result<ParsedPacket> validatePacketBytes(
82 const std::vector<std::uint8_t> &headerData,
83 const std::vector<std::uint8_t> &packetData,
84 const ccsds::PacketErrorControlMode mode,
85 const ccsds::CRC16Config &crcConfig) {
86 if (headerData.size() != 6U) {
87 return ccsds::Error{ccsds::ErrorCode::INVALID_HEADER_DATA,
88 "Cannot deserialize packet: primary header must contain exactly six bytes."};
89 }
90
91 ParsedPacket parsed;
92 const auto headerResult = parsed.header.deserialize(headerData);
93 if (!headerResult) return headerResult.error();
94 if (parsed.header.getVersionNumber() != 0U) {
95 return ccsds::Error{ccsds::ErrorCode::INVALID_HEADER_DATA,
96 "Cannot deserialize packet: unsupported CCSDS packet version."};
97 }
98
99 const auto expectedDataSize = static_cast<std::size_t>(parsed.header.getDataLength()) + 1U;
100 if (packetData.size() != expectedDataSize) {
101 return ccsds::Error{ccsds::ErrorCode::INVALID_DATA,
102 "Cannot deserialize packet: packet body size does not match Packet Data Length."};
103 }
104
105 if (mode == ccsds::PacketErrorControlMode::CRC16) {
106 if (packetData.size() < 2U) {
107 return ccsds::Error{ccsds::ErrorCode::INVALID_DATA,
108 "Cannot deserialize packet: CRC16 mode requires two packet error-control bytes."};
109 }
110 parsed.receivedCRC = static_cast<std::uint16_t>(
111 (static_cast<std::uint16_t>(packetData[packetData.size() - 2U]) << 8U)
112 | packetData.back());
113 std::vector<std::uint8_t> crcInput = headerData;
114 crcInput.insert(crcInput.end(), packetData.begin(), packetData.end() - 2);
115 const auto expectedCRC = ccsds::crc16(crcInput, crcConfig.polynomial,
116 crcConfig.initialValue, crcConfig.finalXorValue);
117 if (expectedCRC != parsed.receivedCRC) {
118 return ccsds::Error{ccsds::ErrorCode::INVALID_CHECKSUM,
119 "Cannot deserialize packet: CRC16 packet error-control mismatch."};
120 }
121 parsed.dataField.assign(packetData.begin(), packetData.end() - 2);
122 } else {
123 parsed.receivedCRC = 0U;
124 parsed.dataField = packetData;
125 }
126
127 if (parsed.header.getAPID() == ccsds::IDLE_APID) {
128 if (parsed.header.getSecondaryHeaderFlag() != 0U) {
129 return ccsds::Error{ccsds::ErrorCode::INVALID_HEADER_DATA,
130 "Cannot deserialize Idle Packet: secondary-header flag must be zero."};
131 }
132 if (parsed.dataField.empty()) {
133 return ccsds::Error{ccsds::ErrorCode::INVALID_DATA,
134 "Cannot deserialize Idle Packet: mission-defined idle user data is required."};
135 }
136 }
137 return parsed;
138 }

155ccsds::ResultBool ccsds::Packet::update() {
156 FORWARD_RESULT(validatePacketForSerialization(m_primaryHeader, m_dataField));
157 if (m_updateStatus || !m_enableUpdatePacket) return true;
158
159 std::vector<std::uint8_t> dataField;
160 ASSIGN_MV(dataField, m_dataField.serialize());
161 const auto packetDataFieldSize = dataField.size() + getPacketErrorControlSize();
162 constexpr auto maximumPacketDataFieldSize =
163 static_cast<std::size_t>(std::numeric_limits<std::uint16_t>::max()) + 1U;
164
165 RET_IF_ERR_MSG(packetDataFieldSize == 0U, ErrorCode::INVALID_DATA,
166 "Cannot finalize packet: Packet Data Field is empty.");
167 RET_IF_ERR_MSG(packetDataFieldSize > maximumPacketDataFieldSize, ErrorCode::INVALID_DATA,
168 "Cannot finalize packet: Packet Data Field exceeds the CCSDS length field.");
169
170 m_primaryHeader.setDataLength(static_cast<std::uint16_t>(packetDataFieldSize - 1U));
171 FORWARD_RESULT(m_primaryHeader.setSequenceCount(m_sequenceCounter & SEQUENCE_COUNT_MASK));
172
173 if (getPacketErrorControlMode() == PacketErrorControlMode::CRC16) {
174 auto crcInput = static_cast<const Header &>(m_primaryHeader).serialize();
175 RET_IF_ERR_MSG(crcInput.size() != 6U, ErrorCode::INVALID_HEADER_DATA,
176 "Cannot finalize packet: primary-header serialization failed.");
177 crcInput.insert(crcInput.end(), dataField.begin(), dataField.end());
178 m_CRC16 = ccsds::crc16(crcInput, m_CRC16Config.polynomial,
179 m_CRC16Config.initialValue, m_CRC16Config.finalXorValue);
180 } else {
181 m_CRC16 = 0U;
182 }
183 m_updateStatus = true;
184 return true;
185}

194ccsds::ResultBool ccsds::Packet::loadFromConfig(const ccsds::Config &cfg) {
195 int versionNumber{};
196 int APID{};
197 bool segmented{};
198 bool configuredType{false};
199 const bool hasConfiguredType = cfg.isKey("ccsds_type");
200
201 RET_IF_ERR_MSG(!cfg.isKey("ccsds_version_number"), ErrorCode::CONFIG_FILE_ERROR,
202 "Config: Missing int field: ccsds_version_number");
203 RET_IF_ERR_MSG(!cfg.isKey("ccsds_APID"), ErrorCode::CONFIG_FILE_ERROR,
204 "Config: Missing int field: ccsds_APID");
205 RET_IF_ERR_MSG(!cfg.isKey("ccsds_segmented"), ErrorCode::CONFIG_FILE_ERROR,
206 "Config: Missing bool field: ccsds_segmented");
207
208 ASSIGN_CP(versionNumber, cfg.get<int>("ccsds_version_number"));
209 ASSIGN_CP(APID, cfg.get<int>("ccsds_APID"));
210 ASSIGN_CP(segmented, cfg.get<bool>("ccsds_segmented"));
211 if (hasConfiguredType) ASSIGN_CP(configuredType, cfg.get<bool>("ccsds_type"));
212
213 RET_IF_ERR_MSG(versionNumber != 0, ErrorCode::CONFIG_FILE_ERROR,
214 "Config: ccsds_version_number must be 0 for CCSDS 133.0-B-2 Space Packets");
215 RET_IF_ERR_MSG(APID < 0 || APID > static_cast<int>(IDLE_APID), ErrorCode::CONFIG_FILE_ERROR,
216 "Config: ccsds_APID must be between 0 and 2047");
217
218 FORWARD_RESULT(m_primaryHeader.setVersionNumber(static_cast<std::uint8_t>(versionNumber)));
219 if (hasConfiguredType) FORWARD_RESULT(m_primaryHeader.setType(static_cast<std::uint8_t>(configuredType)));
220 FORWARD_RESULT(m_primaryHeader.setAPID(static_cast<std::uint16_t>(APID)));
221
222 PacketErrorControlMode errorControl{};
223 ASSIGN_CP(errorControl, packetErrorControlFromConfig(cfg));
224 setPacketErrorControlMode(errorControl);
225
226 const auto sequenceFlag = segmented ? FIRST_SEGMENT : UNSEGMENTED;
227 const auto sequenceCount = static_cast<std::uint16_t>(segmented ? 1U : 0U);
228 FORWARD_RESULT(m_primaryHeader.setSequenceFlags(sequenceFlag));
229 FORWARD_RESULT(m_primaryHeader.setSequenceCount(sequenceCount));
230 m_sequenceCounter = (m_sequenceCounter & PACKET_ERROR_CONTROL_DISABLED_MASK)
231 | (sequenceCount & SEQUENCE_COUNT_MASK);
232
233 if (cfg.isKey("data_field_size")) {
234 int dataFieldSize{};
235 ASSIGN_CP(dataFieldSize, cfg.get<int>("data_field_size"));
236 RET_IF_ERR_MSG(dataFieldSize < 0 || dataFieldSize > std::numeric_limits<std::uint16_t>::max(),
237 ErrorCode::CONFIG_FILE_ERROR,
238 "Config: data_field_size must be between 0 and 65535");
239 m_dataField.setDataPacketSize(static_cast<std::uint16_t>(dataFieldSize));
240 }
241
242 bool defineSecondaryHeader{false};
243 if (cfg.isKey("define_secondary_header")) {
244 ASSIGN_CP(defineSecondaryHeader, cfg.get<bool>("define_secondary_header"));
245 }
246
247 if (defineSecondaryHeader) {
248 FORWARD_RESULT(m_dataField.setSecondaryHeader(cfg));
249 const auto secondary = m_dataField.getSecondaryHeader();
250 RET_IF_ERR_MSG(!secondary, ErrorCode::CONFIG_FILE_ERROR,
251 "Config: define_secondary_header did not create a header.");
252 const auto declaredDirection = secondary->getDirection();
253 if (hasConfiguredType && declaredDirection != PacketDirection::Unspecified) {
254 RET_IF_ERR_MSG(static_cast<std::uint8_t>(configuredType)
255 != packetTypeForDirection(declaredDirection),
256 ErrorCode::CONFIG_FILE_ERROR,
257 "Config: ccsds_type contradicts the concrete secondary-header direction.");
258 }
259 FORWARD_RESULT(setSecondaryHeader(secondary));
260 } else {
261 FORWARD_RESULT(setSecondaryHeader(nullptr));
262 RET_IF_ERR_MSG(!hasConfiguredType && APID != static_cast<int>(IDLE_APID),
263 ErrorCode::CONFIG_FILE_ERROR,
264 "Config: direction-neutral packets require ccsds_type.");
265 }
266
267 if (cfg.isKey("application_data")) {
268 std::vector<std::uint8_t> applicationData{};
269 ASSIGN_CP(applicationData, cfg.get<std::vector<std::uint8_t>>("application_data"));
270 FORWARD_RESULT(m_dataField.setApplicationData(applicationData));
271 }
272
273 if (m_primaryHeader.getAPID() == IDLE_APID) {
274 const auto idleResult = validatePacketForSerialization(m_primaryHeader, m_dataField);
275 if (!idleResult) {
276 return Error{ErrorCode::CONFIG_FILE_ERROR,
277 "Config: invalid Idle Packet: " + idleResult.error().message()};
278 }
279 }
280 m_updateStatus = false;
281 return true;
282}

374ccsds::ResultBuffer ccsds::Packet::serialize() {
375 const auto updateResult = update();
376 if (!updateResult) return updateResult.error();
377 const auto header = static_cast<const Header &>(m_primaryHeader).serialize();
378 RET_IF_ERR_MSG(header.size() != 6U, ErrorCode::INVALID_HEADER_DATA,
379 "Cannot serialize packet: primary-header serialization failed.");
380 std::vector<std::uint8_t> dataField;
381 ASSIGN_MV(dataField, m_dataField.serialize());
382
383 const auto packetDataFieldSize = dataField.size() + getPacketErrorControlSize();
384 constexpr auto maximumPacketDataFieldSize =
385 static_cast<std::size_t>(std::numeric_limits<std::uint16_t>::max()) + 1U;
386 RET_IF_ERR_MSG(packetDataFieldSize == 0U, ErrorCode::INVALID_DATA,
387 "Cannot serialize packet: Packet Data Field is empty.");
388 RET_IF_ERR_MSG(packetDataFieldSize > maximumPacketDataFieldSize, ErrorCode::INVALID_DATA,
389 "Cannot serialize packet: Packet Data Field exceeds the CCSDS length field.");
390 RET_IF_ERR_MSG(static_cast<std::size_t>(m_primaryHeader.getDataLength()) + 1U
391 != packetDataFieldSize,
392 ErrorCode::INVALID_DATA,
393 "Cannot serialize packet: Packet Data Length does not match the finalized data field.");
394
395 std::vector<std::uint8_t> packet;
396 packet.reserve(header.size() + dataField.size() + getPacketErrorControlSize());
397 packet.insert(packet.end(), header.begin(), header.end());
398 packet.insert(packet.end(), dataField.begin(), dataField.end());
399 if (getPacketErrorControlMode() == PacketErrorControlMode::CRC16) {
400 packet.push_back(static_cast<std::uint8_t>((m_CRC16 >> 8U) & 0xFFU));
401 packet.push_back(static_cast<std::uint8_t>(m_CRC16 & 0xFFU));
402 }
403 return packet;
404}

406ccsds::Result<std::size_t> ccsds::Packet::deserializeBoundedWithSecondaryHeader(
407 const std::vector<std::uint8_t> &data,
408 const std::shared_ptr<SecondaryHeaderAbstract> &prototype,
409 const std::int32_t customHeaderSize) {
410 RET_IF_ERR_MSG(!prototype, ErrorCode::INVALID_SECONDARY_HEADER_DATA,
411 "Cannot deserialize packet: secondary-header prototype is null.");
412
413 std::size_t packetSize{};
414 ASSIGN_CP(packetSize, declaredPacketSize(data));
415 const std::vector<std::uint8_t> headerData(data.begin(), data.begin() + 6);
416 const std::vector<std::uint8_t> packetData(
417 data.begin() + 6, data.begin() + static_cast<std::ptrdiff_t>(packetSize));
418 const auto validated = validatePacketBytes(headerData, packetData,
419 getPacketErrorControlMode(), m_CRC16Config);
420 if (!validated) return validated.error();
421 auto parsed = validated.value();
422
423 RET_IF_ERR_MSG(parsed.header.getSecondaryHeaderFlag() == 0U,
424 ErrorCode::INVALID_SECONDARY_HEADER_DATA,
425 "Cannot deserialize typed secondary header: CCSDS secondary-header flag is zero.");
426 if (prototype->getDirection() != PacketDirection::Unspecified) {
427 RET_IF_ERR_MSG(parsed.header.getType()
428 != packetTypeForDirection(prototype->getDirection()),
429 ErrorCode::INVALID_SECONDARY_HEADER_DATA,
430 "Secondary-header direction does not match the CCSDS Packet Type.");
431 }
432
433 DataField parsedField = m_dataField;
434 parsedField.clearContent();
435 std::shared_ptr<SecondaryHeaderAbstract> secondaryHeader;
436 if (prototype->isPusHeader()) {
437 const auto &pusPrototype = static_cast<const pus::SecondaryHeader &>(*prototype);
438 auto cloneResult = parsedField.getPusSecondaryHeaderFactory().clone(pusPrototype);
439 if (!cloneResult) return cloneResult.error();
440 secondaryHeader = cloneResult.value();
441 } else {
442 RET_IF_ERR_MSG(!parsedField.getSecondaryHeaderFactory().typeIsRegistered(prototype->getType()),
443 ErrorCode::INVALID_SECONDARY_HEADER_DATA,
444 "Cannot deserialize packet: custom secondary-header type is not registered: "
445 + prototype->getType());
446 secondaryHeader = parsedField.getSecondaryHeaderFactory().create(prototype->getType());
447 }
448 RET_IF_ERR_MSG(!secondaryHeader, ErrorCode::INVALID_SECONDARY_HEADER_DATA,
449 "Cannot deserialize packet: failed to create secondary-header parser.");
450
451 std::size_t secondaryHeaderSize = secondaryHeader->getSize();
452 if (!secondaryHeader->isPusHeader() && secondaryHeader->variableLength
453 && customHeaderSize > 0) {
454 secondaryHeaderSize = static_cast<std::size_t>(customHeaderSize);
455 }
456 RET_IF_ERR_MSG(secondaryHeaderSize > parsed.dataField.size(),
457 ErrorCode::INVALID_SECONDARY_HEADER_DATA,
458 "Cannot deserialize packet: secondary header exceeds the packet data field.");
459
460 const std::vector<std::uint8_t> secondaryBytes(
461 parsed.dataField.begin(),
462 parsed.dataField.begin() + static_cast<std::ptrdiff_t>(secondaryHeaderSize));
463 FORWARD_RESULT(secondaryHeader->deserialize(secondaryBytes));
464 FORWARD_RESULT(parsedField.setSecondaryHeader(secondaryHeader));
465
466 const std::vector<std::uint8_t> applicationData(
467 parsed.dataField.begin() + static_cast<std::ptrdiff_t>(secondaryHeaderSize),
468 parsed.dataField.end());
469 FORWARD_RESULT(parsedField.setApplicationData(applicationData));
470
471 m_primaryHeader = parsed.header;
472 m_dataField = std::move(parsedField);
473 m_CRC16 = parsed.receivedCRC;
474 m_sequenceCounter = (m_sequenceCounter & PACKET_ERROR_CONTROL_DISABLED_MASK)
475 | (m_primaryHeader.getSequenceCount() & SEQUENCE_COUNT_MASK);
476 m_updateStatus = true;
477 return packetSize;
478}

480ccsds::Result<std::size_t> ccsds::Packet::deserializeBounded(
481 const std::vector<std::uint8_t> &data) {
482 if (const auto prototype = m_dataField.getSecondaryHeader()) {
483 return deserializeBoundedWithSecondaryHeader(data, prototype);
484 }
485
486 std::size_t packetSize{};
487 ASSIGN_CP(packetSize, declaredPacketSize(data));
488 const std::vector<std::uint8_t> headerData(data.begin(), data.begin() + 6);
489 const std::vector<std::uint8_t> packetData(
490 data.begin() + 6, data.begin() + static_cast<std::ptrdiff_t>(packetSize));
491 const auto parseResult = deserialize(headerData, packetData);
492 if (!parseResult) return parseResult.error();
493 return packetSize;
494}

496ccsds::Result<std::size_t> ccsds::Packet::deserializeBounded(
497 const std::vector<std::uint8_t> &data, const std::string &headerType,
498 const std::int32_t headerSize) {
499 RET_IF_ERR_MSG(headerType == "BufferHeader", ErrorCode::INVALID_SECONDARY_HEADER_DATA,
500 "Cannot deserialize packet: BufferHeader requires an explicit byte size.");
501 const bool isPus = pus::SecondaryHeaderFactory::isPusSelector(headerType);
502 RET_IF_ERR_MSG(isPus
503 ? !m_dataField.getPusSecondaryHeaderFactory().typeIsSupported(headerType)
504 : !m_dataField.getSecondaryHeaderFactory().typeIsRegistered(headerType),
505 ErrorCode::INVALID_SECONDARY_HEADER_DATA,
506 "Cannot deserialize packet: unsupported secondary header: " + headerType);
507
508 std::shared_ptr<SecondaryHeaderAbstract> prototype;
509 const auto installed = m_dataField.getSecondaryHeader();
510 if (isPus && installed && installed->isPusHeader() && installed->getType() == headerType) {
511 prototype = installed;
512 } else if (isPus) {
513 auto createResult = m_dataField.getPusSecondaryHeaderFactory().create(headerType);
514 if (!createResult) return createResult.error();
515 prototype = createResult.value();
516 } else {
517 prototype = m_dataField.getSecondaryHeaderFactory().create(headerType);
518 }
519 return deserializeBoundedWithSecondaryHeader(data, prototype, headerSize);
520}

522ccsds::Result<std::size_t> ccsds::Packet::deserializeBounded(
523 const std::vector<std::uint8_t> &data, const std::uint16_t headerDataSizeBytes) {
524 std::size_t packetSize{};
525 ASSIGN_CP(packetSize, declaredPacketSize(data));
526 const std::vector<std::uint8_t> headerData(data.begin(), data.begin() + 6);
527 const std::vector<std::uint8_t> packetData(
528 data.begin() + 6, data.begin() + static_cast<std::ptrdiff_t>(packetSize));
529 const auto validated = validatePacketBytes(headerData, packetData,
530 getPacketErrorControlMode(), m_CRC16Config);
531 if (!validated) return validated.error();
532 auto parsed = validated.value();
533 RET_IF_ERR_MSG(parsed.header.getSecondaryHeaderFlag() == 0U && headerDataSizeBytes != 0U,
534 ErrorCode::INVALID_SECONDARY_HEADER_DATA,
535 "Cannot deserialize opaque secondary header: CCSDS flag is zero.");
536 RET_IF_ERR_MSG(static_cast<std::size_t>(headerDataSizeBytes) > parsed.dataField.size(),
537 ErrorCode::INVALID_SECONDARY_HEADER_DATA,
538 "Cannot deserialize packet: secondary-header size exceeds the packet data field.");
539
540 DataField parsedField = m_dataField;
541 parsedField.clearContent();
542 const std::vector<std::uint8_t> secondaryBytes(
543 parsed.dataField.begin(),
544 parsed.dataField.begin() + static_cast<std::ptrdiff_t>(headerDataSizeBytes));
545 if (!secondaryBytes.empty()) FORWARD_RESULT(parsedField.setSecondaryHeader(secondaryBytes));
546 const std::vector<std::uint8_t> applicationData(
547 parsed.dataField.begin() + static_cast<std::ptrdiff_t>(headerDataSizeBytes),
548 parsed.dataField.end());
549 FORWARD_RESULT(parsedField.setApplicationData(applicationData));
550
551 m_primaryHeader = parsed.header;
552 m_dataField = std::move(parsedField);
553 m_CRC16 = parsed.receivedCRC;
554 m_sequenceCounter = (m_sequenceCounter & PACKET_ERROR_CONTROL_DISABLED_MASK)
555 | (m_primaryHeader.getSequenceCount() & SEQUENCE_COUNT_MASK);
556 m_updateStatus = true;
557 return packetSize;
558}
