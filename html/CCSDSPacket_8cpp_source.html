55 ccsds::Error packetCoreError(const ccsds_status_t status,
56 const char *action) {
57 ccsds::ErrorCode code = ccsds::ErrorCode::UNKNOWN_ERROR;
58 switch (status) {
59 case CCSDS_STATUS_INVALID_HEADER_DATA:
60 code = ccsds::ErrorCode::INVALID_HEADER_DATA;
61 break;
62 case CCSDS_STATUS_INVALID_DATA:
63 case CCSDS_STATUS_BUFFER_TOO_SMALL:
64 code = ccsds::ErrorCode::INVALID_DATA;
65 break;
66 case CCSDS_STATUS_NULL_POINTER:
67 code = ccsds::ErrorCode::NULL_POINTER;
68 break;
69 default:
70 break;
71 }
72 return ccsds::Error{
73 code,
74 std::string(action) + ": " + ccsds_status_name(status)
75 };
76 }

78 ccsds::Error packetViewError(const ccsds_status_t status) {
79 if (status == CCSDS_STATUS_INVALID_CHECKSUM) {
80 return ccsds::Error{
81 ccsds::ErrorCode::INVALID_CHECKSUM,
82 "Cannot deserialize packet: CRC16 packet error-control mismatch."
83 };
84 }
85
86 const auto code = status <= CCSDS_STATUS_CONFIG_FILE_ERROR
87 ? static_cast<ccsds::ErrorCode>(status)
88 : ccsds::ErrorCode::UNKNOWN_ERROR;
89 return ccsds::Error{
90 code,
91 std::string("Cannot deserialize packet: C packet-view validation failed: ")
92 + ccsds_status_name(status)
93 };
94 }

96 ccsds::Result<ParsedPacketView> validatePacketView(
97 const std::uint8_t *data,
98 const std::size_t size,
99 const ccsds::PacketErrorControlMode mode,
100 const ccsds::CRC16Config &crcConfig) {
101 if (size < CCSDS_PRIMARY_HEADER_SIZE) {
102 return ccsds::Error{ccsds::ErrorCode::INVALID_HEADER_DATA,
103 "Cannot deserialize packet: truncated CCSDS primary header."};
104 }
105 if (data == nullptr) {
106 return ccsds::Error{ccsds::ErrorCode::NULL_POINTER,
107 "Cannot deserialize packet: raw buffer pointer is null."};
108 }
109
110 ccsds_primary_header_t coreHeader{};
111 const auto headerStatus =
112 ccsds_primary_header_decode(data, size, &coreHeader);
113 if (headerStatus != CCSDS_STATUS_OK) {
114 return ccsds::Error{ccsds::ErrorCode::INVALID_HEADER_DATA,
115 "Cannot deserialize packet: invalid CCSDS primary header."};
116 }
117 if (coreHeader.version_number != 0U) {
118 return ccsds::Error{ccsds::ErrorCode::INVALID_HEADER_DATA,
119 "Cannot deserialize packet: unsupported CCSDS packet version."};
120 }
121
122 const auto packetSize =
123 CCSDS_PRIMARY_HEADER_SIZE + static_cast<std::size_t>(coreHeader.data_length) + 1U;
124 if (size < packetSize) {
125 return ccsds::Error{ccsds::ErrorCode::INVALID_DATA,
126 "Cannot deserialize packet: truncated packet body for Packet Data Length."};
127 }
128
129 const auto bodySize = packetSize - CCSDS_PRIMARY_HEADER_SIZE;
130 const auto pecSize =
131 mode == ccsds::PacketErrorControlMode::CRC16 ? std::size_t{2U} : std::size_t{0U};
132 if (bodySize < pecSize) {
133 return ccsds::Error{
134 ccsds::ErrorCode::INVALID_DATA,
135 "Cannot deserialize packet: CRC16 mode requires two packet error-control bytes."
136 };
137 }
138 const auto dataFieldSize = bodySize - pecSize;
139 if (coreHeader.apid == CCSDS_IDLE_APID) {
140 if (coreHeader.secondary_header_flag != 0U) {
141 return ccsds::Error{
142 ccsds::ErrorCode::INVALID_HEADER_DATA,
143 "Cannot deserialize Idle Packet: secondary-header flag must be zero."
144 };
145 }
146 if (dataFieldSize == 0U) {
147 return ccsds::Error{
148 ccsds::ErrorCode::INVALID_DATA,
149 "Cannot deserialize Idle Packet: mission-defined idle user data is required."
150 };
151 }
152 }
153
154 const ccsds_crc16_config_t coreCrc{
155 crcConfig.polynomial,
156 crcConfig.initialValue,
157 crcConfig.finalXorValue
158 };
159 ccsds_packet_view_t view{};
160 const auto status = ccsds_packet_view_parse(
161 data,
162 size,
163 mode == ccsds::PacketErrorControlMode::CRC16
164 ? CCSDS_PACKET_ERROR_CONTROL_CRC16
165 : CCSDS_PACKET_ERROR_CONTROL_NONE,
166 &coreCrc,
167 &view);
168 if (status != CCSDS_STATUS_OK) return packetViewError(status);
169
170 ParsedPacketView parsed;
171 const ccsds::PrimaryHeader header{
172 view.primary_header.version_number,
173 view.primary_header.type,
174 view.primary_header.secondary_header_flag,
175 view.primary_header.apid,
176 view.primary_header.sequence_flags,
177 view.primary_header.sequence_count,
178 view.primary_header.data_length
179 };
180 const auto headerResult = parsed.header.setData(header);
181 if (!headerResult) return headerResult.error();
182 parsed.dataField = view.data_field;
183 parsed.receivedCRC = view.received_crc16;
184 parsed.packetSize = view.consumed;
185 return parsed;
186 }

188 ccsds::ResultBool validatePacketForSerialization(
189 const ccsds::Header &header,
190 const ccsds::DataField &dataField) {
191 RET_IF_ERR_MSG(header.getHeaderStatus() == ccsds::INVALID,
192 ccsds::ErrorCode::INVALID_HEADER_DATA,
193 "Cannot serialize packet: primary header is invalid.");
194 RET_IF_ERR_MSG(header.getVersionNumber() != 0U,
195 ccsds::ErrorCode::INVALID_HEADER_DATA,
196 "Cannot serialize packet: unsupported CCSDS packet version.");
197
198 if (header.getAPID() == ccsds::IDLE_APID) {
199 RET_IF_ERR_MSG(header.getSecondaryHeaderFlag() != 0U,
200 ccsds::ErrorCode::INVALID_HEADER_DATA,
201 "Cannot serialize Idle Packet: secondary-header flag must be zero.");
202 RET_IF_ERR_MSG(dataField.getSecondaryHeaderFlag(),
203 ccsds::ErrorCode::INVALID_SECONDARY_HEADER_DATA,
204 "Cannot serialize Idle Packet with a secondary header.");
205 RET_IF_ERR_MSG(dataField.getApplicationDataBytesSize() == 0U,
206 ccsds::ErrorCode::INVALID_DATA,
207 "Cannot serialize Idle Packet without mission-defined idle user data.");
208 return true;
209 }
210
211 const auto secondary = dataField.getSecondaryHeader();
212 RET_IF_ERR_MSG((header.getSecondaryHeaderFlag() != 0U) != static_cast<bool>(secondary),
213 ccsds::ErrorCode::INVALID_SECONDARY_HEADER_DATA,
214 "Cannot serialize packet: primary-header flag and secondary-header state differ.");
215
216 if (secondary && secondary->getDirection() != ccsds::PacketDirection::Unspecified) {
217 RET_IF_ERR_MSG(header.getType()
218 != ccsds::packetTypeForDirection(secondary->getDirection()),
219 ccsds::ErrorCode::INVALID_HEADER_DATA,
220 "Cannot serialize packet: primary-header Packet Type differs from secondary-header direction.");
221 }
222 return true;
223 }

225 ccsds::Result<ParsedPacket> validatePacketBytes(
226 const std::vector<std::uint8_t> &headerData,
227 const std::vector<std::uint8_t> &packetData,
228 const ccsds::PacketErrorControlMode mode,
229 const ccsds::CRC16Config &crcConfig) {
230 if (headerData.size() != 6U) {
231 return ccsds::Error{ccsds::ErrorCode::INVALID_HEADER_DATA,
232 "Cannot deserialize packet: primary header must contain exactly six bytes."};
233 }
234
235 ParsedPacket parsed;
236 const auto headerResult = parsed.header.deserialize(headerData);
237 if (!headerResult) return headerResult.error();
238 if (parsed.header.getVersionNumber() != 0U) {
239 return ccsds::Error{ccsds::ErrorCode::INVALID_HEADER_DATA,
240 "Cannot deserialize packet: unsupported CCSDS packet version."};
241 }
242
243 const auto expectedDataSize = static_cast<std::size_t>(parsed.header.getDataLength()) + 1U;
244 if (packetData.size() != expectedDataSize) {
245 return ccsds::Error{ccsds::ErrorCode::INVALID_DATA,
246 "Cannot deserialize packet: packet body size does not match Packet Data Length."};
247 }
248
249 if (mode == ccsds::PacketErrorControlMode::CRC16) {
250 if (packetData.size() < 2U) {
251 return ccsds::Error{ccsds::ErrorCode::INVALID_DATA,
252 "Cannot deserialize packet: CRC16 mode requires two packet error-control bytes."};
253 }
254 parsed.receivedCRC = static_cast<std::uint16_t>(
255 (static_cast<std::uint16_t>(packetData[packetData.size() - 2U]) << 8U)
256 | packetData.back());
257 std::vector<std::uint8_t> crcInput = headerData;
258 crcInput.insert(crcInput.end(), packetData.begin(), packetData.end() - 2);
259 const auto expectedCRC = ccsds::crc16(crcInput, crcConfig.polynomial,
260 crcConfig.initialValue, crcConfig.finalXorValue);
261 if (expectedCRC != parsed.receivedCRC) {
262 return ccsds::Error{ccsds::ErrorCode::INVALID_CHECKSUM,
263 "Cannot deserialize packet: CRC16 packet error-control mismatch."};
264 }
265 parsed.dataField.assign(packetData.begin(), packetData.end() - 2);
266 } else {
267 parsed.receivedCRC = 0U;
268 parsed.dataField = packetData;
269 }
270
271 if (parsed.header.getAPID() == ccsds::IDLE_APID) {
272 if (parsed.header.getSecondaryHeaderFlag() != 0U) {
273 return ccsds::Error{ccsds::ErrorCode::INVALID_HEADER_DATA,
274 "Cannot deserialize Idle Packet: secondary-header flag must be zero."};
275 }
276 if (parsed.dataField.empty()) {
277 return ccsds::Error{ccsds::ErrorCode::INVALID_DATA,
278 "Cannot deserialize Idle Packet: mission-defined idle user data is required."};
279 }
280 }
281 return parsed;
282 }

299ccsds::ResultBool ccsds::Packet::finalizeDataFieldParts(
300 const std::vector<std::uint8_t> &secondaryHeader,
301 const ccsds_buffer_view_t applicationData) {
302 const auto dataFieldSize = secondaryHeader.size() + applicationData.size;
303 const auto packetDataFieldSize = dataFieldSize + getPacketErrorControlSize();
304 constexpr auto maximumPacketDataFieldSize =
305 static_cast<std::size_t>(std::numeric_limits<std::uint16_t>::max()) + 1U;
306
307 RET_IF_ERR_MSG(packetDataFieldSize == 0U, ErrorCode::INVALID_DATA,
308 "Cannot finalize packet: Packet Data Field is empty.");
309 RET_IF_ERR_MSG(packetDataFieldSize > maximumPacketDataFieldSize, ErrorCode::INVALID_DATA,
310 "Cannot finalize packet: Packet Data Field exceeds the CCSDS length field.");
311
312 auto coreHeader = toCoreHeader(m_primaryHeader);
313 const ccsds_packet_data_parts_t coreDataField{
314 {
315 secondaryHeader.empty() ? nullptr : secondaryHeader.data(),
316 secondaryHeader.size()
317 },
318 applicationData
319 };
320 const auto coreCrc = toCoreCrcConfig(m_CRC16Config);
321 std::uint16_t crc = 0U;
322 std::size_t serializedSize = 0U;
323 const auto status = ccsds_packet_finalize_parts(
324 &coreHeader,
325 static_cast<std::uint16_t>(m_sequenceCounter & SEQUENCE_COUNT_MASK),
326 coreDataField,
327 toCoreErrorControl(getPacketErrorControlMode()),
328 &coreCrc,
329 &crc,
330 &serializedSize);
331 if (status != CCSDS_STATUS_OK) {
332 return packetCoreError(status, "Cannot finalize packet");
333 }
334
335 RET_IF_ERR_MSG(serializedSize
336 != CCSDS_PRIMARY_HEADER_SIZE + packetDataFieldSize,
337 ErrorCode::INVALID_DATA,
338 "Cannot finalize packet: C core returned an unexpected packet size.");
339
340 m_primaryHeader.setDataLength(coreHeader.data_length);
341 FORWARD_RESULT(m_primaryHeader.setSequenceCount(coreHeader.sequence_count));
342 m_CRC16 = crc;
343 return true;
344}

346ccsds::ResultBool ccsds::Packet::update() {
347 FORWARD_RESULT(validatePacketForSerialization(m_primaryHeader, m_dataField));
348 if (m_updateStatus || !m_enableUpdatePacket) return true;
349
350 m_dataField.update();
351 const auto secondaryHeader = m_dataField.getSecondaryHeaderBytes();
352 const auto secondary = m_dataField.getSecondaryHeader();
353 RET_IF_ERR_MSG(secondary && secondaryHeader.size() != secondary->getSize(),
354 ErrorCode::INVALID_SECONDARY_HEADER_DATA,
355 "Secondary header serialization failed or returned an unexpected size.");
356 RET_IF_ERR_MSG(secondaryHeader.size()
357 + m_dataField.getApplicationDataView().size
358 > m_dataField.getDataFieldAbsoluteBytesSize(),
359 ErrorCode::INVALID_DATA,
360 "Serialized packet data field exceeds its configured capacity.");
361
362 FORWARD_RESULT(finalizeDataFieldParts(
363 secondaryHeader, m_dataField.getApplicationDataView()));
364 m_updateStatus = true;
365 return true;
366}

375ccsds::ResultBool ccsds::Packet::loadFromConfig(const ccsds::Config &cfg) {
376 int versionNumber{};
377 int APID{};
378 bool segmented{};
379 bool configuredType{false};
380 const bool hasConfiguredType = cfg.isKey("ccsds_type");
381
382 RET_IF_ERR_MSG(!cfg.isKey("ccsds_version_number"), ErrorCode::CONFIG_FILE_ERROR,
383 "Config: Missing int field: ccsds_version_number");
384 RET_IF_ERR_MSG(!cfg.isKey("ccsds_APID"), ErrorCode::CONFIG_FILE_ERROR,
385 "Config: Missing int field: ccsds_APID");
386 RET_IF_ERR_MSG(!cfg.isKey("ccsds_segmented"), ErrorCode::CONFIG_FILE_ERROR,
387 "Config: Missing bool field: ccsds_segmented");
388
389 ASSIGN_CP(versionNumber, cfg.get<int>("ccsds_version_number"));
390 ASSIGN_CP(APID, cfg.get<int>("ccsds_APID"));
391 ASSIGN_CP(segmented, cfg.get<bool>("ccsds_segmented"));
392 if (hasConfiguredType) ASSIGN_CP(configuredType, cfg.get<bool>("ccsds_type"));
393
394 RET_IF_ERR_MSG(versionNumber != 0, ErrorCode::CONFIG_FILE_ERROR,
395 "Config: ccsds_version_number must be 0 for CCSDS 133.0-B-2 Space Packets");
396 RET_IF_ERR_MSG(APID < 0 || APID > static_cast<int>(IDLE_APID), ErrorCode::CONFIG_FILE_ERROR,
397 "Config: ccsds_APID must be between 0 and 2047");
398
399 FORWARD_RESULT(m_primaryHeader.setVersionNumber(static_cast<std::uint8_t>(versionNumber)));
400 if (hasConfiguredType) FORWARD_RESULT(m_primaryHeader.setType(static_cast<std::uint8_t>(configuredType)));
401 FORWARD_RESULT(m_primaryHeader.setAPID(static_cast<std::uint16_t>(APID)));
402
403 PacketErrorControlMode errorControl{};
404 ASSIGN_CP(errorControl, packetErrorControlFromConfig(cfg));
405 setPacketErrorControlMode(errorControl);
406
407 const auto sequenceFlag = segmented ? FIRST_SEGMENT : UNSEGMENTED;
408 const auto sequenceCount = static_cast<std::uint16_t>(segmented ? 1U : 0U);
409 FORWARD_RESULT(m_primaryHeader.setSequenceFlags(sequenceFlag));
410 FORWARD_RESULT(m_primaryHeader.setSequenceCount(sequenceCount));
411 m_sequenceCounter = (m_sequenceCounter & PACKET_ERROR_CONTROL_DISABLED_MASK)
412 | (sequenceCount & SEQUENCE_COUNT_MASK);
413
414 if (cfg.isKey("data_field_size")) {
415 int dataFieldSize{};
416 ASSIGN_CP(dataFieldSize, cfg.get<int>("data_field_size"));
417 RET_IF_ERR_MSG(dataFieldSize < 0 || dataFieldSize > std::numeric_limits<std::uint16_t>::max(),
418 ErrorCode::CONFIG_FILE_ERROR,
419 "Config: data_field_size must be between 0 and 65535");
420 m_dataField.setDataPacketSize(static_cast<std::uint16_t>(dataFieldSize));
421 }
422
423 bool defineSecondaryHeader{false};
424 if (cfg.isKey("define_secondary_header")) {
425 ASSIGN_CP(defineSecondaryHeader, cfg.get<bool>("define_secondary_header"));
426 }
427
428 if (defineSecondaryHeader) {
429 FORWARD_RESULT(m_dataField.setSecondaryHeader(cfg));
430 const auto secondary = m_dataField.getSecondaryHeader();
431 RET_IF_ERR_MSG(!secondary, ErrorCode::CONFIG_FILE_ERROR,
432 "Config: define_secondary_header did not create a header.");
433 const auto declaredDirection = secondary->getDirection();
434 if (hasConfiguredType && declaredDirection != PacketDirection::Unspecified) {
435 RET_IF_ERR_MSG(static_cast<std::uint8_t>(configuredType)
436 != packetTypeForDirection(declaredDirection),
437 ErrorCode::CONFIG_FILE_ERROR,
438 "Config: ccsds_type contradicts the concrete secondary-header direction.");
439 }
440 FORWARD_RESULT(setSecondaryHeader(secondary));
441 } else {
442 FORWARD_RESULT(setSecondaryHeader(nullptr));
443 RET_IF_ERR_MSG(!hasConfiguredType && APID != static_cast<int>(IDLE_APID),
444 ErrorCode::CONFIG_FILE_ERROR,
445 "Config: direction-neutral packets require ccsds_type.");
446 }
447
448 if (cfg.isKey("application_data")) {
449 std::vector<std::uint8_t> applicationData{};
450 ASSIGN_CP(applicationData, cfg.get<std::vector<std::uint8_t>>("application_data"));
451 FORWARD_RESULT(m_dataField.setApplicationData(applicationData));
452 }
453
454 if (m_primaryHeader.getAPID() == IDLE_APID) {
455 const auto idleResult = validatePacketForSerialization(m_primaryHeader, m_dataField);
456 if (!idleResult) {
457 return Error{ErrorCode::CONFIG_FILE_ERROR,
458 "Config: invalid Idle Packet: " + idleResult.error().message()};
459 }
460 }
461 m_updateStatus = false;
462 return true;
463}

555ccsds::Result<std::size_t> ccsds::Packet::serialize(
556 std::uint8_t *output, const std::size_t capacity) {
557 FORWARD_RESULT(validatePacketForSerialization(m_primaryHeader, m_dataField));
558
559 const auto expectedCapacity = getSerializedSize();
560 RET_IF_ERR_MSG(output == nullptr, ErrorCode::NULL_POINTER,
561 "Cannot serialize packet: output buffer pointer is null.");
562 RET_IF_ERR_MSG(capacity < expectedCapacity, ErrorCode::INVALID_DATA,
563 "Cannot serialize packet: output buffer is too small.");
564
565 m_dataField.update();
566 const auto secondaryHeader = m_dataField.getSecondaryHeaderBytes();
567 const auto secondary = m_dataField.getSecondaryHeader();
568 RET_IF_ERR_MSG(secondary && secondaryHeader.size() != secondary->getSize(),
569 ErrorCode::INVALID_SECONDARY_HEADER_DATA,
570 "Secondary header serialization failed or returned an unexpected size.");
571 const auto applicationData = m_dataField.getApplicationDataView();
572 RET_IF_ERR_MSG(secondaryHeader.size() + applicationData.size
573 > m_dataField.getDataFieldAbsoluteBytesSize(),
574 ErrorCode::INVALID_DATA,
575 "Serialized packet data field exceeds its configured capacity.");
576
577 if (!m_updateStatus && m_enableUpdatePacket) {
578 FORWARD_RESULT(finalizeDataFieldParts(secondaryHeader, applicationData));
579 m_updateStatus = true;
580 }
581
582 const auto dataFieldSize = secondaryHeader.size() + applicationData.size;
583 const auto packetDataFieldSize = dataFieldSize + getPacketErrorControlSize();
584 constexpr auto maximumPacketDataFieldSize =
585 static_cast<std::size_t>(std::numeric_limits<std::uint16_t>::max()) + 1U;
586 RET_IF_ERR_MSG(packetDataFieldSize == 0U, ErrorCode::INVALID_DATA,
587 "Cannot serialize packet: Packet Data Field is empty.");
588 RET_IF_ERR_MSG(packetDataFieldSize > maximumPacketDataFieldSize, ErrorCode::INVALID_DATA,
589 "Cannot serialize packet: Packet Data Field exceeds the CCSDS length field.");
590 RET_IF_ERR_MSG(static_cast<std::size_t>(m_primaryHeader.getDataLength()) + 1U
591 != packetDataFieldSize,
592 ErrorCode::INVALID_DATA,
593 "Cannot serialize packet: Packet Data Length does not match the finalized data field.");
594
595 const auto required = CCSDS_PRIMARY_HEADER_SIZE + packetDataFieldSize;
596 RET_IF_ERR_MSG(required != expectedCapacity, ErrorCode::INVALID_DATA,
597 "Cannot serialize packet: finalized size differs from the expected size.");
598
599 const auto coreHeader = toCoreHeader(m_primaryHeader);
600 const ccsds_packet_data_parts_t coreDataField{
601 {
602 secondaryHeader.empty() ? nullptr : secondaryHeader.data(),
603 secondaryHeader.size()
604 },
605 applicationData
606 };
607 std::size_t written = 0U;
608 const auto status = ccsds_packet_encode_parts(
609 &coreHeader,
610 coreDataField,
611 toCoreErrorControl(getPacketErrorControlMode()),
612 m_CRC16,
613 output,
614 capacity,
615 &written);
616 if (status != CCSDS_STATUS_OK) {
617 return packetCoreError(status, "Cannot serialize packet");
618 }
619 RET_IF_ERR_MSG(written != required, ErrorCode::INVALID_DATA,
620 "Cannot serialize packet: C core returned an unexpected packet size.");
621 return written;
622}

632ccsds::Result<std::size_t> ccsds::Packet::deserializeBoundedWithSecondaryHeader(
633 const std::uint8_t *data,
634 const std::size_t size,
635 const std::shared_ptr<SecondaryHeaderAbstract> &prototype,
636 const std::int32_t customHeaderSize) {
637 RET_IF_ERR_MSG(!prototype, ErrorCode::INVALID_SECONDARY_HEADER_DATA,
638 "Cannot deserialize packet: secondary-header prototype is null.");
639
640 const auto validated =
641 validatePacketView(data, size, getPacketErrorControlMode(), m_CRC16Config);
642 if (!validated) return validated.error();
643 const auto parsed = validated.value();
644
645 RET_IF_ERR_MSG(parsed.header.getSecondaryHeaderFlag() == 0U,
646 ErrorCode::INVALID_SECONDARY_HEADER_DATA,
647 "Cannot deserialize typed secondary header: CCSDS secondary-header flag is zero.");
648 if (prototype->getDirection() != PacketDirection::Unspecified) {
649 RET_IF_ERR_MSG(parsed.header.getType()
650 != packetTypeForDirection(prototype->getDirection()),
651 ErrorCode::INVALID_SECONDARY_HEADER_DATA,
652 "Secondary-header direction does not match the CCSDS Packet Type.");
653 }
654
655 DataField parsedField = m_dataField;
656 parsedField.clearContent();
657 std::shared_ptr<SecondaryHeaderAbstract> secondaryHeader;
658 if (prototype->isPusHeader()) {
659 const auto &pusPrototype = static_cast<const pus::SecondaryHeader &>(*prototype);
660 auto cloneResult = parsedField.getPusSecondaryHeaderFactory().clone(pusPrototype);
661 if (!cloneResult) return cloneResult.error();
662 secondaryHeader = cloneResult.value();
663 } else {
664 RET_IF_ERR_MSG(!parsedField.getSecondaryHeaderFactory().typeIsRegistered(prototype->getType()),
665 ErrorCode::INVALID_SECONDARY_HEADER_DATA,
666 "Cannot deserialize packet: custom secondary-header type is not registered: "
667 + prototype->getType());
668 secondaryHeader = parsedField.getSecondaryHeaderFactory().create(prototype->getType());
669 }
670 RET_IF_ERR_MSG(!secondaryHeader, ErrorCode::INVALID_SECONDARY_HEADER_DATA,
671 "Cannot deserialize packet: failed to create secondary-header parser.");
672
673 std::size_t secondaryHeaderSize = secondaryHeader->getSize();
674 if (!secondaryHeader->isPusHeader() && secondaryHeader->variableLength
675 && customHeaderSize > 0) {
676 secondaryHeaderSize = static_cast<std::size_t>(customHeaderSize);
677 }
678 RET_IF_ERR_MSG(secondaryHeaderSize > parsed.dataField.size,
679 ErrorCode::INVALID_SECONDARY_HEADER_DATA,
680 "Cannot deserialize packet: secondary header exceeds the packet data field.");
681
682 std::vector<std::uint8_t> secondaryBytes;
683 if (secondaryHeaderSize != 0U) {
684 secondaryBytes.assign(parsed.dataField.data,
685 parsed.dataField.data + secondaryHeaderSize);
686 }
687 FORWARD_RESULT(secondaryHeader->deserialize(secondaryBytes));
688 FORWARD_RESULT(parsedField.setSecondaryHeader(secondaryHeader));
689
690 const auto applicationSize = parsed.dataField.size - secondaryHeaderSize;
691 if (applicationSize == 0U) {
692 FORWARD_RESULT(parsedField.setApplicationData(std::vector<std::uint8_t>{}));
693 } else {
694 FORWARD_RESULT(parsedField.setApplicationData(
695 parsed.dataField.data + secondaryHeaderSize, applicationSize));
696 }
697
698 m_primaryHeader = parsed.header;
699 m_dataField = std::move(parsedField);
700 m_CRC16 = parsed.receivedCRC;
701 m_sequenceCounter = (m_sequenceCounter & PACKET_ERROR_CONTROL_DISABLED_MASK)
702 | (m_primaryHeader.getSequenceCount() & SEQUENCE_COUNT_MASK);
703 m_updateStatus = true;
704 return parsed.packetSize;
705}
