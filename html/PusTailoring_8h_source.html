23 [[nodiscard]] inline std::string selector(const Revision revision,
24 const PacketDirection direction) {
25 const char *rev = revision == Revision::A ? "revA"
26 : revision == Revision::C ? "revC" : "unspecified";
27 const char *dir = direction == PacketDirection::Telecommand ? "TC"
28 : direction == PacketDirection::Telemetry ? "TM" : "unspecified";
29 return std::string{"PUS:"} + rev + ":" + dir;
30 }

67 [[nodiscard]] inline bool validIdentifierWidth(const std::uint8_t width) noexcept {
68 return width == 0U || width == 1U || width == 2U || width == 4U;
69 }

71 [[nodiscard]] inline ResultBool validateTailoring(const rev_a::TcTailoring &tailoring) {
72 RET_IF_ERR_MSG(!validIdentifierWidth(tailoring.sourceIdOctets),
73 ErrorCode::INVALID_SECONDARY_HEADER_DATA,
74 "PUS-A TC source-ID width must be 0, 1, 2, or 4 octets.");
75 return true;
76 }
77
78 [[nodiscard]] inline ResultBool validateTailoring(const rev_a::TmTailoring &tailoring) {
79 RET_IF_ERR_MSG(!validIdentifierWidth(tailoring.destinationIdOctets),
80 ErrorCode::INVALID_SECONDARY_HEADER_DATA,
81 "PUS-A TM destination-ID width must be 0, 1, 2, or 4 octets.");
82 if (tailoring.timestampPresent) {
83 FORWARD_RESULT(time::validate(tailoring.cuc));
84 } else {
85 RET_IF_ERR_MSG(tailoring.cuc.epoch != time::Epoch::Unspecified
86 || tailoring.cuc.pField != time::PFieldMode::Implicit
87 || tailoring.cuc.coarseOctets != 0U
88 || tailoring.cuc.fineOctets != 0U,
89 ErrorCode::INVALID_SECONDARY_HEADER_DATA,
90 "Disabled PUS-A TM time requires an empty CUC configuration.");
91 }
92 return true;
93 }
