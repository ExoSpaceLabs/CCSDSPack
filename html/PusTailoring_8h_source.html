18 enum class Revision : std::uint8_t {
19 Unspecified = 0,
20 A = 1,
21 C = 2

25 [[nodiscard]] inline std::string selector(const Revision revision,
26 const PacketDirection direction) {
27 const char *rev = revision == Revision::A ? "revA"
28 : revision == Revision::C ? "revC" : "unspecified";
29 const char *dir = direction == PacketDirection::Telecommand ? "TC"
30 : direction == PacketDirection::Telemetry ? "TM" : "unspecified";
31 return std::string{"PUS:"} + rev + ":" + dir;
32 }

73 [[nodiscard]] inline ResultBool validateTailoring(const rev_a::TcTailoring &tailoring) {

75 tailoring.sourceIdOctets,

84 [[nodiscard]] inline ResultBool validateTailoring(const rev_a::TmTailoring &tailoring) {
85 const ccsds_pus_a_tm_tailoring_t core{
86 tailoring.destinationIdOctets,
87 static_cast<std::uint8_t>(tailoring.packetSubcounterPresent ? 1U : 0U),
88 static_cast<std::uint8_t>(tailoring.timestampPresent ? 1U : 0U),
89 {
90 static_cast<ccsds_cuc_epoch_t>(tailoring.cuc.epoch),
91 static_cast<ccsds_cuc_pfield_mode_t>(tailoring.cuc.pField),
92 tailoring.cuc.coarseOctets,
93 tailoring.cuc.fineOctets
94 },
95 tailoring.secondaryHeaderSpareOctets
96 };
97 const auto status = ccsds_pus_a_tm_validate_tailoring(&core);
98 RET_IF_ERR_MSG(status == CCSDS_STATUS_PUS_INVALID_IDENTIFIER_WIDTH,
99 ErrorCode::INVALID_SECONDARY_HEADER_DATA,
100 "PUS-A TM destination-ID width must be 0, 1, 2, or 4 octets.");
101 RET_IF_ERR_MSG(status == CCSDS_STATUS_PUS_DISABLED_TIME_CONFIG,
102 ErrorCode::INVALID_SECONDARY_HEADER_DATA,
103 "Disabled PUS-A TM time requires an empty CUC configuration.");
104 if (status != CCSDS_STATUS_OK) {
105 FORWARD_RESULT(time::validate(tailoring.cuc));
106 return Error{ErrorCode::INVALID_SECONDARY_HEADER_DATA,
107 "Invalid PUS-A TM tailoring."};
108 }
109 return true;
110 }
