134 template <typename HeaderT, typename... Args>
135 [[nodiscard]] Result<std::size_t> deserializeBounded(
136 const std::vector<std::uint8_t> &data, Args&&... args) {
137 static_assert(std::is_base_of<SecondaryHeaderAbstract, HeaderT>::value,
138 "HeaderT must derive from ccsds::SecondaryHeaderAbstract");
139 Packet staged = *this;
140 auto header = std::make_shared<HeaderT>(std::forward<Args>(args)...);
141 const auto attach = staged.setSecondaryHeader(header);
142 if (!attach) return attach.error();
143 const auto parsed = staged.deserializeBounded(data);
144 if (!parsed) return parsed.error();
145 *this = std::move(staged);
146 return parsed.value();
147 }
