9static ccsds_status_t packet_error_control_size(
10 const ccsds_packet_error_control_t error_control,
11 size_t *size_out) {
12 if (size_out == NULL) return CCSDS_STATUS_NULL_POINTER;
13 if (error_control == CCSDS_PACKET_ERROR_CONTROL_NONE) {
14 *size_out = 0U;
15 return CCSDS_STATUS_OK;
16 }
17 if (error_control == CCSDS_PACKET_ERROR_CONTROL_CRC16) {
18 *size_out = 2U;
19 return CCSDS_STATUS_OK;
20 }
21 return CCSDS_STATUS_INVALID_DATA;
22}

24static ccsds_status_t validate_packet_data_field(
25 const ccsds_primary_header_t *header,
26 const ccsds_packet_data_parts_t data_field,
27 const size_t pec_size,
28 size_t *packet_data_field_size_out) {
29 size_t data_field_size;
30 size_t packet_data_field_size;
31
32 if (header == NULL || packet_data_field_size_out == NULL) {
33 return CCSDS_STATUS_NULL_POINTER;
34 }
35 if ((data_field.first.data == NULL && data_field.first.size != 0U)
36 || (data_field.second.data == NULL && data_field.second.size != 0U)) {
37 return CCSDS_STATUS_NULL_POINTER;
38 }
39 if (data_field.first.size > (size_t)UINT16_MAX + (size_t)1U
40 || data_field.second.size > (size_t)UINT16_MAX + (size_t)1U
41 || data_field.second.size
42 > (size_t)UINT16_MAX + (size_t)1U - data_field.first.size) {
43 return CCSDS_STATUS_INVALID_DATA;
44 }
45 data_field_size = data_field.first.size + data_field.second.size;
46 if (data_field_size > (size_t)UINT16_MAX + (size_t)1U - pec_size) {
47 return CCSDS_STATUS_INVALID_DATA;
48 }
49
50 packet_data_field_size = data_field_size + pec_size;
51 if (packet_data_field_size == 0U
52 || packet_data_field_size > (size_t)UINT16_MAX + (size_t)1U) {
53 return CCSDS_STATUS_INVALID_DATA;
54 }
55
56 if (header->version_number != 0U) {
57 return CCSDS_STATUS_INVALID_HEADER_DATA;
58 }
59 if (header->apid == CCSDS_IDLE_APID) {
60 if (header->secondary_header_flag != 0U) {
61 return CCSDS_STATUS_INVALID_HEADER_DATA;
62 }
63 if (data_field_size == 0U) {
64 return CCSDS_STATUS_INVALID_DATA;
65 }
66 }
67
68 *packet_data_field_size_out = packet_data_field_size;
69 return CCSDS_STATUS_OK;
70}

72ccsds_status_t ccsds_packet_finalize_parts(
73 ccsds_primary_header_t *header,
74 const uint16_t sequence_count,
75 const ccsds_packet_data_parts_t data_field,
76 const ccsds_packet_error_control_t error_control,
77 const ccsds_crc16_config_t *crc_config,
78 uint16_t *crc_out,
79 size_t *serialized_size_out) {
80 ccsds_primary_header_t staged;
81 ccsds_crc16_config_t config;
82 uint8_t header_bytes[CCSDS_PRIMARY_HEADER_SIZE];
83 size_t pec_size;
84 size_t packet_data_field_size;
85 uint16_t crc = 0U;
86 uint16_t crc_state;
87 ccsds_status_t status;
88
89 if (header == NULL || crc_out == NULL || serialized_size_out == NULL) {
90 return CCSDS_STATUS_NULL_POINTER;
91 }
92 if (sequence_count > CCSDS_SEQUENCE_COUNT_MAX) {
93 return CCSDS_STATUS_INVALID_HEADER_DATA;
94 }
95
96 status = packet_error_control_size(error_control, &pec_size);
97 if (status != CCSDS_STATUS_OK) return status;
98
99 staged = *header;
100 staged.sequence_count = sequence_count;
101
102 status = validate_packet_data_field(
103 &staged, data_field, pec_size, &packet_data_field_size);
104 if (status != CCSDS_STATUS_OK) return status;
105
106 staged.data_length = (uint16_t)(packet_data_field_size - 1U);
107 status = ccsds_primary_header_validate(&staged);
108 if (status != CCSDS_STATUS_OK) return status;
109
110 if (error_control == CCSDS_PACKET_ERROR_CONTROL_CRC16) {
111 config = crc_config != NULL ? *crc_config : ccsds_crc16_default_config();
112
113 status = ccsds_primary_header_encode(
114 &staged, header_bytes, sizeof(header_bytes));
115 if (status != CCSDS_STATUS_OK) return status;
116
117 crc_state = config.initial_value;
118 status = ccsds_crc16_update(
119 &crc_state, header_bytes, sizeof(header_bytes), config.polynomial);
120 if (status != CCSDS_STATUS_OK) return status;
121 status = ccsds_crc16_update(
122 &crc_state, data_field.first.data, data_field.first.size, config.polynomial);
123 if (status != CCSDS_STATUS_OK) return status;
124 status = ccsds_crc16_update(
125 &crc_state, data_field.second.data, data_field.second.size, config.polynomial);
126 if (status != CCSDS_STATUS_OK) return status;
127 crc = (uint16_t)(crc_state ^ config.final_xor_value);
128 }
129
130 *header = staged;
131 *crc_out = crc;
132 *serialized_size_out = CCSDS_PRIMARY_HEADER_SIZE + packet_data_field_size;
133 return CCSDS_STATUS_OK;
134}

136ccsds_status_t ccsds_packet_encode_parts(
137 const ccsds_primary_header_t *header,
138 const ccsds_packet_data_parts_t data_field,
139 const ccsds_packet_error_control_t error_control,
140 const uint16_t crc16,
141 uint8_t *output,
142 const size_t capacity,
143 size_t *written_out) {
144 size_t pec_size;
145 size_t packet_data_field_size;
146 size_t required;
147 ccsds_status_t status;
148
149 if (header == NULL || written_out == NULL) {
150 return CCSDS_STATUS_NULL_POINTER;
151 }
152
153 status = packet_error_control_size(error_control, &pec_size);
154 if (status != CCSDS_STATUS_OK) return status;
155 status = validate_packet_data_field(
156 header, data_field, pec_size, &packet_data_field_size);
157 if (status != CCSDS_STATUS_OK) return status;
158 status = ccsds_primary_header_validate(header);
159 if (status != CCSDS_STATUS_OK) return status;
160
161 if ((size_t)header->data_length + 1U != packet_data_field_size) {
162 return CCSDS_STATUS_INVALID_DATA;
163 }
164
165 required = CCSDS_PRIMARY_HEADER_SIZE + packet_data_field_size;
166 if (capacity < required) return CCSDS_STATUS_BUFFER_TOO_SMALL;
167 if (output == NULL) return CCSDS_STATUS_NULL_POINTER;
168
169 if (data_field.first.size != 0U) {
170 memmove(output + CCSDS_PRIMARY_HEADER_SIZE,
171 data_field.first.data,
172 data_field.first.size);
173 }
174 if (data_field.second.size != 0U) {
175 memmove(output + CCSDS_PRIMARY_HEADER_SIZE + data_field.first.size,
176 data_field.second.data,
177 data_field.second.size);
178 }
179
180 status = ccsds_primary_header_encode(header, output, capacity);
181 if (status != CCSDS_STATUS_OK) return status;
182
183 if (pec_size != 0U) {
184 ccsds_store_be16(
185 output + CCSDS_PRIMARY_HEADER_SIZE
186 + data_field.first.size + data_field.second.size,
187 crc16);
188 }
189
190 *written_out = required;
191 return CCSDS_STATUS_OK;
192}
