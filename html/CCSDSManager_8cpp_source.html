14 bool sameSecondaryContract(const ccsds::Packet &lhs,
15 const ccsds::Packet &rhs) noexcept {
16 const auto lhsHeader = lhs.getSecondaryHeader();
17 const auto rhsHeader = rhs.getSecondaryHeader();
18 if (static_cast<bool>(lhsHeader) != static_cast<bool>(rhsHeader)) return false;
19 if (!lhsHeader) return true;
20 if (lhsHeader->getType() != rhsHeader->getType()
21 || lhsHeader->getDirection() != rhsHeader->getDirection()
22 || lhsHeader->isPusHeader() != rhsHeader->isPusHeader()) {
23 return false;
24 }
25 if (!lhsHeader->isPusHeader()) return true;
26 return ccsds::pus::sameTailoring(
27 static_cast<const ccsds::pus::SecondaryHeader &>(*lhsHeader),
28 static_cast<const ccsds::pus::SecondaryHeader &>(*rhsHeader));
29 }
30}

37std::uint16_t ccsds::Manager::packetIdentifier(const Packet &packet) {
38 const auto &header = packet.getPrimaryHeader();
39 return static_cast<std::uint16_t>(
40 (static_cast<std::uint16_t>(header.getVersionNumber()) << 13U)
41 | (static_cast<std::uint16_t>(header.getType()) << 12U)
42 | (static_cast<std::uint16_t>(header.getSecondaryHeaderFlag()) << 11U)
43 | header.getAPID());
44}

55ccsds::ResultBool ccsds::Manager::validatePacketIdentifier(const Packet &packet) const {
56 if (!hasIdentifierBinding()) return true;
57 const auto &bound = m_templateIsSet ? m_templatePacket : m_packets.front();
58 RET_IF_ERR_MSG(packetIdentifier(packet) != packetIdentifier(bound),
59 ErrorCode::INVALID_HEADER_DATA,
60 "Packet identifier does not match the Manager-bound stream identifier");
61 RET_IF_ERR_MSG(packet.getPacketErrorControlMode() != bound.getPacketErrorControlMode(),
62 ErrorCode::INVALID_DATA,
63 "Packet error-control mode does not match the Manager template");
64 RET_IF_ERR_MSG(!sameSecondaryContract(packet, bound),
65 ErrorCode::INVALID_SECONDARY_HEADER_DATA,
66 "Secondary-header type or tailoring does not match the Manager template");
67 return true;
68}

85 if (!getAutoSequenceCountEnable()) return;

90ccsds::ResultBool ccsds::Manager::setPacketTemplate(Packet packet) {
91 RET_IF_ERR_MSG(m_templateIsSet, ErrorCode::TEMPLATE_SET_FAILURE,
92 "Cannot set Template as it is already set, please clear Manager first");
93 RET_IF_ERR_MSG(packet.getPrimaryHeader().getHeaderStatus() == INVALID,
94 ErrorCode::INVALID_HEADER_DATA, "Cannot set an invalid packet template");
95 m_templatePacket = std::move(packet);
96 m_sequenceCount = (m_sequenceCount & AUTO_SEQUENCE_DISABLED_MASK)
97 | (m_templatePacket.getPrimaryHeader().getSequenceCount() & CCSDS_SEQUENCE_COUNT_MAX);
98 m_templatePacket.setUpdatePacketEnable(false);
99 m_validator.clear();
100 m_validator.setTemplatePacket(m_templatePacket);
101 m_validator.configure(true, true, true);
102 m_validateEnable = true;
103 m_templateIsSet = true;
104 return true;
105}

142ccsds::ResultBool ccsds::Manager::setApplicationData(const std::vector<std::uint8_t> &data) {
143 RET_IF_ERR_MSG(data.empty(), ErrorCode::NO_DATA,
144 "Cannot set Application data, Provided data is empty");
145 RET_IF_ERR_MSG(!m_templateIsSet, ErrorCode::INVALID_HEADER_DATA,
146 "Cannot set Application data, No template has been set");
147 RET_IF_ERR_MSG(m_templatePacket.getPrimaryHeader().getHeaderStatus() == INVALID,
148 ErrorCode::INVALID_HEADER_DATA,
149 "Cannot set Application data using an invalid packet template");
150
151 const auto maxBytesPerPacket = m_templatePacket.getDataFieldMaximumSize();
152 RET_IF_ERR_MSG(maxBytesPerPacket == 0U, ErrorCode::INVALID_APPLICATION_DATA,
153 "Cannot segment application data into a zero-sized packet data field");
154 std::size_t packetCount{};
155 const auto countStatus = ccsds_segmentation_packet_count(
156 data.size(), static_cast<std::size_t>(maxBytesPerPacket), &packetCount);
157 RET_IF_ERR_MSG(countStatus != CCSDS_STATUS_OK, ErrorCode::INVALID_APPLICATION_DATA,
158 "Cannot plan CCSDS application-data segmentation");
159
160 std::vector<Packet> generated;
161 generated.reserve(packetCount);
162 const auto startCount = getSequenceCount();
163 const bool autoSequence = getAutoSequenceCountEnable();
164
165 for (std::size_t index = 0U; index < packetCount; ++index) {
166 ccsds_segment_plan_t plan{};
167 const auto planStatus = ccsds_segmentation_plan(
168 data.size(), static_cast<std::size_t>(maxBytesPerPacket), index,
169 startCount, autoSequence ? 1 : 0, &plan);
170 RET_IF_ERR_MSG(planStatus != CCSDS_STATUS_OK, ErrorCode::INVALID_APPLICATION_DATA,
171 "Cannot plan CCSDS application-data segment");
172
173 Packet packet = m_templatePacket;
174 const std::vector<std::uint8_t> chunk(
175 data.begin() + static_cast<std::ptrdiff_t>(plan.offset),
176 data.begin() + static_cast<std::ptrdiff_t>(plan.offset + plan.size));
177 packet.setSequenceFlags(static_cast<ESequenceFlag>(plan.sequence_flags));
178 FORWARD_RESULT(packet.setSequenceCount(plan.sequence_count));
179 FORWARD_RESULT(packet.setApplicationData(chunk));
180 packet.setUpdatePacketEnable(m_updateEnable);
181 generated.push_back(std::move(packet));
182 }
183
184 m_packets = std::move(generated);
185 const auto nextCount = ccsds_sequence_after_packets(
186 startCount, packetCount, autoSequence ? 1 : 0);
187 m_sequenceCount = (m_sequenceCount & AUTO_SEQUENCE_DISABLED_MASK) | nextCount;
188 return true;
189}

217ccsds::ResultBuffer ccsds::Manager::getPacketsBuffer() const {
218 const auto prefixSize = ccsds_packet_stream_prefix_size(m_syncPattEnable ? 1 : 0);
219 std::size_t totalSize = 0U;
220 for (const auto &packet : m_packets) {
221 const auto packetSize = packet.getSerializedSize();
222 RET_IF_ERR_MSG(packetSize > std::numeric_limits<std::size_t>::max() - prefixSize,
223 ErrorCode::INVALID_DATA,
224 "Cannot serialize packet stream: frame size overflow.");
225 const auto frameSize = prefixSize + packetSize;
226 RET_IF_ERR_MSG(frameSize > std::numeric_limits<std::size_t>::max() - totalSize,
227 ErrorCode::INVALID_DATA,
228 "Cannot serialize packet stream: total size overflow.");
229 totalSize += frameSize;
230 }
231
232 std::vector<std::uint8_t> buffer(totalSize);
233 std::size_t offset = 0U;
234 for (auto packet : m_packets) {
235 std::size_t prefixWritten = 0U;
236 const auto prefixStatus = ccsds_packet_stream_write_prefix(
237 m_syncPattEnable ? 1 : 0,
238 m_syncPattern,
239 buffer.empty() ? nullptr : buffer.data() + offset,
240 buffer.size() - offset,
241 &prefixWritten);
242 RET_IF_ERR_MSG(prefixStatus != CCSDS_STATUS_OK, ErrorCode::INVALID_DATA,
243 "Cannot serialize packet stream synchronization prefix.");
244 offset += prefixWritten;
245
246 const auto written = packet.serialize(
247 buffer.data() + offset, buffer.size() - offset);
248 if (!written) return written.error();
249 offset += written.value();
250 }
251
252 RET_IF_ERR_MSG(offset != buffer.size(), ErrorCode::INVALID_DATA,
253 "Cannot serialize packet stream: unexpected final size.");
254 return buffer;
255}

257ccsds::ResultBuffer ccsds::Manager::getApplicationDataBuffer() {
258 RET_IF_ERR_MSG(m_packets.empty(), ErrorCode::NO_DATA,
259 "Cannot get Application data, no packets have been set.");
260
261 std::size_t totalSize = 0U;
262 for (const auto &packet : m_packets) {
263 const auto view = packet.getDataField().getApplicationDataView();
264 RET_IF_ERR_MSG(view.size > std::numeric_limits<std::size_t>::max() - totalSize,
265 ErrorCode::INVALID_DATA,
266 "Cannot reconstruct Application data: total size overflow.");
267 totalSize += view.size;
268 }
269
270 std::vector<std::uint8_t> data(totalSize);
271 ccsds_reassembly_state_t reassembly{};
272 ccsds_reassembly_reset(&reassembly, m_validateEnable ? 1 : 0);
273
274 for (std::size_t index = 0U; index < m_packets.size(); ++index) {
275 const auto &packet = m_packets[index];
276 if (m_validateEnable) {
277 RET_IF_ERR_MSG(!m_validator.validate(packet), ErrorCode::VALIDATION_FAILURE,
278 "Validation failure for packet at index " + std::to_string(index));
279 }
280
281 const auto &header = packet.getPrimaryHeader();
282 const ccsds_primary_header_t coreHeader{
283 header.getVersionNumber(),
284 header.getType(),
285 header.getSecondaryHeaderFlag(),
286 header.getAPID(),
287 header.getSequenceFlags(),
288 header.getSequenceCount(),
289 header.getDataLength()
290 };
291 const auto applicationData = packet.getDataField().getApplicationDataView();
292 std::size_t writtenNow = 0U;
293 int complete = 0;
294 const auto status = ccsds_reassembly_accept(
295 &reassembly, &coreHeader, applicationData,
296 data.empty() ? nullptr : data.data(), data.size(),
297 &writtenNow, &complete);
298 (void)writtenNow;
299 (void)complete;
300 RET_IF_ERR_MSG(status != CCSDS_STATUS_OK, ErrorCode::VALIDATION_FAILURE,
301 "Application-data reassembly failed at packet index "
302 + std::to_string(index));
303 }
304
305 RET_IF_ERR_MSG(reassembly.written != data.size(), ErrorCode::INVALID_DATA,
306 "Application-data reassembly produced an unexpected size.");
307 return data;
308}

320ccsds::ResultBool ccsds::Manager::addPacket(Packet packet) {
321 RET_IF_ERR_MSG(packet.getPrimaryHeader().getHeaderStatus() == INVALID,
322 ErrorCode::INVALID_HEADER_DATA,
323 "Cannot add packet with an invalid primary header");
324 FORWARD_RESULT(validatePacketIdentifier(packet));
325 if (m_validateEnable && !m_updateEnable) {
326 auto validator = m_validator;
327 if (!m_templateIsSet && m_packets.empty()) validator.configure(true, true, false);
328 RET_IF_ERR_MSG(!validator.validate(packet), ErrorCode::VALIDATION_FAILURE,
329 "packet is not valid");
330 m_validator = std::move(validator);
331 }
332 packet.setUpdatePacketEnable(m_updateEnable);
333 m_packets.push_back(std::move(packet));
334 syncSequenceCountFromPacket(m_packets.back());
335 return true;
336}

360ccsds::ResultBool ccsds::Manager::load(
361 const std::uint8_t *data, const std::size_t size) {
362 RET_IF_ERR_MSG(data == nullptr, ErrorCode::NULL_POINTER,
363 "Cannot load packet stream, raw buffer pointer is null");
364 RET_IF_ERR_MSG(size < 7U, ErrorCode::INVALID_DATA,
365 "invalid packet buffer size");
366
367 Manager staged = *this;
368 ccsds_packet_stream_t stream{};
369 const auto initStatus = ccsds_packet_stream_init(
370 &stream, data, size,
371 CCSDS_PACKET_ERROR_CONTROL_NONE,
372 nullptr,
373 staged.m_syncPattEnable ? 1 : 0,
374 staged.m_syncPattern);
375 RET_IF_ERR_MSG(initStatus != CCSDS_STATUS_OK,
376 static_cast<ErrorCode>(initStatus),
377 "Cannot initialize CCSDS packet stream.");
378
379 while (ccsds_packet_stream_remaining(&stream) != 0U) {
380 ccsds_packet_view_t view{};
381 std::size_t frameConsumed = 0U;
382 const auto streamStatus =
383 ccsds_packet_stream_next(&stream, &view, &frameConsumed);
384 (void)frameConsumed;
385 RET_IF_ERR_MSG(streamStatus != CCSDS_STATUS_OK,
386 static_cast<ErrorCode>(streamStatus),
387 "Cannot parse next CCSDS packet from stream.");
388
389 Packet packet = staged.boundParserPacket();
390 packet.setPacketErrorControlMode(staged.boundPacketErrorControlMode());
391 std::size_t consumed{};
392 ASSIGN_CP(consumed, packet.deserializeBounded(view.packet.data, view.packet.size));
393 RET_IF_ERR_MSG(consumed != view.packet.size, ErrorCode::INVALID_DATA,
394 "Parsed packet size differs from stream framing.");
395 FORWARD_RESULT(staged.addPacket(std::move(packet)));
396 }
397
398 *this = std::move(staged);
399 return true;
400}
