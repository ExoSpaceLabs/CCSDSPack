11 bool sameSecondaryContract(const ccsds::Packet &lhs,
12 const ccsds::Packet &rhs) noexcept {
13 const auto lhsHeader = lhs.getSecondaryHeader();
14 const auto rhsHeader = rhs.getSecondaryHeader();
15 if (static_cast<bool>(lhsHeader) != static_cast<bool>(rhsHeader)) return false;
16 if (!lhsHeader) return true;
17 if (lhsHeader->getType() != rhsHeader->getType()
18 || lhsHeader->getDirection() != rhsHeader->getDirection()
19 || lhsHeader->isPusHeader() != rhsHeader->isPusHeader()) {
20 return false;
21 }
22 if (!lhsHeader->isPusHeader()) return true;
23 return ccsds::pus::sameTailoring(
24 static_cast<const ccsds::pus::SecondaryHeader &>(*lhsHeader),
25 static_cast<const ccsds::pus::SecondaryHeader &>(*rhsHeader));
26 }
27}

34std::uint16_t ccsds::Manager::packetIdentifier(const Packet &packet) {
35 const auto &header = packet.getPrimaryHeader();
36 return static_cast<std::uint16_t>(
37 (static_cast<std::uint16_t>(header.getVersionNumber()) << 13U)
38 | (static_cast<std::uint16_t>(header.getType()) << 12U)
39 | (static_cast<std::uint16_t>(header.getSecondaryHeaderFlag()) << 11U)
40 | header.getAPID());
41}

52ccsds::ResultBool ccsds::Manager::validatePacketIdentifier(const Packet &packet) const {
53 if (!hasIdentifierBinding()) return true;
54 const auto &bound = m_templateIsSet ? m_templatePacket : m_packets.front();
55 RET_IF_ERR_MSG(packetIdentifier(packet) != packetIdentifier(bound),
56 ErrorCode::INVALID_HEADER_DATA,
57 "Packet identifier does not match the Manager-bound stream identifier");
58 RET_IF_ERR_MSG(packet.getPacketErrorControlMode() != bound.getPacketErrorControlMode(),
59 ErrorCode::INVALID_DATA,
60 "Packet error-control mode does not match the Manager template");
61 RET_IF_ERR_MSG(!sameSecondaryContract(packet, bound),
62 ErrorCode::INVALID_SECONDARY_HEADER_DATA,
63 "Secondary-header type or tailoring does not match the Manager template");
64 return true;
65}

94ccsds::ResultBool ccsds::Manager::setPacketTemplate(Packet packet) {
95 RET_IF_ERR_MSG(m_templateIsSet, ErrorCode::TEMPLATE_SET_FAILURE,
96 "Cannot set Template as it is already set, please clear Manager first");
97 RET_IF_ERR_MSG(packet.getPrimaryHeader().getHeaderStatus() == INVALID,
98 ErrorCode::INVALID_HEADER_DATA, "Cannot set an invalid packet template");
99 m_templatePacket = std::move(packet);
100 m_sequenceCount = (m_sequenceCount & AUTO_SEQUENCE_DISABLED_MASK)
101 | (m_templatePacket.getPrimaryHeader().getSequenceCount() & SEQUENCE_COUNT_MASK);
102 m_templatePacket.setUpdatePacketEnable(false);
103 m_validator.clear();
104 m_validator.setTemplatePacket(m_templatePacket);
105 m_validator.configure(true, true, true);
106 m_validateEnable = true;
107 m_templateIsSet = true;
108 return true;
109}

146ccsds::ResultBool ccsds::Manager::setApplicationData(const std::vector<std::uint8_t> &data) {
147 RET_IF_ERR_MSG(data.empty(), ErrorCode::NO_DATA,
148 "Cannot set Application data, Provided data is empty");
149 RET_IF_ERR_MSG(!m_templateIsSet, ErrorCode::INVALID_HEADER_DATA,
150 "Cannot set Application data, No template has been set");
151 RET_IF_ERR_MSG(m_templatePacket.getPrimaryHeader().getHeaderStatus() == INVALID,
152 ErrorCode::INVALID_HEADER_DATA,
153 "Cannot set Application data using an invalid packet template");
154
155 const auto maxBytesPerPacket = m_templatePacket.getDataFieldMaximumSize();
156 RET_IF_ERR_MSG(maxBytesPerPacket == 0U, ErrorCode::INVALID_APPLICATION_DATA,
157 "Cannot segment application data into a zero-sized packet data field");
158 const auto packetCount =
159 (data.size() + static_cast<std::size_t>(maxBytesPerPacket) - 1U)
160 / static_cast<std::size_t>(maxBytesPerPacket);
161
162 std::vector<Packet> generated;
163 generated.reserve(packetCount);
164 auto nextCount = getSequenceCount();
165 std::size_t offset = 0U;
166 for (std::size_t index = 0U; index < packetCount; ++index) {
167 Packet packet = m_templatePacket;
168 const auto bytes = std::min<std::size_t>(maxBytesPerPacket, data.size() - offset);
169 const std::vector<std::uint8_t> chunk(
170 data.begin() + static_cast<std::ptrdiff_t>(offset),
171 data.begin() + static_cast<std::ptrdiff_t>(offset + bytes));
172 ESequenceFlag flags = UNSEGMENTED;
173 if (packetCount > 1U) {
174 if (index == 0U) flags = FIRST_SEGMENT;
175 else if (index + 1U == packetCount) flags = LAST_SEGMENT;
176 else flags = CONTINUING_SEGMENT;
177 }
178 packet.setSequenceFlags(flags);
179 FORWARD_RESULT(packet.setSequenceCount(nextCount));
180 FORWARD_RESULT(packet.setApplicationData(chunk));
181 packet.setUpdatePacketEnable(m_updateEnable);
182 generated.push_back(std::move(packet));
183 if (getAutoSequenceCountEnable()) nextCount = static_cast<std::uint16_t>((nextCount + 1U) & SEQUENCE_COUNT_MASK);
184 offset += bytes;
185 }
186 m_packets = std::move(generated);
187 m_sequenceCount = (m_sequenceCount & AUTO_SEQUENCE_DISABLED_MASK) | nextCount;
188 return true;
189}

204ccsds::ResultBuffer ccsds::Manager::getPacketBufferAtIndex(const std::uint16_t index) {
205 RET_IF_ERR_MSG(index >= m_packets.size(), ErrorCode::INVALID_DATA,
206 "Cannot get packet, index is out of bounds");
207 auto packet = m_packets[index];
208 if (m_validateEnable) {
209 const auto updateResult = packet.update();
210 if (!updateResult) return updateResult.error();
211 RET_IF_ERR_MSG(!m_validator.validate(packet), ErrorCode::VALIDATION_FAILURE,
212 "Validation failure for packet at index " + std::to_string(index));
213 }
214 return packet.serialize();
215}

217ccsds::ResultBuffer ccsds::Manager::getPacketsBuffer() const {
218 std::vector<std::uint8_t> buffer;
219 for (auto packet : m_packets) {
220 if (m_syncPattEnable) {
221 buffer.push_back(static_cast<std::uint8_t>((m_syncPattern >> 24U) & 0xFFU));
222 buffer.push_back(static_cast<std::uint8_t>((m_syncPattern >> 16U) & 0xFFU));
223 buffer.push_back(static_cast<std::uint8_t>((m_syncPattern >> 8U) & 0xFFU));
224 buffer.push_back(static_cast<std::uint8_t>(m_syncPattern & 0xFFU));
225 }
226 std::vector<std::uint8_t> packetBuffer;
227 ASSIGN_MV(packetBuffer, packet.serialize());
228 buffer.insert(buffer.end(), packetBuffer.begin(), packetBuffer.end());
229 }
230 return buffer;
231}

258ccsds::ResultBool ccsds::Manager::addPacket(Packet packet) {
259 RET_IF_ERR_MSG(packet.getPrimaryHeader().getHeaderStatus() == INVALID,
260 ErrorCode::INVALID_HEADER_DATA,
261 "Cannot add packet with an invalid primary header");
262 FORWARD_RESULT(validatePacketIdentifier(packet));
263 if (m_validateEnable && !m_updateEnable) {
264 auto validator = m_validator;
265 if (!m_templateIsSet && m_packets.empty()) validator.configure(true, true, false);
266 RET_IF_ERR_MSG(!validator.validate(packet), ErrorCode::VALIDATION_FAILURE,
267 "packet is not valid");
268 m_validator = std::move(validator);
269 }
270 packet.setUpdatePacketEnable(m_updateEnable);
271 m_packets.push_back(std::move(packet));
272 syncSequenceCountFromPacket(m_packets.back());
273 return true;
274}

291ccsds::ResultBool ccsds::Manager::load(const std::vector<std::uint8_t> &packetsBuffer) {
292 RET_IF_ERR_MSG(packetsBuffer.size() < 7U, ErrorCode::INVALID_DATA,
293 "invalid packet buffer size");
294 Manager staged = *this;
295 std::size_t offset{0U};
296 while (offset < packetsBuffer.size()) {
297 if (staged.m_syncPattEnable) {
298 RET_IF_ERR_MSG(packetsBuffer.size() - offset < 4U, ErrorCode::INVALID_DATA,
299 "Truncated sync pattern.");
300 const std::uint32_t value =
301 (static_cast<std::uint32_t>(packetsBuffer[offset]) << 24U)
302 | (static_cast<std::uint32_t>(packetsBuffer[offset + 1U]) << 16U)
303 | (static_cast<std::uint32_t>(packetsBuffer[offset + 2U]) << 8U)
304 | static_cast<std::uint32_t>(packetsBuffer[offset + 3U]);
305 RET_IF_ERR_MSG(value != staged.m_syncPattern, ErrorCode::INVALID_DATA,
306 "Sync Pattern mismatch.");
307 offset += 4U;
308 }
309 RET_IF_ERR_MSG(packetsBuffer.size() - offset < 6U, ErrorCode::INVALID_DATA,
310 "Truncated CCSDS primary header.");
311 const std::vector<std::uint8_t> remaining(
312 packetsBuffer.begin() + static_cast<std::ptrdiff_t>(offset), packetsBuffer.end());
313 Packet packet = staged.boundParserPacket();
314 packet.setPacketErrorControlMode(staged.boundPacketErrorControlMode());
315 std::size_t consumed{};
316 ASSIGN_CP(consumed, packet.deserializeBounded(remaining));
317 FORWARD_RESULT(staged.addPacket(std::move(packet)));
318 offset += consumed;
319 }
320 *this = std::move(staged);
321 return true;
322}

336ccsds::ResultBool ccsds::Manager::readTemplate(const std::string &filename) {
337 Packet templatePacket;
338 templatePacket.setUpdatePacketEnable(false);
339 if (stringEndsWith(filename, ".bin")) {
340 std::vector<std::uint8_t> buffer;
341 ASSIGN_CP(buffer, readBinaryFile(filename));
342 FORWARD_RESULT(templatePacket.deserialize(buffer));
343 } else if (stringEndsWith(filename, ".cfg")) {
344 FORWARD_RESULT(templatePacket.loadFromConfigFile(filename));
345 } else {
346 return Error{INVALID_DATA,
347 "Cannot load template, invalid file provided [supported extensions [.bin, .cfg]]"};
348 }
349 return setPacketTemplate(std::move(templatePacket));
350}
