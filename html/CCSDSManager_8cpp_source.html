25std::uint16_t CCSDS::Manager::packetIdentifier(const Packet &packet) {
26 const auto &header = packet.getPrimaryHeader();
27 return static_cast<std::uint16_t>(
28 (static_cast<std::uint16_t>(header.getVersionNumber()) << 13U)
29 | (static_cast<std::uint16_t>(header.getType()) << 12U)
30 | (static_cast<std::uint16_t>(header.getDataFieldHeaderFlag()) << 11U)
31 | header.getAPID());
32}

70CCSDS::ResultBool CCSDS::Manager::setPacketTemplate(Packet packet) {
71 RET_IF_ERR_MSG(m_templateIsSet, ErrorCode::TEMPLATE_SET_FAILURE,
72 "Cannot set Template as it is already set, please clear Manager first");
73 RET_IF_ERR_MSG(packet.getPrimaryHeader().getHeaderStatus() == INVALID,
74 ErrorCode::INVALID_HEADER_DATA, "Cannot set an invalid packet template");
75
76 packet.update();
77 RET_IF_ERR_MSG(packet.getPrimaryHeader().getHeaderStatus() == INVALID,
78 ErrorCode::INVALID_HEADER_DATA, "Cannot finalize an invalid packet template");
79
80 m_templatePacket = std::move(packet);
81 m_sequenceCount = (m_sequenceCount & AUTO_SEQUENCE_DISABLED_MASK)
82 | (m_templatePacket.getPrimaryHeader().getSequenceCount()
83 & SEQUENCE_COUNT_MASK);
84 m_templatePacket.setUpdatePacketEnable(false);
85 m_validator.clear();
86 m_validator.setTemplatePacket(m_templatePacket);
87 m_validator.configure(true, true, true);
88 m_validateEnable = true;
89 m_templateIsSet = true;
90 return true;
91}

135CCSDS::ResultBool CCSDS::Manager::setApplicationData(
136 const std::vector<std::uint8_t> &data) {
137 RET_IF_ERR_MSG(data.empty(), ErrorCode::NO_DATA,
138 "Cannot set Application data, Provided data is empty");
139 RET_IF_ERR_MSG(!m_templateIsSet, ErrorCode::INVALID_HEADER_DATA,
140 "Cannot set Application data, No template has been set");
141 RET_IF_ERR_MSG(m_templatePacket.getPrimaryHeader().getHeaderStatus() == INVALID,
142 ErrorCode::INVALID_HEADER_DATA,
143 "Cannot set Application data using an invalid packet template");
144
145 const auto maxBytesPerPacket = m_templatePacket.getDataFieldMaximumSize();
146 RET_IF_ERR_MSG(maxBytesPerPacket == 0U, ErrorCode::INVALID_APPLICATION_DATA,
147 "Cannot segment application data into a zero-sized packet data field");
148
149 const auto packetCount =
150 (data.size() + static_cast<std::size_t>(maxBytesPerPacket) - 1U)
151 / static_cast<std::size_t>(maxBytesPerPacket);
152
153 std::vector<Packet> generated;
154 generated.reserve(packetCount);
155 auto nextCount = getSequenceCount();
156
157 std::size_t offset = 0U;
158 for (std::size_t index = 0U; index < packetCount; ++index) {
159 Packet packet = m_templatePacket;
160 const auto bytes = std::min<std::size_t>(maxBytesPerPacket, data.size() - offset);
161 const std::vector<std::uint8_t> chunk(
162 data.begin() + static_cast<std::ptrdiff_t>(offset),
163 data.begin() + static_cast<std::ptrdiff_t>(offset + bytes));
164
165 ESequenceFlag flags = UNSEGMENTED;
166 if (packetCount > 1U) {
167 if (index == 0U) flags = FIRST_SEGMENT;
168 else if (index + 1U == packetCount) flags = LAST_SEGMENT;
169 else flags = CONTINUING_SEGMENT;
170 }
171
172 packet.setSequenceFlags(flags);
173 FORWARD_RESULT(packet.setSequenceCount(nextCount));
174 FORWARD_RESULT(packet.setApplicationData(chunk));
175 packet.setUpdatePacketEnable(m_updateEnable);
176 generated.push_back(std::move(packet));
177
178 if (getAutoSequenceCountEnable()) {
179 nextCount = static_cast<std::uint16_t>((nextCount + 1U) & SEQUENCE_COUNT_MASK);
180 }
181 offset += bytes;
182 }
183
184 m_packets = std::move(generated);
185 m_sequenceCount = (m_sequenceCount & AUTO_SEQUENCE_DISABLED_MASK) | nextCount;
186 return true;
187}

225std::vector<std::uint8_t> CCSDS::Manager::getPacketsBuffer() const {
226 std::vector<std::uint8_t> buffer;
227 for (auto packet : m_packets) {
228 if (m_syncPattEnable) {
229 buffer.push_back(static_cast<std::uint8_t>((m_syncPattern >> 24U) & 0xFFU));
230 buffer.push_back(static_cast<std::uint8_t>((m_syncPattern >> 16U) & 0xFFU));
231 buffer.push_back(static_cast<std::uint8_t>((m_syncPattern >> 8U) & 0xFFU));
232 buffer.push_back(static_cast<std::uint8_t>(m_syncPattern & 0xFFU));
233 }
234
235 const auto packetBuffer = packet.serialize();
236 if (packetBuffer.empty()) return {};
237 buffer.insert(buffer.end(), packetBuffer.begin(), packetBuffer.end());
238 }
239 return buffer;
240}

279CCSDS::ResultBool CCSDS::Manager::addPacket(Packet packet) {
280 RET_IF_ERR_MSG(packet.getPrimaryHeader().getHeaderStatus() == INVALID,
281 ErrorCode::INVALID_HEADER_DATA,
282 "Cannot add packet with an invalid primary header");
283 FORWARD_RESULT(validatePacketIdentifier(packet));
284
285 if (m_validateEnable && !m_updateEnable) {
286 auto validator = m_validator;
287 if (!m_templateIsSet && m_packets.empty()) {
288 validator.configure(true, true, false);
289 }
290 RET_IF_ERR_MSG(!validator.validate(packet), ErrorCode::VALIDATION_FAILURE,
291 "packet is not valid");
292 m_validator = std::move(validator);
293 }
294
295 packet.setUpdatePacketEnable(m_updateEnable);
296 m_packets.push_back(std::move(packet));
297 syncSequenceCountFromPacket(m_packets.back());
298 return true;
299}

311 Manager staged = *this;
312 for (const auto &packet : packets) {
313 const auto result = staged.addPacket(packet);
314 if (!result) return result.error();
315 }
316 *this = std::move(staged);

320CCSDS::ResultBool CCSDS::Manager::load(
321 const std::vector<std::uint8_t> &packetsBuffer) {
322 RET_IF_ERR_MSG(packetsBuffer.size() < 7U, ErrorCode::INVALID_DATA,
323 "invalid packet buffer size");
324
325 Manager staged = *this;
326 std::size_t offset{0U};
327 while (offset < packetsBuffer.size()) {
328 if (staged.m_syncPattEnable) {
329 RET_IF_ERR_MSG(packetsBuffer.size() - offset < 4U, ErrorCode::INVALID_DATA,
330 "Truncated sync pattern.");
331 const std::uint32_t value =
332 (static_cast<std::uint32_t>(packetsBuffer[offset]) << 24U)
333 | (static_cast<std::uint32_t>(packetsBuffer[offset + 1U]) << 16U)
334 | (static_cast<std::uint32_t>(packetsBuffer[offset + 2U]) << 8U)
335 | static_cast<std::uint32_t>(packetsBuffer[offset + 3U]);
336 RET_IF_ERR_MSG(value != staged.m_syncPattern, ErrorCode::INVALID_DATA,
337 "Sync Pattern mismatch.");
338 offset += 4U;
339 }
340
341 RET_IF_ERR_MSG(packetsBuffer.size() - offset < 6U, ErrorCode::INVALID_DATA,
342 "Truncated CCSDS primary header.");
343
344 const std::vector<std::uint8_t> remaining(
345 packetsBuffer.begin() + static_cast<std::ptrdiff_t>(offset),
346 packetsBuffer.end());
347 Packet packet;
348 packet.setPacketErrorControlMode(staged.boundPacketErrorControlMode());
349 std::size_t consumed{};
350 ASSIGN_CP(consumed, packet.deserializeBounded(remaining));
351 const auto addResult = staged.addPacket(std::move(packet));
352 if (!addResult) return addResult.error();
353 offset += consumed;
354 }
355
356 *this = std::move(staged);
357 return true;
358}

376CCSDS::ResultBool CCSDS::Manager::readTemplate(const std::string &filename) {
377 Packet templatePacket;
378 templatePacket.setUpdatePacketEnable(false);
379 if (stringEndsWith(filename, ".bin")) {
380 std::vector<std::uint8_t> buffer;
381 ASSIGN_CP(buffer, readBinaryFile(filename));
382 FORWARD_RESULT(templatePacket.deserialize(buffer));
383 } else if (stringEndsWith(filename, ".cfg")) {
384 FORWARD_RESULT(templatePacket.loadFromConfigFile(filename));
385 } else {
386 return Error{INVALID_DATA,
387 "Cannot load template, invalid file provided [supported extensions [.bin, .cfg]]"};
388 }
389 FORWARD_RESULT(setPacketTemplate(std::move(templatePacket)));
390 return true;
391}
