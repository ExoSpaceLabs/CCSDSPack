16void printHelpDecoder() {

41struct DecoderConfig {
42 bool validationEnable{};
43 bool syncPatternEnable{};
44 std::uint32_t syncPattern{0x1ACFFC1DU};
45};

47CCSDS::Result<DecoderConfig> readDecoderConfig(const Config &cfg) {
48 DecoderConfig settings;
49 RET_IF_ERR_MSG(!cfg.isKey("validation_enable"),
50 static_cast<CCSDS::ErrorCode>(CONFIG_MISSING_PARAMETER),
51 "Config: Missing bool field: validation_enable");
52 ASSIGN_CP(settings.validationEnable, cfg.get<bool>("validation_enable"));
53
54 if (cfg.isKey("sync_pattern_enable")) {
55 ASSIGN_CP(settings.syncPatternEnable, cfg.get<bool>("sync_pattern_enable"));
56 }
57 if (settings.syncPatternEnable && cfg.isKey("sync_pattern")) {
58 int encoded{};
59 ASSIGN_CP(encoded, cfg.get<int>("sync_pattern"));
60 settings.syncPattern = static_cast<std::uint32_t>(encoded);
61 }
62 return settings;
63}

66int main(const int argc, char *argv[]) {
67 const std::string appName = "ccsds_decoder";
68 std::unordered_map<std::string, std::string> allowed{
69 {"h", "help"}, {"v", "verbose"}, {"i", "input"},
70 {"o", "output"}, {"c", "config"}, {"e", "packet-error-control"},
71 {"t", "trailing-output"}
72 };
73 const std::set<std::string> booleanArgs{"verbose", "help"};
74 std::unordered_map<std::string, std::string> args{
75 {"verbose", "false"}, {"help", "false"}
76 };
77
78 const auto start = std::chrono::high_resolution_clock::now();
79 if (const auto result = parseArguments(argc, argv, allowed, args, booleanArgs); !result) {
80 return printError(result.error());
81 }
82 if (args["help"] == "true") {
83 printHelpDecoder();
84 return 0;
85 }
86
87 const auto requirePath = [&](const char *key, const char *description)
88 -> CCSDS::Result<std::string> {
89 const auto it = args.find(key);
90 if (it == args.end() || it->second.empty()) {
91 return CCSDS::Error{static_cast<CCSDS::ErrorCode>(ARG_PARSE_ERROR),
92 std::string(description) + " must be specified"};
93 }
94 return it->second;
95 };
96
97 std::string input;
98 std::string output;
99 std::string configFile;
100 {
101 auto result = requirePath("input", "Input file");
102 if (!result) { printHelpDecoder(); return printError(result.error()); }
103 input = result.value();
104 }
105 {
106 auto result = requirePath("output", "Output file");
107 if (!result) { printHelpDecoder(); return printError(result.error()); }
108 output = result.value();
109 }
110 {
111 auto result = requirePath("config", "Config file");
112 if (!result) { printHelpDecoder(); return printError(result.error()); }
113 configFile = result.value();
114 }
115
116 if (!fileExists(input)) {
117 return printError(CCSDS::Error{static_cast<CCSDS::ErrorCode>(ARG_PARSE_ERROR),
118 "Input file does not exist: " + input});
119 }
120 if (!fileExists(configFile)) {
121 return printError(CCSDS::Error{static_cast<CCSDS::ErrorCode>(ARG_PARSE_ERROR),
122 "Config file does not exist: " + configFile});
123 }
124
125 customConsole(appName, "reading CCSDS configuration file: " + configFile);
126 Config cfg;
127 if (const auto result = cfg.load(configFile); !result) return printError(result.error());
128
129 DecoderConfig settings;
130 {
131 const auto result = readDecoderConfig(cfg);
132 if (!result) return printError(result.error());
133 settings = result.value();
134 }
135
136 CCSDS::Packet templatePacket;
137 if (const auto result = templatePacket.loadFromConfig(cfg); !result) {
138 return printError(result.error());
139 }
140 if (const auto it = args.find("packet-error-control"); it != args.end()) {
141 const auto result = parsePacketErrorControlMode(it->second);
142 if (!result) return printError(result.error());
143 templatePacket.setPacketErrorControlMode(result.value());
144 }
145
146 CCSDS::Manager manager;
147 if (const auto result = manager.setPacketTemplate(std::move(templatePacket)); !result) {
148 return printError(result.error());
149 }
150 manager.setSyncPattern(settings.syncPattern);
151 manager.setSyncPatternEnable(settings.syncPatternEnable);
152 manager.setAutoValidateEnable(false);
153
154 std::vector<std::uint8_t> inputBytes;
155 customConsole(appName, "reading data from " + input);
156 {
157 const auto result = readBinaryFile(input);
158 if (!result) return printError(result.error());
159 inputBytes = result.value();
160 }
161
162 PacketStreamLayout layout;
163 {
164 const auto result = inspectPacketStream(inputBytes, settings.syncPatternEnable,
165 settings.syncPattern, true);
166 if (!result) return printError(result.error());
167 layout = result.value();
168 }
169 if (layout.packets.empty()) {
170 return printError(CCSDS::Error{static_cast<CCSDS::ErrorCode>(INVALID_INPUT_DATA),
171 "Input does not begin with a complete CCSDS packet"});
172 }
173
174 const std::vector<std::uint8_t> packetPrefix(
175 inputBytes.begin(),
176 inputBytes.begin() + static_cast<std::ptrdiff_t>(layout.consumedBytes));
177 const std::vector<std::uint8_t> trailing(
178 inputBytes.begin() + static_cast<std::ptrdiff_t>(layout.consumedBytes),
179 inputBytes.end());
180
181 customConsole(appName,
182 "decoding " + std::to_string(layout.packets.size())
183 + " packet(s) using "
184 + packetErrorControlModeName(manager.getTemplate().getPacketErrorControlMode())
185 + " packet error control");
186 if (const auto result = manager.load(packetPrefix); !result) {
187 return printError(result.error());
188 }
189
190 if (!trailing.empty()) {
191 customConsole(appName,
192 "leaving " + std::to_string(trailing.size())
193 + " trailing byte(s) unconsumed", "WARN");
194 }
195 if (const auto it = args.find("trailing-output"); it != args.end()) {
196 if (const auto result = writeBinaryFile(trailing, it->second); !result) {
197 return printError(result.error());
198 }
199 }
200
201 if (args["verbose"] == "true") printPackets(manager);
202
203 manager.setAutoValidateEnable(settings.validationEnable);
204 std::vector<std::uint8_t> outputData;
205 {
206 const auto result = manager.getApplicationDataBuffer();
207 if (!result) return printError(result.error());
208 outputData = result.value();
209 }
210
211 customConsole(appName, "writing data to " + output);
212 if (const auto result = writeBinaryFile(outputData, output); !result) {
213 return printError(result.error());
214 }
215
216 const auto duration = std::chrono::duration_cast<std::chrono::microseconds>(
217 std::chrono::high_resolution_clock::now() - start);
218 customConsole(appName, "execution time: " + std::to_string(duration.count()) + " [us]");
219 customConsole(appName, "[ Exit code 0 ]");
220 return 0;
221}
