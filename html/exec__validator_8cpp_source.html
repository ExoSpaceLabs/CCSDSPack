56ccsds::Result<ValidatorSettings>
57loadValidatorSettings(const std::unordered_map<std::string, std::string> &args) {
58 ValidatorSettings settings;
59
60 const auto configIt = args.find("config");
61 if (configIt != args.end()) {
62 if (!ccsds::fileExists(configIt->second)) {
63 return ccsds::Error{static_cast<ccsds::ErrorCode>(ARG_PARSE_ERROR),
64 "Configuration file does not exist: " + configIt->second};
65 }
66
67 ccsds::Config cfg;
68 const auto loadResult = cfg.load(configIt->second);
69 if (!loadResult) return loadResult.error();
70
71 const auto templateResult = settings.templatePacket.loadFromConfig(cfg);
72 if (!templateResult) return templateResult.error();
73
74 settings.mode = settings.templatePacket.getPacketErrorControlMode();
75 settings.hasTemplate = true;
76
77 if (cfg.isKey("sync_pattern_enable")) {
78 const auto syncEnableResult = cfg.get<bool>("sync_pattern_enable");
79 if (!syncEnableResult) return syncEnableResult.error();
80 settings.syncPatternEnable = syncEnableResult.value();
81 }
82
83 if (settings.syncPatternEnable && cfg.isKey("sync_pattern")) {
84 const auto syncResult = cfg.get<int>("sync_pattern");
85 if (!syncResult) return syncResult.error();
86 settings.syncPattern = static_cast<std::uint32_t>(syncResult.value());
87 }
88 }
89
90 const auto modeIt = args.find("packet-error-control");
91 if (modeIt != args.end()) {
92 const auto modeResult = parsePacketErrorControlMode(modeIt->second);
93 if (!modeResult) return modeResult.error();
94 settings.mode = modeResult.value();
95 if (settings.hasTemplate) {
96 const auto applied = applyPacketErrorControlMode(settings.templatePacket, settings.mode);
97 if (!applied) return applied.error();
98 }
99 }
100
101 return settings;
102}

104void emitValidationReport(const ccsds::ValidationReport &validation,
105 const bool verbose,
106 std::ostringstream &report) {
107 for (const auto &check : validation) {
108 if (!verbose && check.passed) continue;
109 std::ostringstream line;
110 line << " [REPORT] " << std::left << std::setw(37)
111 << ccsds::validationCodeName(check.code)
112 << " : " << (check.passed ? "PASSED" : "FAILED") << '\n';
113 report << line.str();
114 std::cout << line.str();
115 }
116}

118void emitParseFailure(const ccsds::Error &error, const bool pusHeaderExpected) {
119 const auto &message = error.message();
120 if (pusHeaderExpected) {
121 std::cout << " [REPORT] PUS secondary header : FAILED\n";
122 } else if (message.find("packet version") != std::string::npos) {
123 std::cout << " [REPORT] CCSDS version : FAILED\n";
124 } else {
125 std::cout << " [REPORT] Packet parse : FAILED\n";
126 }
127 std::cerr << " " << message << std::endl;
128}

130ccsds::Packet parserPacket(const ValidatorSettings &settings) {
131 ccsds::Packet packet = settings.hasTemplate ? settings.templatePacket : ccsds::Packet{};
132 packet.setPacketErrorControlMode(settings.mode);
133 // data_field_size is a generation/template capacity, not a wire-level CCSDS
134 // invariant. Keep the template's identifier/PEC/header schema but make the
135 // validator parser permissive enough to classify an oversized/mismatched packet.
136 packet.setDataFieldSize(std::numeric_limits<std::uint16_t>::max());
137 return packet;
138}

140bool expectsPusHeader(const ccsds::Packet &packet) {
141 const auto header = packet.getSecondaryHeader();
142 return header && header->isPusHeader();

147int main(const int argc, char *argv[]) {
148 const std::string appName = "ccsds_validator";
149 std::unordered_map<std::string, std::string> allowed{
150 {"h", "help"}, {"v", "verbose"}, {"i", "input"},
151 {"c", "config"}, {"p", "print-packets"},
152 {"e", "packet-error-control"}
153 };
154 const std::set<std::string> booleanArgs{"verbose", "help", "print-packets"};
155 std::unordered_map<std::string, std::string> args{
156 {"verbose", "false"}, {"help", "false"}, {"print-packets", "false"}
157 };
158
159 const auto start = std::chrono::high_resolution_clock::now();
160 const auto argumentResult = parseArguments(argc, argv, allowed, args, booleanArgs);
161 if (!argumentResult) return printError(argumentResult.error());
162
163 if (args["help"] == "true") {
164 printHelp();
165 return 0;
166 }
167
168 const auto inputIt = args.find("input");
169 if (inputIt == args.end() || inputIt->second.empty()) {
170 printHelp();
171 return printError(ccsds::Error{static_cast<ccsds::ErrorCode>(ARG_PARSE_ERROR),
172 "Input file must be specified"});
173 }
174 if (!ccsds::fileExists(inputIt->second)) {
175 return printError(ccsds::Error{static_cast<ccsds::ErrorCode>(ARG_PARSE_ERROR),
176 "Input file does not exist: " + inputIt->second});
177 }
178
179 const auto settingsResult = loadValidatorSettings(args);
180 if (!settingsResult) return printError(settingsResult.error());
181 const ValidatorSettings settings = settingsResult.value();
182
183 const auto inputResult = ccsds::readBinaryFile(inputIt->second);
184 if (!inputResult) return printError(inputResult.error());
185 const std::vector<std::uint8_t> inputBytes = inputResult.value();
186
187 const auto layoutResult = inspectPacketStream(inputBytes,
188 settings.syncPatternEnable,
189 settings.syncPattern,
190 true);
191 if (!layoutResult) {
192 std::cout << " [REPORT] Packet Data Length : FAILED\n";
193 std::cerr << " " << layoutResult.error().message() << std::endl;
194 customConsole(appName, "Packets validation [FAILED]");
195 return PACKET_VALIDATION_FAILED;
196 }
197 const PacketStreamLayout layout = layoutResult.value();
198
199 if (layout.packets.empty()) {
200 std::cout << " [REPORT] Packet Data Length : FAILED\n";
201 std::cerr << " Input does not contain a complete CCSDS packet" << std::endl;
202 customConsole(appName, "Packets validation [FAILED]");
203 return PACKET_VALIDATION_FAILED;
204 }
205
206 const bool verbose = args["verbose"] == "true";
207 const bool printPacketsEnabled = args["print-packets"] == "true";
208 bool overallResult{true};
209 std::vector<std::size_t> failedPackets;
210
211 ccsds::Validator validator;
212 validator.configure(true, true, settings.hasTemplate);
213 if (settings.hasTemplate) validator.setTemplatePacket(settings.templatePacket);
214
215 for (std::size_t index = 0U; index < layout.packets.size(); ++index) {
216 const auto &slice = layout.packets[index];
217 const std::vector<std::uint8_t> packetBytes(
218 inputBytes.begin() + static_cast<std::ptrdiff_t>(slice.offset),
219 inputBytes.begin() + static_cast<std::ptrdiff_t>(slice.offset + slice.size));
220
221 std::ostringstream packetReport;
222 packetReport << "[ CCSDS VALIDATOR ] Packet " << index + 1U << '\n';
223 if (verbose) std::cout << packetReport.str();
224
225 ccsds::Packet packet = parserPacket(settings);
226 const bool pusExpected = expectsPusHeader(packet);
227 const auto parsed = packet.deserializeBounded(packetBytes);
228 if (!parsed) {
229 overallResult = false;
230 failedPackets.push_back(index + 1U);
231 emitParseFailure(parsed.error(), pusExpected);
232 continue;
233 }
234
235 const auto validation = validator.validate(packet);
236 emitValidationReport(validation, verbose, packetReport);
237
238 if (!validation.valid()) {
239 overallResult = false;
240 failedPackets.push_back(index + 1U);
241 }
242
243 if (printPacketsEnabled) ccsds::printPacket(packet);
244 }
245
246 const std::size_t trailingBytes = inputBytes.size() - layout.consumedBytes;
247 if (trailingBytes != 0U) {
248 overallResult = false;
249 std::cout << " [REPORT] Packet Data Length : FAILED\n";
250 std::cerr << " " << trailingBytes
251 << " trailing byte(s) do not form a complete packet" << std::endl;
252 }
253
254 if (!failedPackets.empty()) {
255 std::cerr << "[ CCSDS VALIDATOR ] Failed packet(s):";
256 for (const auto packetIndex : failedPackets) std::cerr << ' ' << packetIndex;
257 std::cerr << std::endl;
258 }
259
260 customConsole(appName,
261 std::string("packet error control: ")
262 + packetErrorControlModeName(settings.mode));
263 customConsole(appName,
264 std::string("Packets validation [")
265 + (overallResult ? "PASSED" : "FAILED") + "]");
266
267 const auto duration = std::chrono::duration_cast<std::chrono::microseconds>(
268 std::chrono::high_resolution_clock::now() - start);
269 customConsole(appName, "execution time: " + std::to_string(duration.count()) + " [us]");
270
271 if (!overallResult) {
272 customConsole(appName, "[ Exit code 18 ]");
273 return PACKET_VALIDATION_FAILED;
274 }
275
276 customConsole(appName, "[ Exit code 0 ]");
277 return 0;
278}
