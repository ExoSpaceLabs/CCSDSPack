7static ccsds_status_t validate_fields(const ccsds_pus_tc_fields_t *fields,
8 const uint8_t source_id_octets) {
9 if (fields == NULL) {
10 return CCSDS_STATUS_NULL_POINTER;
11 }
12 if (fields->acknowledgement_flags > 0x0FU) {
13 return CCSDS_STATUS_PUS_INVALID_ACK_FLAGS;
14 }
15 if (!ccsds_pus_identifier_fits(fields->source_id, source_id_octets)) {
16 return CCSDS_STATUS_PUS_IDENTIFIER_OVERFLOW;
17 }
18 return CCSDS_STATUS_OK;
19}

21static void write_spare(uint8_t *output, const size_t offset,
22 const uint8_t spare_octets) {
23 uint8_t index;
24 for (index = 0U; index < spare_octets; ++index) {
25 output[offset + (size_t)index] = 0U;
26 }
27}

40size_t ccsds_pus_a_tc_encoded_size(const ccsds_pus_a_tc_tailoring_t *tailoring) {
41 if (ccsds_pus_a_tc_validate_tailoring(tailoring) != CCSDS_STATUS_OK) {

56ccsds_status_t ccsds_pus_a_tc_encode(
57 const ccsds_pus_tc_fields_t *fields,
58 const ccsds_pus_a_tc_tailoring_t *tailoring,
59 uint8_t *output,
60 const size_t capacity,
61 size_t *written_out) {
62 size_t offset = 0U;
63 size_t required;
64 ccsds_status_t status;
65
66 if (tailoring == NULL || written_out == NULL) {
67 return CCSDS_STATUS_NULL_POINTER;
68 }
69 status = ccsds_pus_a_tc_validate_tailoring(tailoring);
70 if (status != CCSDS_STATUS_OK) return status;
71 status = validate_fields(fields, tailoring->source_id_octets);
72 if (status != CCSDS_STATUS_OK) return status;
73
74 required = ccsds_pus_a_tc_encoded_size(tailoring);
75 if (capacity < required) return CCSDS_STATUS_BUFFER_TOO_SMALL;
76 if (output == NULL) return CCSDS_STATUS_NULL_POINTER;
77
78 output[offset++] = (uint8_t)(0x10U | fields->acknowledgement_flags);
79 output[offset++] = fields->service_type;
80 output[offset++] = fields->service_subtype;
81 ccsds_store_uint(output + offset, tailoring->source_id_octets,
82 fields->source_id, CCSDS_BYTE_ORDER_BIG);
83 offset += (size_t)tailoring->source_id_octets;
84 write_spare(output, offset, tailoring->secondary_header_spare_octets);
85 offset += (size_t)tailoring->secondary_header_spare_octets;
86
87 *written_out = offset;
88 return CCSDS_STATUS_OK;
89}

91ccsds_status_t ccsds_pus_a_tc_decode(
92 const uint8_t *data,
93 const size_t size,
94 const ccsds_pus_a_tc_tailoring_t *tailoring,
95 ccsds_pus_tc_fields_t *fields_out) {
96 ccsds_status_t status;
97 size_t expected;
98
99 if (tailoring == NULL || fields_out == NULL) {
100 return CCSDS_STATUS_NULL_POINTER;
101 }
102 status = ccsds_pus_a_tc_validate_tailoring(tailoring);
103 if (status != CCSDS_STATUS_OK) return status;
104
105 expected = ccsds_pus_a_tc_encoded_size(tailoring);
106 if (size != expected) return CCSDS_STATUS_PUS_SIZE_MISMATCH;
107 if (data == NULL) return CCSDS_STATUS_NULL_POINTER;
108 if ((data[0] & 0xF0U) != 0x10U) return CCSDS_STATUS_PUS_INVALID_VERSION;
109 if (!ccsds_pus_spare_is_zero(data, size, tailoring->secondary_header_spare_octets)) {
110 return CCSDS_STATUS_PUS_NONZERO_SPARE;
111 }
112
113 fields_out->acknowledgement_flags = (uint8_t)(data[0] & 0x0FU);
114 fields_out->service_type = data[1];
115 fields_out->service_subtype = data[2];
116 fields_out->source_id = (uint32_t)ccsds_load_uint(
117 data + 3U, tailoring->source_id_octets, CCSDS_BYTE_ORDER_BIG);
118 return CCSDS_STATUS_OK;
119}
