8ccsds::ResultBuffer ccsds::DataField::serialize() {
9 update();
10 const auto secondaryHeader = static_cast<const DataField &>(*this).getSecondaryHeaderBytes();
11 RET_IF_ERR_MSG(m_secondaryHeader && secondaryHeader.size() != m_secondaryHeader->getSize(),
12 ErrorCode::INVALID_SECONDARY_HEADER_DATA,
13 "Secondary header serialization failed or returned an unexpected size.");
14 RET_IF_ERR_MSG(secondaryHeader.size() + m_applicationData.size() > m_dataPacketSize,
15 ErrorCode::INVALID_DATA,
16 "Serialized packet data field exceeds its configured capacity.");
17 std::vector<std::uint8_t> fullData{};
18 fullData.reserve(secondaryHeader.size() + m_applicationData.size());
19 fullData.insert(fullData.end(), secondaryHeader.begin(), secondaryHeader.end());
20 fullData.insert(fullData.end(), m_applicationData.begin(), m_applicationData.end());
21 return fullData;
22}

71ccsds::ResultBool ccsds::DataField::setApplicationData(const std::uint8_t *pData,
72 const std::size_t &sizeData) {
73 RET_IF_ERR_MSG(!pData, ErrorCode::NULL_POINTER, "Application data is nullptr");
74 RET_IF_ERR_MSG(sizeData < 1U, ErrorCode::INVALID_APPLICATION_DATA,
75 "Application data size cannot be less than one");
76 const auto secondaryHeaderSize = m_secondaryHeader
77 ? static_cast<std::size_t>(m_secondaryHeader->getSize()) : 0U;
78 RET_IF_ERR_MSG(sizeData + secondaryHeaderSize > m_dataPacketSize,
79 ErrorCode::INVALID_APPLICATION_DATA,
80 "Application data field exceeds available size");
81 m_applicationData.assign(pData, pData + sizeData);
82 m_secondaryHeaderUpdated = false;
83 return true;
84}

118ccsds::ResultBool ccsds::DataField::setSecondaryHeader(
119 const std::vector<std::uint8_t> &data, const std::string &pType) {
120 RET_IF_ERR_MSG(data.size() + m_applicationData.size() > m_dataPacketSize,
121 ErrorCode::INVALID_SECONDARY_HEADER_DATA,
122 "Secondary header data exceeds available size");
123 std::shared_ptr<SecondaryHeaderAbstract> header;
124 if (pus::SecondaryHeaderFactory::isPusSelector(pType)) {
125 auto result = m_pusSecondaryHeaderFactory.create(pType);
126 if (!result) return result.error();
127 header = result.value();
128 } else {
129 RET_IF_ERR_MSG(!m_secondaryHeaderFactory.typeIsRegistered(pType),
130 ErrorCode::INVALID_SECONDARY_HEADER_DATA,
131 "Secondary header type is not registered: " + pType);
132 header = m_secondaryHeaderFactory.create(pType);
133 }
134 RET_IF_ERR_MSG(!header, ErrorCode::INVALID_SECONDARY_HEADER_DATA,
135 "Failed to create secondary header of type: " + pType);
136 if (!header->variableLength) {
137 RET_IF_ERR_MSG(data.size() != header->getSize(), ErrorCode::INVALID_SECONDARY_HEADER_DATA,
138 "Secondary header data size mismatch for type: " + pType);
139 }
140 FORWARD_RESULT(header->deserialize(data));
141 m_secondaryHeader = std::move(header);
142 m_secondaryHeaderType = pType;
143 m_secondaryHeaderUpdated = false;
144 return true;
145}

161ccsds::ResultBool ccsds::DataField::setSecondaryHeader(const ccsds::Config &cfg) {
162 RET_IF_ERR_MSG(!cfg.isKey("secondary_header_type"), ErrorCode::CONFIG_FILE_ERROR,
163 "Config: Missing string field: secondary_header_type");
164 std::string type{};
165 ASSIGN_CP(type, cfg.get<std::string>("secondary_header_type"));
166 RET_IF_ERR_MSG(type == "PusA" || type == "PusB" || type == "PusC",
167 ErrorCode::CONFIG_FILE_ERROR,
168 "Legacy PusA/PusB/PusC selectors were removed in v2; use a canonical PUS selector.");
169 std::shared_ptr<SecondaryHeaderAbstract> header;
170 if (pus::SecondaryHeaderFactory::isPusSelector(type)) {
171 auto result = m_pusSecondaryHeaderFactory.create(type);
172 if (!result) return result.error();
173 header = result.value();
174 } else {
175 RET_IF_ERR_MSG(!m_secondaryHeaderFactory.typeIsRegistered(type),
176 ErrorCode::INVALID_SECONDARY_HEADER_DATA,
177 "Secondary header type is not registered: " + type);
178 header = m_secondaryHeaderFactory.create(type);
179 }
180 RET_IF_ERR_MSG(!header, ErrorCode::INVALID_SECONDARY_HEADER_DATA,
181 "Failed to create secondary header of type: " + type);
182 FORWARD_RESULT(header->loadFromConfig(cfg));
183 m_secondaryHeader = std::move(header);
184 m_secondaryHeaderType = m_secondaryHeader->getType();
185 m_secondaryHeaderUpdated = false;
186 return true;
187}

190ccsds::ResultBool ccsds::DataField::setSecondaryHeader(
191 std::shared_ptr<SecondaryHeaderAbstract> header) {
192 if (header) {
193 RET_IF_ERR_MSG(static_cast<std::size_t>(header->getSize()) + m_applicationData.size()
194 > m_dataPacketSize,
195 ErrorCode::INVALID_SECONDARY_HEADER_DATA,
196 "Secondary header exceeds available data-field capacity.");
197 RET_IF_ERR_MSG(!header->isPusHeader()
198 && pus::SecondaryHeaderFactory::isPusSelector(header->getType()),
199 ErrorCode::INVALID_SECONDARY_HEADER_DATA,
200 "Custom secondary headers cannot use the reserved PUS: namespace.");
201 }
202 m_secondaryHeader = std::move(header);
203 m_secondaryHeaderType = m_secondaryHeader ? m_secondaryHeader->getType() : std::string{};
204 m_secondaryHeaderUpdated = false;
205 return true;
