18 const ccsds_primary_header_t *header,

20 uint8_t *output,

24 ccsds_sequence_validator_t staged_sequence;
25 size_t available;
26 int complete;
27 ccsds_status_t status;
28
29 if (state == NULL || header == NULL
30 || written_now_out == NULL || complete_out == NULL) {
31 return CCSDS_STATUS_NULL_POINTER;
32 }
33 if (application_data.size != 0U && application_data.data == NULL) {
34 return CCSDS_STATUS_NULL_POINTER;
35 }
36 if (ccsds_primary_header_validate(header) != CCSDS_STATUS_OK) {
37 return CCSDS_STATUS_INVALID_HEADER_DATA;
38 }
39 if (state->written > capacity) {
40 return CCSDS_STATUS_INVALID_DATA;
41 }
42 available = capacity - state->written;
43 if (application_data.size > available) {
44 return CCSDS_STATUS_BUFFER_TOO_SMALL;
45 }
46 if (application_data.size != 0U && output == NULL) {
47 return CCSDS_STATUS_NULL_POINTER;
48 }
49
50 staged_sequence = state->sequence;
51 if (state->validate_sequence != 0U) {
52 if (!ccsds_sequence_flags_valid(&staged_sequence, header->sequence_flags)
53 || !ccsds_sequence_count_valid(&staged_sequence, header->sequence_count)) {
54 return CCSDS_STATUS_VALIDATION_FAILURE;
55 }
56 status = ccsds_sequence_validator_accept(&staged_sequence, header);
57 if (status != CCSDS_STATUS_OK) {
58 return status;
59 }
60 }
61
62 complete = header->sequence_flags == 3U || header->sequence_flags == 2U;
63
64 if (application_data.size != 0U) {
65 memcpy(output + state->written,
66 application_data.data,
67 application_data.size);
68 }
69
70 state->sequence = staged_sequence;
71 state->written += application_data.size;
72 *written_now_out = application_data.size;
73 *complete_out = complete;
74 return CCSDS_STATUS_OK;
75}
