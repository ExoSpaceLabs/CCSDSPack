18ccsds::ResultBool ccsds::Config::load(const std::string &filename) {
19 std::ifstream file(filename);
20 RET_IF_ERR_MSG(!file.is_open(),ccsds::ErrorCode::CONFIG_FILE_ERROR, "Failed to open configuration file");
21
22 std::unordered_map<std::string, ConfigValue> parsedValues;
23 std::string line;
24 while (std::getline(file, line)) {
25 if (line.empty() || line.front() == '#') continue;
26
27 auto [key, type, valueStr] = parseLine(line);
28 RET_IF_ERR_MSG(key.empty() || type.empty(),ccsds::ErrorCode::CONFIG_FILE_ERROR, "Failed to parse configuration file");
29 ConfigValue value;
30 if (type == "string") {
31 value = valueStr;
32 } else if (type == "int" || type == "uint") {
33 const bool negative = !valueStr.empty() && valueStr.front() == '-';
34 std::string_view encoded{valueStr};
35 std::uint8_t base = 10U;
36 if (negative) encoded.remove_prefix(1U);
37 if (encoded.size() > 2U && encoded[0] == '0'
38 && (encoded[1] == 'x' || encoded[1] == 'X')) {
39 encoded.remove_prefix(2U);
40 base = 16U;
41 }
42 RET_IF_ERR_MSG(encoded.empty(), ccsds::ErrorCode::CONFIG_FILE_ERROR,
43 "Config: Invalid integer value for key: " + key);
44
45 std::uint64_t parsed{};
46 const auto conversion = std::from_chars(encoded.data(),
47 encoded.data() + encoded.size(),
48 parsed, base);
49 RET_IF_ERR_MSG(conversion.ec != std::errc{}
50 || conversion.ptr != encoded.data() + encoded.size(),
51 ccsds::ErrorCode::CONFIG_FILE_ERROR,
52 "Config: Invalid integer value for key: " + key);
53 if (type == "uint") {
54 RET_IF_ERR_MSG(negative, ccsds::ErrorCode::CONFIG_FILE_ERROR,
55 "Config: Unsigned value cannot be negative for key: " + key);
56 value = parsed;
57 } else {
58 constexpr auto positiveLimit = static_cast<std::uint64_t>(INT32_MAX);
59 constexpr auto negativeLimit = positiveLimit + 1U;
60 RET_IF_ERR_MSG((!negative && parsed > positiveLimit)
61 || (negative && parsed > negativeLimit),
62 ccsds::ErrorCode::CONFIG_FILE_ERROR,
63 "Config: Integer value is out of range for key: " + key);
64 value = negative ? static_cast<int>(-static_cast<std::int64_t>(parsed))
65 : static_cast<int>(parsed);
66 }
67 } else if (type == "float") {
68 errno = 0;
69 char *end{};
70 const float parsed = std::strtof(valueStr.c_str(), &end);
71 RET_IF_ERR_MSG(errno == ERANGE || end == valueStr.c_str() || *end != '\0',
72 ccsds::ErrorCode::CONFIG_FILE_ERROR,
73 "Config: Invalid float value for key: " + key);
74 value = parsed;
75 } else if (type == "bool") {
76 RET_IF_ERR_MSG(valueStr != "true" && valueStr != "false"
77 && valueStr != "1" && valueStr != "0",
78 ccsds::ErrorCode::CONFIG_FILE_ERROR,
79 "Config: Invalid bool value for key: " + key);
80 value = valueStr == "true" || valueStr == "1";
81 } else if (type == "bytes") {
82 ASSIGN_CP( value, parseBytes(valueStr) );
83 } else {
84 return ccsds::Error{ccsds::ErrorCode::CONFIG_FILE_ERROR, " unknown type: " + type};
85 }
86
87 parsedValues[key] = value;
88 }
89
90 values = std::move(parsedValues);
91 return true;
92}

101std::tuple<std::string, std::string, std::string> ccsds::Config::parseLine(
102 const std::string& line) {
103 auto colonPos = line.find(':');
104 auto equalPos = line.find('=');
105
106 if (colonPos == std::string::npos || equalPos == std::string::npos || equalPos < colonPos)
107 return {"", "", ""};
108
109 std::string key = line.substr(0, colonPos);
110 std::string type = line.substr(colonPos + 1, equalPos - colonPos - 1);
111 std::string value = line.substr(equalPos + 1);
112
113 if (!value.empty() && value.front() == '"' && value.back() == '"')
114 value = value.substr(1, value.size() - 2);
115
116 return {key, type, value};
117}

119ccsds::ResultBuffer ccsds::Config::parseBytes(const std::string &valueStr) {
120 std::vector<uint8_t> result{};
121 RET_IF_ERR_MSG(valueStr.empty() || valueStr.front() != '[' || valueStr.back() != ']',
122 ccsds::ErrorCode::CONFIG_FILE_ERROR,
123 "Config: Invalid buffer formatting []");
124
125 if (valueStr == "[]" || valueStr == "[ ]") {
126 return result;
127 }
128
129 // Strip surrounding [ ... ]
130 std::string inner = valueStr.substr(1, valueStr.size() - 2);
131 std::stringstream ss(inner);
132 std::string token;
133
134 while (std::getline(ss, token, ',')) {
135 // Remove all spaces inside each token
136 token.erase(std::remove_if(token.begin(), token.end(),
137 [](unsigned char c){ return std::isspace(c); }),
138 token.end());
139
140 // Empty token after trimming is invalid (e.g., "[12, ,34]")
141 if (token.empty()) {
142 return ccsds::Error{ccsds::ErrorCode::CONFIG_FILE_ERROR, "Invalid byte value: <empty>"};
143 }
144
145 std::uint8_t base = 10;
146 std::string_view sv{token};
147
148 // Allow 0x / 0X prefix for hex
149 if (sv.size() > 2 && sv[0] == '0' && (sv[1] == 'x' || sv[1] == 'X')) {
150 sv.remove_prefix(2);
151 base = 16;
152 if (sv.empty()) {
153 return ccsds::Error{ccsds::ErrorCode::CONFIG_FILE_ERROR, "Invalid byte value: 0x"};
154 }
155 }
156
157 // Parse without exceptions
158 unsigned int tmp = 0U;
159 const char* first = sv.data();
160 const char* last = sv.data() + sv.size();
161 auto res = std::from_chars(first, last, tmp, base);
162
163 // Valid if parsed ok, consumed all characters, and fits in a byte
164 if (res.ec != std::errc{} || res.ptr != last || tmp > 0xFFu) {
165 return ccsds::Error{ccsds::ErrorCode::CONFIG_FILE_ERROR,
166 std::string("Invalid byte value: ") + std::string(token)};
167 }
168 result.push_back(static_cast<uint8_t>(tmp));
169 }
170
171 return result;
172}
