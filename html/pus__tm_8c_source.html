7static int cuc_is_empty(const ccsds_cuc_config_t *config) {
8 return config->epoch == CCSDS_CUC_EPOCH_UNSPECIFIED
9 && config->pfield_mode == CCSDS_CUC_PFIELD_IMPLICIT
10 && config->coarse_octets == 0U
11 && config->fine_octets == 0U;
12}

14static ccsds_status_t validate_time_tailoring(const uint8_t timestamp_present,
15 const ccsds_cuc_config_t *config) {
16 if (timestamp_present > 1U) {
17 return CCSDS_STATUS_PUS_INVALID_BOOLEAN;
18 }
19 if (timestamp_present != 0U) {
20 return ccsds_cuc_validate(config);
21 }
22 if (!cuc_is_empty(config)) {
23 return CCSDS_STATUS_PUS_DISABLED_TIME_CONFIG;
24 }
25 return CCSDS_STATUS_OK;
26}

28static void write_spare(uint8_t *output, const size_t offset,
29 const uint8_t spare_octets) {
30 uint8_t index;
31 for (index = 0U; index < spare_octets; ++index) {
32 output[offset + (size_t)index] = 0U;
33 }
34}

36ccsds_status_t ccsds_pus_a_tm_validate_tailoring(
37 const ccsds_pus_a_tm_tailoring_t *tailoring) {
38 ccsds_status_t status;
39
40 if (tailoring == NULL) return CCSDS_STATUS_NULL_POINTER;
41 if (!ccsds_pus_identifier_width_is_valid(tailoring->destination_id_octets)) {
42 return CCSDS_STATUS_PUS_INVALID_IDENTIFIER_WIDTH;
43 }
44 if (tailoring->packet_subcounter_present > 1U) {
45 return CCSDS_STATUS_PUS_INVALID_BOOLEAN;
46 }
47 status = validate_time_tailoring(tailoring->timestamp_present, &tailoring->cuc);
48 return status;
49}

57size_t ccsds_pus_a_tm_encoded_size(const ccsds_pus_a_tm_tailoring_t *tailoring) {
58 if (ccsds_pus_a_tm_validate_tailoring(tailoring) != CCSDS_STATUS_OK) {
59 return 0U;
60 }
61 return (size_t)3U
62 + (tailoring->packet_subcounter_present != 0U ? 1U : 0U)
63 + (size_t)tailoring->destination_id_octets
64 + (tailoring->timestamp_present != 0U
65 ? ccsds_cuc_encoded_size(&tailoring->cuc) : 0U)
66 + (size_t)tailoring->secondary_header_spare_octets;
67}

79ccsds_status_t ccsds_pus_a_tm_encode(
80 const ccsds_pus_a_tm_fields_t *fields,
81 const ccsds_pus_a_tm_tailoring_t *tailoring,
82 uint8_t *output,
83 const size_t capacity,
84 size_t *written_out) {
85 size_t offset = 0U;
86 size_t required;
87 size_t cuc_written = 0U;
88 ccsds_status_t status;
89
90 if (fields == NULL || tailoring == NULL || written_out == NULL) {
91 return CCSDS_STATUS_NULL_POINTER;
92 }
93
94 status = ccsds_pus_a_tm_validate_tailoring(tailoring);
95 if (status != CCSDS_STATUS_OK) return status;
96 if (!ccsds_pus_identifier_fits(fields->destination_id,
97 tailoring->destination_id_octets)) {
98 return CCSDS_STATUS_PUS_IDENTIFIER_OVERFLOW;
99 }
100
101 required = ccsds_pus_a_tm_encoded_size(tailoring);
102 if (capacity < required) return CCSDS_STATUS_BUFFER_TOO_SMALL;
103 if (output == NULL) return CCSDS_STATUS_NULL_POINTER;
104
105 output[offset++] = 0x10U;
106 output[offset++] = fields->service_type;
107 output[offset++] = fields->service_subtype;
108
109 if (tailoring->packet_subcounter_present != 0U) {
110 output[offset++] = fields->packet_subcounter;
111 }
112
113 ccsds_store_uint(output + offset, tailoring->destination_id_octets,
114 fields->destination_id, CCSDS_BYTE_ORDER_BIG);
115 offset += (size_t)tailoring->destination_id_octets;
116
117 if (tailoring->timestamp_present != 0U) {
118 status = ccsds_cuc_encode(
119 &fields->timestamp,
120 &tailoring->cuc,
121 output + offset,
122 capacity - offset,
123 &cuc_written);
124 if (status != CCSDS_STATUS_OK) return status;
125 offset += cuc_written;
126 }
127
128 write_spare(output, offset, tailoring->secondary_header_spare_octets);
129 offset += (size_t)tailoring->secondary_header_spare_octets;
130
131 *written_out = offset;
132 return CCSDS_STATUS_OK;
133}

135ccsds_status_t ccsds_pus_a_tm_decode(
136 const uint8_t *data,
137 const size_t size,
138 const ccsds_pus_a_tm_tailoring_t *tailoring,
139 ccsds_pus_a_tm_fields_t *fields_out) {
140 size_t offset = 0U;
141 size_t timestamp_size = 0U;
142 size_t expected;
143 ccsds_status_t status;
144
145 if (tailoring == NULL || fields_out == NULL) {
146 return CCSDS_STATUS_NULL_POINTER;
147 }
148
149 status = ccsds_pus_a_tm_validate_tailoring(tailoring);
150 if (status != CCSDS_STATUS_OK) return status;
151 expected = ccsds_pus_a_tm_encoded_size(tailoring);
152 if (size != expected) return CCSDS_STATUS_PUS_SIZE_MISMATCH;
153 if (data == NULL) return CCSDS_STATUS_NULL_POINTER;
154 if (data[0] != 0x10U) return CCSDS_STATUS_PUS_INVALID_VERSION;
155 if (!ccsds_pus_spare_is_zero(
156 data, size, tailoring->secondary_header_spare_octets)) {
157 return CCSDS_STATUS_PUS_NONZERO_SPARE;
158 }
159
160 fields_out->service_type = data[1];
161 fields_out->service_subtype = data[2];
162 offset = 3U;
163
164 if (tailoring->packet_subcounter_present != 0U) {
165 fields_out->packet_subcounter = data[offset++];
166 } else {
167 fields_out->packet_subcounter = 0U;
168 }
169
170 fields_out->destination_id = (uint32_t)ccsds_load_uint(
171 data + offset, tailoring->destination_id_octets, CCSDS_BYTE_ORDER_BIG);
172 offset += (size_t)tailoring->destination_id_octets;
173
174 if (tailoring->timestamp_present != 0U) {
175 timestamp_size = ccsds_cuc_encoded_size(&tailoring->cuc);
176 status = ccsds_cuc_decode(
177 data + offset, timestamp_size, &tailoring->cuc, &fields_out->timestamp);
178 if (status != CCSDS_STATUS_OK) return status;
179 } else {
180 fields_out->timestamp.coarse = UINT64_C(0);
181 fields_out->timestamp.fine = UINT64_C(0);
182 }
183
184 return CCSDS_STATUS_OK;
185}

187ccsds_status_t ccsds_pus_c_tm_encode(
188 const ccsds_pus_c_tm_fields_t *fields,
189 const ccsds_pus_c_tm_tailoring_t *tailoring,
190 uint8_t *output,
191 const size_t capacity,
192 size_t *written_out) {
193 size_t offset = 0U;
194 size_t required;
195 size_t cuc_written = 0U;
196 ccsds_status_t status;
197
198 if (fields == NULL || tailoring == NULL || written_out == NULL) {
199 return CCSDS_STATUS_NULL_POINTER;
200 }
201
202 status = ccsds_pus_c_tm_validate_tailoring(tailoring);
203 if (status != CCSDS_STATUS_OK) return status;
204 if (fields->time_reference_status > 0x0FU) {
205 return CCSDS_STATUS_PUS_INVALID_TIME_REFERENCE_STATUS;
206 }
207 if (!ccsds_pus_identifier_fits(fields->destination_id, 2U)) {
208 return CCSDS_STATUS_PUS_IDENTIFIER_OVERFLOW;
209 }
210
211 required = ccsds_pus_c_tm_encoded_size(tailoring);
212 if (capacity < required) return CCSDS_STATUS_BUFFER_TOO_SMALL;
213 if (output == NULL) return CCSDS_STATUS_NULL_POINTER;
214
215 output[offset++] = (uint8_t)(0x20U | fields->time_reference_status);
216 output[offset++] = fields->service_type;
217 output[offset++] = fields->service_subtype;
218 ccsds_store_be16(output + offset, fields->message_type_counter);
219 offset += 2U;
220 ccsds_store_be16(output + offset, (uint16_t)fields->destination_id);
221 offset += 2U;
222
223 if (tailoring->timestamp_present != 0U) {
224 status = ccsds_cuc_encode(
225 &fields->timestamp,
226 &tailoring->cuc,
227 output + offset,
228 capacity - offset,
229 &cuc_written);
230 if (status != CCSDS_STATUS_OK) return status;
231 offset += cuc_written;
232 }
233
234 write_spare(output, offset, tailoring->secondary_header_spare_octets);
235 offset += (size_t)tailoring->secondary_header_spare_octets;
236
237 *written_out = offset;
238 return CCSDS_STATUS_OK;
239}

262 return CCSDS_STATUS_PUS_NONZERO_SPARE;

266 fields_out->service_type = data[1];
267 fields_out->service_subtype = data[2];
268 fields_out->message_type_counter = ccsds_load_be16(data + 3U);
269 fields_out->destination_id = (uint32_t)ccsds_load_be16(data + 5U);
270
271 if (tailoring->timestamp_present != 0U) {
272 timestamp_size = ccsds_cuc_encoded_size(&tailoring->cuc);
273 status = ccsds_cuc_decode(
274 data + 7U, timestamp_size, &tailoring->cuc, &fields_out->timestamp);
275 if (status != CCSDS_STATUS_OK) return status;
276 } else {
277 fields_out->timestamp.coarse = UINT64_C(0);
278 fields_out->timestamp.fine = UINT64_C(0);
279 }
280
281 return CCSDS_STATUS_OK;
