7static uint64_t maximum_for_octets(const uint8_t octets) {
8 if (octets == 0U) {
9 return UINT64_C(0);
10 }
11 if (octets >= 8U) {
12 return UINT64_MAX;
13 }
14 return (UINT64_C(1) << ((uint64_t)octets * UINT64_C(8))) - UINT64_C(1);
15}

17static uint8_t cuc_pfield(const ccsds_cuc_config_t *config) {
18 const uint8_t epoch_bits =
19 config->epoch == CCSDS_CUC_EPOCH_CCSDS_1958_TAI ? 0x10U : 0x20U;
20 return (uint8_t)(epoch_bits
21 | (uint8_t)((config->coarse_octets - 1U) << 2U)
22 | config->fine_octets);
23}

25ccsds_status_t ccsds_cuc_validate(const ccsds_cuc_config_t *config) {
26 if (config == NULL) {
27 return CCSDS_STATUS_NULL_POINTER;
28 }
29 if (config->epoch != CCSDS_CUC_EPOCH_CCSDS_1958_TAI
30 && config->epoch != CCSDS_CUC_EPOCH_AGENCY_DEFINED) {
31 return CCSDS_STATUS_CUC_INVALID_EPOCH;
32 }
33 if (config->pfield_mode != CCSDS_CUC_PFIELD_IMPLICIT
34 && config->pfield_mode != CCSDS_CUC_PFIELD_EXPLICIT) {
35 return CCSDS_STATUS_CUC_INVALID_PFIELD_MODE;
36 }
37 if (config->coarse_octets < 1U || config->coarse_octets > 4U) {
38 return CCSDS_STATUS_CUC_INVALID_COARSE_WIDTH;
39 }
40 if (config->fine_octets > 3U) {
41 return CCSDS_STATUS_CUC_INVALID_FINE_WIDTH;
42 }
43 return CCSDS_STATUS_OK;
44}

46size_t ccsds_cuc_encoded_size(const ccsds_cuc_config_t *config) {
47 if (config == NULL) {
48 return 0U;
49 }
50 return (size_t)config->coarse_octets
51 + (size_t)config->fine_octets
52 + (config->pfield_mode == CCSDS_CUC_PFIELD_EXPLICIT ? 1U : 0U);
53}

55ccsds_status_t ccsds_cuc_encode(const ccsds_cuc_time_t *value,
56 const ccsds_cuc_config_t *config,
57 uint8_t *output,
58 const size_t capacity,
59 size_t *written_out) {
60 size_t offset = 0U;
61 size_t required;
62 ccsds_status_t status;
63
64 if (value == NULL || config == NULL || written_out == NULL) {
65 return CCSDS_STATUS_NULL_POINTER;
66 }
67
68 status = ccsds_cuc_validate(config);
69 if (status != CCSDS_STATUS_OK) {
70 return status;
71 }
72 if (value->coarse > maximum_for_octets(config->coarse_octets)) {
73 return CCSDS_STATUS_CUC_COARSE_OVERFLOW;
74 }
75 if (value->fine > maximum_for_octets(config->fine_octets)) {
76 return CCSDS_STATUS_CUC_FINE_OVERFLOW;
77 }
78
79 required = ccsds_cuc_encoded_size(config);
80 if (capacity < required) {
81 return CCSDS_STATUS_BUFFER_TOO_SMALL;
82 }
83 if (output == NULL) {
84 return CCSDS_STATUS_NULL_POINTER;
85 }
86
87 if (config->pfield_mode == CCSDS_CUC_PFIELD_EXPLICIT) {
88 output[offset++] = cuc_pfield(config);
89 }
90
91 ccsds_store_be_uint(output + offset, config->coarse_octets, value->coarse);
92 offset += (size_t)config->coarse_octets;
93 if (config->fine_octets != 0U) {
94 ccsds_store_be_uint(output + offset, config->fine_octets, value->fine);
95 offset += (size_t)config->fine_octets;
96 }
97
98 *written_out = offset;
99 return CCSDS_STATUS_OK;
100}

102ccsds_status_t ccsds_cuc_decode(const uint8_t *data,
103 const size_t size,
104 const ccsds_cuc_config_t *config,
105 ccsds_cuc_time_t *value_out) {
106 size_t offset = 0U;
107 const size_t expected = ccsds_cuc_encoded_size(config);
108 ccsds_status_t status;
109
110 if (config == NULL || value_out == NULL) {
111 return CCSDS_STATUS_NULL_POINTER;
112 }
113
114 status = ccsds_cuc_validate(config);
115 if (status != CCSDS_STATUS_OK) {
116 return status;
117 }
118 if (size != expected) {
119 return CCSDS_STATUS_CUC_SIZE_MISMATCH;
120 }
121 if (data == NULL) {
122 return CCSDS_STATUS_NULL_POINTER;
123 }
124
125 if (config->pfield_mode == CCSDS_CUC_PFIELD_EXPLICIT) {
126 if (data[0] != cuc_pfield(config)) {
127 return CCSDS_STATUS_CUC_PFIELD_MISMATCH;
128 }
129 offset = 1U;
130 }
131
132 value_out->coarse = ccsds_load_be_uint(data + offset, config->coarse_octets);
133 offset += (size_t)config->coarse_octets;
134 value_out->fine = config->fine_octets == 0U
135 ? UINT64_C(0)
136 : ccsds_load_be_uint(data + offset, config->fine_octets);
137
138 return CCSDS_STATUS_OK;
139}
